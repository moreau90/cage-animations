  function clamp(x,a,b){ return Math.max(a, Math.min(b,x)); }

  // Smooth a cyclic array with a 3-point weighted average [0.25, 0.5, 0.25].
  // Wraps around at boundaries for seamless looping animations.
  // Multiple passes approximate a Gaussian filter.
  function smoothArray(arr, passes = 3) {
    const n = arr.length;
    let result = arr.slice();
    for (let p = 0; p < passes; p++) {
      const tmp = result.slice();
      for (let i = 0; i < n; i++) {
        result[i] = 0.25 * tmp[(i-1+n)%n] + 0.5 * tmp[i] + 0.25 * tmp[(i+1)%n];
      }
    }
    return result;
  }

  function rotateAroundAxis(px, py, pz, ox, oy, oz, ax, ay, az, angle) {
    let x = px - ox, y = py - oy, z = pz - oz;
    const c = Math.cos(angle), s = Math.sin(angle), t = 1 - c;
    const len = Math.hypot(ax, ay, az) || 1;
    ax/=len; ay/=len; az/=len;

    const nx = (t*ax*ax + c)*x + (t*ax*ay - s*az)*y + (t*ax*az + s*ay)*z;
    const ny = (t*ax*ay + s*az)*x + (t*ay*ay + c)*y + (t*ay*az - s*ax)*z;
    const nz = (t*ax*az - s*ay)*x + (t*ay*az + s*ax)*y + (t*az*az + c)*z;
    return [nx + ox, ny + oy, nz + oz];
  }

  // Compute the rotation that maps fromDir→toDir, then apply it to applyToDir.
  // Used for animation retargeting: extracts the "delta rotation" from the FBX
  // animation and applies it to the cage rest direction, preserving cage proportions.
  function applyDeltaRotation(fromDir, toDir, applyToDir) {
    let ax = vcross(fromDir, toDir);
    const axLen = vlen(ax);
    const d = Math.max(-1, Math.min(1, vdot(fromDir, toDir)));
    const ang = Math.acos(d);
    if (Math.abs(ang) < 1e-4) return applyToDir; // near-identical: no rotation
    if (axLen < 1e-6) {
      // Near-180° rotation: cross product degenerate, pick any perpendicular axis
      ax = Math.abs(fromDir[1]) < 0.9
        ? vcross(fromDir, [0, 1, 0])
        : vcross(fromDir, [1, 0, 0]);
      if (vlen(ax) < 1e-6) return applyToDir; // extreme degenerate, bail
    }
    ax = vnorm(ax);
    const r = rotateAroundAxis(applyToDir[0], applyToDir[1], applyToDir[2],
      0, 0, 0, ax[0], ax[1], ax[2], ang);
    return vnorm(r);
  }

  // Compute axis-angle rotation MATRIX from restDir to targetDir.
  // Returns a 3x3 rotation matrix (flat array). Used for analytical limb composition.
  function computeIKRotMatrix(restDir, targetDir) {
    let ax = vcross(restDir, targetDir);
    const axLen = vlen(ax);
    const d = Math.max(-1, Math.min(1, vdot(restDir, targetDir)));
    const ang = Math.acos(d);
    if (Math.abs(ang) < 1e-4) return [1,0,0, 0,1,0, 0,0,1]; // identity
    if (axLen < 1e-6) {
      // Near-180° rotation: pick any perpendicular axis
      ax = Math.abs(restDir[1]) < 0.9
        ? vcross(restDir, [0, 1, 0])
        : vcross(restDir, [1, 0, 0]);
      if (vlen(ax) < 1e-6) return [1,0,0, 0,1,0, 0,0,1];
    }
    ax = vnorm(ax);
    return mat3_rotAxis(ax[0], ax[1], ax[2], ang);
  }

  function vnorm(a){
    const L = vlen(a);
    return (L > 1e-12) ? [a[0]/L, a[1]/L, a[2]/L] : [1,0,0];
  }

  // Find shoulder joint center as average midpoint of K closest arm↔torso cage pairs.
  // Returns { jointCenter, armAnchor, shoulderCi } where jointCenter is the driving pivot.
  function findShoulderAnchorFromTorso(armIndices, torsoIndices, cageRest, K_PAIRS){
    K_PAIRS = K_PAIRS || 5;
    // Collect all arm↔torso distances
    const pairs = [];
    for (const ai of armIndices) {
      const p = cageRest[ai];
      for (const ti of torsoIndices) {
        const q = cageRest[ti];
        const dx = p[0]-q[0], dy = p[1]-q[1], dz = p[2]-q[2];
        pairs.push({ ai, ti, d2: dx*dx + dy*dy + dz*dz });
      }
    }
    pairs.sort((a,b) => a.d2 - b.d2);

    // Average the midpoints of the K closest pairs
    const N = Math.min(K_PAIRS, pairs.length);
    let mx=0, my=0, mz=0, ax=0, ay=0, az=0;
    for (let i = 0; i < N; i++) {
      const ap = cageRest[pairs[i].ai], tp = cageRest[pairs[i].ti];
      mx += (ap[0]+tp[0])*0.5;  my += (ap[1]+tp[1])*0.5;  mz += (ap[2]+tp[2])*0.5;
      ax += ap[0]; ay += ap[1]; az += ap[2];
    }
    const jointCenter = [mx/N, my/N, mz/N];
    const armAnchor = [ax/N, ay/N, az/N];
    const shoulderCi = pairs[0] ? pairs[0].ai : (armIndices[0] ?? -1);
    return { shoulder: jointCenter, armAnchor, shoulderCi, d2: pairs[0]?.d2 || 0 };
  }

  // Farthest arm cage point from shoulder (to define arm direction).
  function farthestPointFrom(armIndices, origin, cageRest){
    let bestCi = armIndices[0] ?? -1;
    let bestD2 = -1;
    for (const ci of armIndices) {
      const p = cageRest[ci];
      const dx = p[0]-origin[0], dy = p[1]-origin[1], dz = p[2]-origin[2];
      const d2 = dx*dx + dy*dy + dz*dz;
      if (d2 > bestD2) { bestD2 = d2; bestCi = ci; }
    }
    return cageRest[bestCi] ? [...cageRest[bestCi]] : [origin[0]+1, origin[1], origin[2]];
  }

  // Split arm cage verts by projection along arm axis into upper/forearm/hand.
  function splitArmByProjection(allArmCage, shoulder, dir, upperArmLen, forearmLen, ridUpper, ridFore, ridHand, regionId, cageRest){
    const upper=[], fore=[], hand=[];
    for (const ci of allArmCage) {
      const p = cageRest[ci];
      const t = (p[0]-shoulder[0])*dir[0] + (p[1]-shoulder[1])*dir[1] + (p[2]-shoulder[2])*dir[2];
      if (t <= upperArmLen) { upper.push(ci); regionId[ci]=ridUpper; }
      else if (t <= upperArmLen + forearmLen) { fore.push(ci); regionId[ci]=ridFore; }
      else { hand.push(ci); regionId[ci]=ridHand; }
    }
    return { upper, fore, hand };
  }

  // Projection-based arm region split using cage-derived shoulder anchors.
  // Splits by projection along the arm axis (shoulder → farthest point),
  // not radial distance. This is stable and mirror-consistent.
  function resplitArmRegions() {
    if (!cageRest || !allLeftArmCage.length || !allRightArmCage.length) return;

    const upperArmLen = +$('upperArmLen').value;
    const forearmLen  = +$('forearmLen').value;

    // Torso cage indices (region 0)
    const torsoIdx = regionIndices[0] || indicesByRegion(0);
    if (!torsoIdx.length) return;

    // Left: shoulder anchor + arm axis
    const L = findShoulderAnchorFromTorso(allLeftArmCage, torsoIdx, cageRest);
    const lFar = farthestPointFrom(allLeftArmCage, L.shoulder, cageRest);
    let lDir = vnorm(vsub(lFar, L.shoulder));

    // Right: shoulder anchor + arm axis
    const Rr = findShoulderAnchorFromTorso(allRightArmCage, torsoIdx, cageRest);
    const rFar = farthestPointFrom(allRightArmCage, Rr.shoulder, cageRest);
    let rDir = vnorm(vsub(rFar, Rr.shoulder));

    // Enforce mirror-consistency: left dir x<0, right dir x>0
    if (lDir[0] > 0) lDir = vscale(lDir, -1);
    if (rDir[0] < 0) rDir = vscale(rDir, -1);

    // Store cage-derived anchors globally so arm driving uses the same pivots
    // Symmetrize L/R to eliminate cage mesh asymmetry artifacts
    {
      const avgShX = (Math.abs(L.shoulder[0]) + Math.abs(Rr.shoulder[0])) / 2;
      const avgShY = (L.shoulder[1] + Rr.shoulder[1]) / 2;
      const avgShZ = (L.shoulder[2] + Rr.shoulder[2]) / 2;
      cageShoulderL = [-avgShX, avgShY, avgShZ];
      cageShoulderR = [ avgShX, avgShY, avgShZ];

      const avgDX = (Math.abs(lDir[0]) + Math.abs(rDir[0])) / 2;
      const avgDY = (lDir[1] + rDir[1]) / 2;
      const avgDZ = (lDir[2] + rDir[2]) / 2;
      cageArmDirL = vnorm([-avgDX, avgDY, avgDZ]);
      cageArmDirR = vnorm([ avgDX, avgDY, avgDZ]);
    }

    // Split by projection along arm axis
    const lSplit = splitArmByProjection(allLeftArmCage,  L.shoulder, lDir, upperArmLen, forearmLen, 7, 9, 12, regionId, cageRest);
    const rSplit = splitArmByProjection(allRightArmCage, Rr.shoulder, rDir, upperArmLen, forearmLen, 8, 10, 13, regionId, cageRest);

    R.l_upperarm = lSplit.upper;
    R.l_forearm  = lSplit.fore;

    R.r_upperarm = rSplit.upper;

    regionIndices[8]  = R.r_upperarm;
    regionIndices[9]  = R.l_forearm;
    regionIndices[10] = R.r_forearm;

    // Rebuild mesh bindings and region weights with updated regionId
    const built = buildBindKNNRegion(meshRestPos, cageRest, regionId, null, 8);

  // Rotate a direction vector around an axis (no origin offset)
  function rotVec3(vx, vy, vz, ax, ay, az, angle) {
    return rotateAroundAxis(vx, vy, vz, 0, 0, 0, ax, ay, az, angle);
  }

  // Apply rotation around an ARBITRARY axis to cage vertices
  function applyRotAxis(indices, pivot, ax, ay, az, ang) {
    const [px,py,pz] = pivot;
    for (const i of indices) {
      let x=cagePos[i*3], y=cagePos[i*3+1], z=cagePos[i*3+2];
      [x,y,z] = rotateAroundAxis(x,y,z, px,py,pz, ax,ay,az, ang);
      cagePos[i*3]=x; cagePos[i*3+1]=y; cagePos[i*3+2]=z;
    }
  }

  // Gradient rotation for joint-proximal cage verts (e.g. thigh near hip).
  // Verts at the pivot end (frac≈0) get MIN_FRAC of the angle.
  // Verts at the far end (frac≈1) get full angle.
  // This prevents inner-thigh cage verts from swinging as hard as knee-level verts,
  // reducing the crotch gap artifact.
  function applyRotAxisGradient(indices, pivot, ax, ay, az, ang, boneDir, boneLen) {
    const [px, py, pz] = pivot;
    const [dx, dy, dz] = boneDir;
    const MIN_FRAC = 0.1; // hip-proximal verts get 10% of rotation
    for (const i of indices) {
      // Project vert offset onto bone direction to get frac along bone
      const vx = cagePos[i*3] - px;
      const vy = cagePos[i*3+1] - py;
      const vz = cagePos[i*3+2] - pz;
      const proj = vx*dx + vy*dy + vz*dz;
      let frac = proj / boneLen;
      frac = Math.max(0, Math.min(1, frac));
      // Linear gradient: MIN_FRAC at hip, 1.0 at knee
      const gradFrac = MIN_FRAC + (1.0 - MIN_FRAC) * frac;
      const vertAng = ang * gradFrac;
      if (Math.abs(vertAng) > 1e-6) {
        let x = cagePos[i*3], y = cagePos[i*3+1], z = cagePos[i*3+2];
        [x, y, z] = rotateAroundAxis(x, y, z, px, py, pz, ax, ay, az, vertAng);
        cagePos[i*3] = x; cagePos[i*3+1] = y; cagePos[i*3+2] = z;
      }
    }
  }

  function applyRotX(indices, pivot, ang){
    const [px,py,pz]=pivot;
    for(const i of indices){
      let x=cagePos[i*3], y=cagePos[i*3+1], z=cagePos[i*3+2];
      [x,y,z] = rotateAroundAxis(x,y,z, px,py,pz, 1,0,0, ang);
      cagePos[i*3]=x; cagePos[i*3+1]=y; cagePos[i*3+2]=z;

  function applyRotY(indices, pivot, ang){
    const [px,py,pz]=pivot;
    for(const i of indices){

  function applyRotZ(indices, pivot, ang){
    const [px,py,pz]=pivot;
    for(const i of indices){

  function translate(indices, dx,dy,dz){
    for(const i of indices){
      cagePos[i*3]+=dx; cagePos[i*3+1]+=dy; cagePos[i*3+2]+=dz;

  function regionCenter(indices){
    let sx=0,sy=0,sz=0;
    const n=Math.max(1, indices.length);
    for(const i of indices){
      sx+=cagePos[i*3]; sy+=cagePos[i*3+1]; sz+=cagePos[i*3+2];
    }
    return [sx/n, sy/n, sz/n];
  }

  // ---------------- Quaternion Math ([x, y, z, w] format) ----------------
  function quat_mul(a, b) {
    return [
      a[3]*b[0] + a[0]*b[3] + a[1]*b[2] - a[2]*b[1],
      a[3]*b[1] - a[0]*b[2] + a[1]*b[3] + a[2]*b[0],
      a[3]*b[2] + a[0]*b[1] - a[1]*b[0] + a[2]*b[3],
      a[3]*b[3] - a[0]*b[0] - a[1]*b[1] - a[2]*b[2]
    ];
  }
  function quat_conjugate(q) { return [-q[0], -q[1], -q[2], q[3]]; }

  function quat_slerp(a, b, t) {
    // Shortest-path: flip b if dot < 0
    let dot = a[0]*b[0] + a[1]*b[1] + a[2]*b[2] + a[3]*b[3];
    const b2 = (dot < 0) ? [-b[0], -b[1], -b[2], -b[3]] : b;
    dot = Math.abs(dot);
    if (dot > 0.9995) {
      // Nearly identical — lerp + normalize
      const r = [
        a[0] + (b2[0] - a[0]) * t,
        a[1] + (b2[1] - a[1]) * t,
        a[2] + (b2[2] - a[2]) * t,
        a[3] + (b2[3] - a[3]) * t
      ];
      const inv = 1 / Math.hypot(r[0], r[1], r[2], r[3]);
      return [r[0]*inv, r[1]*inv, r[2]*inv, r[3]*inv];
    }
    const theta = Math.acos(dot);
    const sinTheta = Math.sin(theta);
    const wa = Math.sin((1 - t) * theta) / sinTheta;
    const wb = Math.sin(t * theta) / sinTheta;
    return [
      wa*a[0] + wb*b2[0], wa*a[1] + wb*b2[1],
      wa*a[2] + wb*b2[2], wa*a[3] + wb*b2[3]
    ];
  }

  // Swing-twist decomposition: given a delta quaternion and a twist axis,
  // extract the signed twist angle (rotation around that axis).
  function swingTwistDecompose(qDelta, twistAxis) {
    // Project quaternion imaginary part onto twist axis
    const proj = qDelta[0]*twistAxis[0] + qDelta[1]*twistAxis[1] + qDelta[2]*twistAxis[2];
    // Twist quaternion = normalize([proj*ax, proj*ay, proj*az, w])
    let tx = proj * twistAxis[0], ty = proj * twistAxis[1], tz = proj * twistAxis[2];
    const tw = qDelta[3];
    const len = Math.hypot(tx, ty, tz, tw);
    if (len < 1e-10) return 0;
    tx /= len; ty /= len; tz /= len;
    const twn = tw / len;
    // Signed angle: 2 * atan2(|imaginary|, w), with sign from projection direction
    const sinHalf = Math.hypot(tx, ty, tz);
    let angle = 2 * Math.atan2(sinHalf, Math.abs(twn));
    // Sign: if proj is negative, twist is in opposite direction
    if (proj < 0) angle = -angle;
    // Handle w < 0 (angle > 180°)
    if (twn < 0) angle = (angle > 0) ? angle - 2*Math.PI : angle + 2*Math.PI;
    return angle;
  }

  // Convenience: extract twist angle of a bone between rest and current quaternions
  // along the rest-pose bone direction.
  // Uses POSITION-SWING RESIDUAL method: removes the shortest-arc rotation that
  // accounts for the direction change (what positions already capture), leaving
  // only the bone-axis twist. This is CHARACTER-INDEPENDENT — different bind poses
  // (T-pose, A-pose, etc.) produce the same twist angle for the same animation.
  function extractQuatTwist(restQ, curQ, boneDir) {
    let qDelta = quat_mul(curQ, quat_conjugate(restQ));
    // Ensure shortest-path quaternion (avoid >180° wraparound like -349°)
    if (qDelta[3] < 0) qDelta = [-qDelta[0], -qDelta[1], -qDelta[2], -qDelta[3]];
    // Position-swing residual: remove positional swing, keep only twist
    const d_cur = quat_rotate_vec(qDelta, boneDir);
    const q_swing = shortest_arc_quat(boneDir, d_cur);
    const q_residual = quat_mul(quat_conjugate(q_swing), qDelta);
    return swingTwistDecompose(q_residual, boneDir);
  }

  // Rotate a 3D vector by a unit quaternion: q * [v,0] * conj(q)
  function quat_rotate_vec(q, v) {
    const qx = q[0], qy = q[1], qz = q[2], qw = q[3];
    const vx = v[0], vy = v[1], vz = v[2];
    const cx = qy*vz - qz*vy, cy = qz*vx - qx*vz, cz = qx*vy - qy*vx;
    const c2x = qy*cz - qz*cy, c2y = qz*cx - qx*cz, c2z = qx*cy - qy*cx;
    return [vx + 2*(qw*cx + c2x), vy + 2*(qw*cy + c2y), vz + 2*(qw*cz + c2z)];
  }

  // Shortest-arc unit quaternion rotating unit vector a to unit vector b.
  // Used by position-swing residual to compute the "positional" swing.
  function shortest_arc_quat(a, b) {
    const d = a[0]*b[0] + a[1]*b[1] + a[2]*b[2];
    if (d > 0.999999) return [0, 0, 0, 1]; // Same direction
    if (d < -0.999999) {
      // Opposite — find any perpendicular axis
      let perp = vcross(a, [1, 0, 0]);
      if (vlen(perp) < 1e-6) perp = vcross(a, [0, 1, 0]);
      perp = vnorm(perp);
      return [perp[0], perp[1], perp[2], 0]; // 180° rotation
    }
    const c = vcross(a, b);
    const w = 1 + d;
    const inv = 1 / Math.hypot(c[0], c[1], c[2], w);
    return [c[0]*inv, c[1]*inv, c[2]*inv, w*inv];
  }

  // Extract LOCAL twist of a bone relative to its parent.
  // Unlike getJointTwistAtTime (world twist), this removes the parent chain's
  // contribution — isolating only the bone's own axial rotation.
  // Used for forearm pronation/supination (our analytical composition already
  // propagates upper arm twist, so we must not double-count it).
  function getLocalJointTwistAtTime(boneName, parentName, childName, t) {
    const boneQ = getJointQuaternionAtTime(boneName, t);
    const parentQ = getJointQuaternionAtTime(parentName, t);
    if (!boneQ || !parentQ) return 0;
    const restBoneQ = jointPosData.rest_quats[boneName];
    const restParentQ = jointPosData.rest_quats[parentName];
    if (!restBoneQ || !restParentQ) return 0;
    // Local quaternion = conj(parent_world) * bone_world
    const localCurr = quat_mul(quat_conjugate(parentQ), boneQ);
    const localRest = quat_mul(quat_conjugate(restParentQ), restBoneQ);
    // Local delta (ensure shortest-path to avoid >180° wraparound)
    let localDelta = quat_mul(localCurr, quat_conjugate(localRest));
    if (localDelta[3] < 0) localDelta = [-localDelta[0], -localDelta[1], -localDelta[2], -localDelta[3]];
    // Bone direction in parent's rest local frame
    const rp = jointPosData.rest_pose;
    if (!rp[boneName] || !rp[childName]) return 0;
    const worldBoneDir = vnorm(vsub(rp[childName], rp[boneName]));
    const localBoneDir = quat_rotate_vec(quat_conjugate(restParentQ), worldBoneDir);
    // Position-swing residual (character-independent)
    const d_cur_local = quat_rotate_vec(localDelta, localBoneDir);
    const q_swing = shortest_arc_quat(localBoneDir, d_cur_local);
    const q_residual = quat_mul(quat_conjugate(q_swing), localDelta);
    return swingTwistDecompose(q_residual, localBoneDir);
  }

  // ---------------- 3x3 Matrix Math (row-major Float64Array[9]) ----------------
  function mat3_create() { return new Float64Array(9); }
  function mat3_identity() { const m = mat3_create(); m[0]=m[4]=m[8]=1; return m; }

  function mat3_rotX(angle) {
    const c = Math.cos(angle), s = Math.sin(angle), m = mat3_create();
    m[0]=1; m[4]=c; m[5]=-s; m[7]=s; m[8]=c;
    return m;
  }

  function mat3_rotY(angle) {
    const c = Math.cos(angle), s = Math.sin(angle), m = mat3_create();
    m[0]=c; m[2]=s; m[4]=1; m[6]=-s; m[8]=c;
    return m;
  }

  function mat3_rotZ(angle) {
    const c = Math.cos(angle), s = Math.sin(angle), m = mat3_create();
    m[0]=c; m[1]=-s; m[3]=s; m[4]=c; m[8]=1;
    return m;
  }

  // Rotation matrix around an arbitrary unit axis (Rodrigues)
  function mat3_rotAxis(ax, ay, az, angle) {
    const c = Math.cos(angle), s = Math.sin(angle), t = 1-c;
    const len = Math.hypot(ax,ay,az) || 1;
    ax/=len; ay/=len; az/=len;
    return [
      t*ax*ax+c,    t*ax*ay-s*az, t*ax*az+s*ay,
      t*ax*ay+s*az, t*ay*ay+c,    t*ay*az-s*ax,
      t*ax*az-s*ay, t*ay*az+s*ax, t*az*az+c
    ];
  }

  function mat3_mul(A, B) {
    const C = mat3_create();
    for (let i = 0; i < 3; i++)
      for (let j = 0; j < 3; j++)
        C[i*3+j] = A[i*3]*B[j] + A[i*3+1]*B[3+j] + A[i*3+2]*B[6+j];
    return C;
  }

  function mat3_transpose(A) {
    const T = mat3_create();
    for (let i = 0; i < 3; i++)
      for (let j = 0; j < 3; j++)
        T[i*3+j] = A[j*3+i];
    return T;
  }

  function mat3_det(A) {
    return A[0]*(A[4]*A[8]-A[5]*A[7])
         - A[1]*(A[3]*A[8]-A[5]*A[6])
         + A[2]*(A[3]*A[7]-A[4]*A[6]);
  }

  function mat3_mulVec3(M, x, y, z) {
    return [
      M[0]*x + M[1]*y + M[2]*z,
      M[3]*x + M[4]*y + M[5]*z,
      M[6]*x + M[7]*y + M[8]*z
    ];
  }

  // Jacobi eigendecomposition for 3x3 symmetric matrix
  // Returns {eigenvalues: [3], V: Float64Array(9)} where columns of V are eigenvectors
  function symmetricEigen3x3(S) {
    const A = Float64Array.from(S);
    const V = mat3_identity();

    for (let iter = 0; iter < 30; iter++) {
      // Find largest off-diagonal element
      let maxVal = 0, p = 0, q = 1;
      for (let i = 0; i < 3; i++)
        for (let j = i+1; j < 3; j++)
          if (Math.abs(A[i*3+j]) > maxVal) { maxVal = Math.abs(A[i*3+j]); p = i; q = j; }

      if (maxVal < 1e-12) break;

      // Compute Jacobi rotation angle
      const app = A[p*3+p], aqq = A[q*3+q], apq = A[p*3+q];
      const theta = (Math.abs(app - aqq) < 1e-14)
        ? Math.PI / 4
        : 0.5 * Math.atan2(2 * apq, app - aqq);
      const c = Math.cos(theta), s = Math.sin(theta);

      // Right multiply: B = A * G(p,q,theta)
      const B = Float64Array.from(A);
      for (let i = 0; i < 3; i++) {
        B[i*3+p] =  c*A[i*3+p] + s*A[i*3+q];
        B[i*3+q] = -s*A[i*3+p] + c*A[i*3+q];
      }
      // Left multiply: A' = G^T * B
      for (let j = 0; j < 3; j++) {
        A[p*3+j] =  c*B[p*3+j] + s*B[q*3+j];
        A[q*3+j] = -s*B[p*3+j] + c*B[q*3+j];
      }

      // Accumulate eigenvectors: V = V * G
      for (let i = 0; i < 3; i++) {
        const vip = V[i*3+p], viq = V[i*3+q];
        V[i*3+p] =  c*vip + s*viq;
        V[i*3+q] = -s*vip + c*viq;
      }
    }

    return { eigenvalues: [A[0], A[4], A[8]], V };
  }

  // SVD for 3x3: H = U * diag(S) * V^T
  function svd3x3(H) {
    const Ht = mat3_transpose(H);
    const HtH = mat3_mul(Ht, H);

    const { eigenvalues, V } = symmetricEigen3x3(HtH);
    const sigma = eigenvalues.map(e => Math.sqrt(Math.max(0, e)));

    // U = H * V * diag(1/sigma)
    const HV = mat3_mul(H, V);
    const U = mat3_create();
    for (let j = 0; j < 3; j++) {
      const invS = sigma[j] > 1e-8 ? 1 / sigma[j] : 0;
      for (let i = 0; i < 3; i++) {
        U[i*3+j] = HV[i*3+j] * invS;
      }
    }

    return { U, S: sigma, V };
  }

  // Gram-Schmidt orthonormalize a 3x3 matrix (row-major) to ensure it's a proper rotation.
  // Returns a new Float64Array(9) with orthonormal rows forming a right-handed frame.
  function mat3_orthonormalize(M) {
    // Extract columns
    let c0 = [M[0], M[3], M[6]];
    let c1 = [M[1], M[4], M[7]];
    let c2 = [M[2], M[5], M[8]];

    // Normalize column 0
    let len = Math.hypot(c0[0], c0[1], c0[2]);
    if (len < 1e-12) { c0 = [1, 0, 0]; len = 1; }
    c0 = [c0[0]/len, c0[1]/len, c0[2]/len];

    // Column 1: subtract projection onto c0, normalize
    let dot = c1[0]*c0[0] + c1[1]*c0[1] + c1[2]*c0[2];
    c1 = [c1[0] - dot*c0[0], c1[1] - dot*c0[1], c1[2] - dot*c0[2]];
    len = Math.hypot(c1[0], c1[1], c1[2]);
    if (len < 1e-12) {
      // c1 was parallel to c0, pick an arbitrary perpendicular
      c1 = Math.abs(c0[0]) < 0.9 ? [1,0,0] : [0,1,0];
      dot = c1[0]*c0[0] + c1[1]*c0[1] + c1[2]*c0[2];
      c1 = [c1[0] - dot*c0[0], c1[1] - dot*c0[1], c1[2] - dot*c0[2]];
      len = Math.hypot(c1[0], c1[1], c1[2]);
    }
    c1 = [c1[0]/len, c1[1]/len, c1[2]/len];

    // Column 2: cross product of c0 and c1 (right-handed)
    c2 = [
      c0[1]*c1[2] - c0[2]*c1[1],
      c0[2]*c1[0] - c0[0]*c1[2],
      c0[0]*c1[1] - c0[1]*c1[0]
    ];

    const R = new Float64Array(9);
    R[0]=c0[0]; R[1]=c1[0]; R[2]=c2[0];
    R[3]=c0[1]; R[4]=c1[1]; R[5]=c2[1];
    R[6]=c0[2]; R[7]=c1[2]; R[8]=c2[2];
    return R;
  }

  // ---------------- Quaternion helpers for rotation blending ----------------
  // q = [x, y, z, w] (Hamilton convention)

  function mat3_to_quat(M) {
    // Convert 3x3 row-major rotation matrix to unit quaternion
    const m00=M[0], m01=M[1], m02=M[2];
    const m10=M[3], m11=M[4], m12=M[5];
    const m20=M[6], m21=M[7], m22=M[8];
    const tr = m00 + m11 + m22;
    let x, y, z, w;
    if (tr > 0) {
      const s = 0.5 / Math.sqrt(tr + 1.0);
      w = 0.25 / s;
      x = (m21 - m12) * s;
      y = (m02 - m20) * s;
      z = (m10 - m01) * s;
    } else if (m00 > m11 && m00 > m22) {
      const s = 2.0 * Math.sqrt(1.0 + m00 - m11 - m22);
      w = (m21 - m12) / s;
      x = 0.25 * s;
      y = (m01 + m10) / s;
      z = (m02 + m20) / s;
    } else if (m11 > m22) {
      const s = 2.0 * Math.sqrt(1.0 + m11 - m00 - m22);
      w = (m02 - m20) / s;
      x = (m01 + m10) / s;
      y = 0.25 * s;
      z = (m12 + m21) / s;
    } else {
      const s = 2.0 * Math.sqrt(1.0 + m22 - m00 - m11);
      w = (m10 - m01) / s;
      x = (m02 + m20) / s;
      y = (m12 + m21) / s;
      z = 0.25 * s;
    }
    const len = Math.hypot(x, y, z, w) || 1;
    return [x/len, y/len, z/len, w/len];
  }

  // ---------------- Dual Quaternion Skinning (DQS) ----------------
  // Encodes rigid transform (R, t) as dual quaternion [qr(4), qd(4)].
  // Blends rotation + translation together — no volume loss at joints.

  function rigid_to_dq(q, t) {
    // q = rotation quaternion [x,y,z,w], t = translation [tx,ty,tz]
    // dual part = 0.5 * (tx,ty,tz,0) * q_real (quaternion multiply)
    const qx=q[0], qy=q[1], qz=q[2], qw=q[3];
    const tx=t[0], ty=t[1], tz=t[2];
    return {
      qr: [qx, qy, qz, qw],
      qd: [
        0.5*( tx*qw + ty*qz - tz*qy),
        0.5*(-tx*qz + ty*qw + tz*qx),
        0.5*( tx*qy - ty*qx + tz*qw),
        0.5*(-tx*qx - ty*qy - tz*qz)
      ]
    };
  }

  function dq_apply(qr, qd, vx, vy, vz) {
    // Apply normalized dual quaternion to vertex.
    // Rotation: q * v * conj(q)
    const ax=qr[0], ay=qr[1], az=qr[2], aw=qr[3];
    const tx2 = 2*(ay*vz - az*vy);
    const ty2 = 2*(az*vx - ax*vz);
    const tz2 = 2*(ax*vy - ay*vx);
    const rx = vx + aw*tx2 + (ay*tz2 - az*ty2);
    const ry = vy + aw*ty2 + (az*tx2 - ax*tz2);
    const rz = vz + aw*tz2 + (ax*ty2 - ay*tx2);
    // Translation: t = 2 * (qr.w * qd.xyz - qd.w * qr.xyz + cross(qr.xyz, qd.xyz))
    const ttx = 2*( qd[0]*aw - qd[3]*ax + qd[2]*ay - qd[1]*az);
    const tty = 2*( qd[1]*aw - qd[3]*ay + qd[0]*az - qd[2]*ax);
    const ttz = 2*( qd[2]*aw - qd[3]*az + qd[1]*ax - qd[0]*ay);
    return [rx + ttx, ry + tty, rz + ttz];
  }

  function quat_rotateVec3(q, vx, vy, vz) {
    // Rotate vector by quaternion: q * v * q^-1
    const qx=q[0], qy=q[1], qz=q[2], qw=q[3];
    // t = 2 * cross(q.xyz, v)
    const tx = 2*(qy*vz - qz*vy);
    const ty = 2*(qz*vx - qx*vz);
    const tz = 2*(qx*vy - qy*vx);
    return [
      vx + qw*tx + (qy*tz - qz*ty),
      vy + qw*ty + (qz*tx - qx*tz),
      vz + qw*tz + (qx*ty - qy*tx)
    ];
  }

  // Kabsch algorithm: best-fit rigid transform (R, t) mapping rest → current
  // for the cage vertices at given indices.
  // Always produces a valid rotation — no fallbacks to identity.
  function computeRigidTransformKabsch(indices, regionId) {
    const n = indices.length;

    // Compute centroids
    let c0x=0, c0y=0, c0z=0, ctx=0, cty=0, ctz=0;
    for (const i of indices) {
      c0x += cageRest[i][0]; c0y += cageRest[i][1]; c0z += cageRest[i][2];
      ctx += cagePos[i*3];   cty += cagePos[i*3+1]; ctz += cagePos[i*3+2];
    }
    c0x/=n; c0y/=n; c0z/=n;
    ctx/=n; cty/=n; ctz/=n;

    // Build covariance H = Σ (rest - c0)(current - ct)^T
    const H = mat3_create();
    for (const i of indices) {
      const dx0 = cageRest[i][0]-c0x, dy0 = cageRest[i][1]-c0y, dz0 = cageRest[i][2]-c0z;
      const dxt = cagePos[i*3]-ctx,   dyt = cagePos[i*3+1]-cty, dzt = cagePos[i*3+2]-ctz;

      H[0] += dx0*dxt; H[1] += dx0*dyt; H[2] += dx0*dzt;
      H[3] += dy0*dxt; H[4] += dy0*dyt; H[5] += dy0*dzt;
      H[6] += dz0*dxt; H[7] += dz0*dyt; H[8] += dz0*dzt;
    }

    const { U, S, V } = svd3x3(H);

    // Fix degenerate SVD: when points are nearly coplanar, one singular value
    // is near zero, making that column of U all zeros (from invS=0 in svd3x3).
    // Reconstruct it as the cross product of the other two columns.
    const sMax = Math.max(S[0], S[1], S[2]);
    if (sMax > 1e-10) {
      let minI = 0;
      if (S[1] < S[minI]) minI = 1;
      if (S[2] < S[minI]) minI = 2;

      if (S[minI] < sMax * 1e-3) {
        const a = (minI + 1) % 3, b = (minI + 2) % 3;
        U[0*3+minI] = U[1*3+a]*U[2*3+b] - U[2*3+a]*U[1*3+b];
        U[1*3+minI] = U[2*3+a]*U[0*3+b] - U[0*3+a]*U[2*3+b];
        U[2*3+minI] = U[0*3+a]*U[1*3+b] - U[1*3+a]*U[0*3+b];
      }
    }

    // R = V * U^T
    const Ut = mat3_transpose(U);
    let R = mat3_mul(V, Ut);

    // Fix reflection: ensure det(R) > 0
    if (mat3_det(R) < 0) {
      const Vfix = Float64Array.from(V);
      let minI = 0;
      if (S[1] < S[minI]) minI = 1;
      if (S[2] < S[minI]) minI = 2;
      Vfix[0*3+minI] *= -1;
      Vfix[1*3+minI] *= -1;
      Vfix[2*3+minI] *= -1;
      R = mat3_mul(Vfix, Ut);
    }

    // Always orthonormalize R to guarantee a valid rotation matrix.
    // This handles any numerical drift from SVD without ever falling back to identity.
    R = mat3_orthonormalize(R);

    // t = centroid_current - R * centroid_rest
    const Rc0 = mat3_mulVec3(R, c0x, c0y, c0z);
    const t = [ctx-Rc0[0], cty-Rc0[1], ctz-Rc0[2]];

    return { R, t, cRest: [c0x, c0y, c0z], cCurr: [ctx, cty, ctz] };
  }

  // Compute world-space rigid transforms for all regions (independent Kabsch per region).
  // Each region's cage verts are already driven with full hierarchical rotation by
  // driveCageFromMixamo(), so world-space Kabsch recovers the composed transform directly.
  function computeAllRegionTransforms() {
    const transforms = {};
    const posMode = armState && armState.positionBased;
    const legAnalytical = legState && legState.positionBased;
    for (const rid in regionIndices) {
      const r = +rid;
      if (r === 5 || r === 6) continue;   // feet use shin transform
      // Arm regions ALWAYS use analytical composition (Kabsch unreliable on thin clusters)
      if (r === 7 || r === 8) continue;
      if (r === 9 || r === 10) continue;
      if (r === 12 || r === 13) continue;
      // Leg regions: use analytical in position mode (Kabsch gives bad results with
      // asymmetric cage + diagonal rotation axes from delta retargeting)
      if (legAnalytical && (r === 1 || r === 2)) continue;  // thighs
      // Shins: prefer Kabsch from cage verts (vert-level IK positions them correctly).
      // Analytical chain (R_kneeBend * R_thigh) diverges at extreme angles because
      // delta-retarget knee bend ≠ actual cage knee bend (16.5° error at 105° knee).
      // if (legAnalytical && (r === 3 || r === 4)) continue;  // DISABLED: use Kabsch instead
      if (headState && r === 11) continue;  // head analytical in position mode
      const indices = regionIndices[rid];
      if (indices && indices.length) {
        transforms[r] = computeRigidTransformKabsch(indices, r);
        if (r === 3 || r === 4) transforms[r]._source = 'kabsch';
      }
    }

    // ── ANALYTICAL LEG TRANSFORMS (position mode) ──
    // Uses stored body-rotated hip pivots (movedHipL/R) instead of Kabsch-derived
    // pivots. The torso Kabsch is approximate because the torso has non-rigid
    // deformation (different twist at top vs bottom). Using exact pivots eliminates
    // cage-mesh misalignment at the hip joint.
    if (legAnalytical && transforms[0]) {
      const R_t = transforms[0].R, t_t = transforms[0].t;
      const gc = lastGroundCorrection; // ground comp Y offset (applied to cage, propagate to mesh)

      // One-shot: compare Kabsch-derived vs stored pivot
      if (!window._pivotDiagDone) {
        window._pivotDiagDone = true;
        const hipL = P.l_hip;
        const hipT = mat3_mulVec3(R_t, hipL[0], hipL[1], hipL[2]);
        const kabschHip = [hipT[0]+t_t[0], hipT[1]+t_t[1], hipT[2]+t_t[2]];
        const mhL = legState.movedHipL || hipL;
        const storedHip = [mhL[0], mhL[1]+gc, mhL[2]];
        const diff = Math.sqrt((kabschHip[0]-storedHip[0])**2+(kabschHip[1]-storedHip[1])**2+(kabschHip[2]-storedHip[2])**2);
        console.log('=== PIVOT COMPARISON (Kabsch vs stored) ===');
        console.log('  L hip Kabsch: [' + kabschHip.map(v=>v.toFixed(4)).join(',') + ']');
        console.log('  L hip stored: [' + storedHip.map(v=>v.toFixed(4)).join(',') + ']');
        console.log('  displacement: ' + diff.toFixed(4) + ' (' + (diff*100).toFixed(1) + '% of mesh if ~1.0 scale)');
        if (armState && armState.movedShoulderL) {
          const shL = cageShoulderL;
          const shT = mat3_mulVec3(R_t, shL[0], shL[1], shL[2]);
          const kabschSh = [shT[0]+t_t[0], shT[1]+t_t[1], shT[2]+t_t[2]];
          const msL = armState.movedShoulderL;
          const storedSh = [msL[0], msL[1]+gc, msL[2]];
          const shDiff = Math.sqrt((kabschSh[0]-storedSh[0])**2+(kabschSh[1]-storedSh[1])**2+(kabschSh[2]-storedSh[2])**2);
          console.log('  L shoulder Kabsch: [' + kabschSh.map(v=>v.toFixed(4)).join(',') + ']');
          console.log('  L shoulder stored: [' + storedSh.map(v=>v.toFixed(4)).join(',') + ']');
          console.log('  displacement: ' + shDiff.toFixed(4));
        }
        // Shin transform source diagnostic (deferred to after transforms[1] exists)
        if (transforms[3]) {
          console.log('  L shin transform source: ' + (transforms[3]._source || 'analytical'));
        }
      }

      // Left thigh: R_thigh = legIK directly (NOT composed with R_t).
      // R_thigh already includes body lean via bodyRotDir in the rest direction.
      // Hip pivot from stored body-rotated position + ground correction.
      {
        const R_thigh = legState.lThighR;
        const R_noTwist = legState.lSwingOnlyR;
        const hip = P.l_hip;
        // Use exact body-rotated hip pivot (matches cage vert-level path)
        const mh = legState.movedHipL || hip;
        const hipWorld = [mh[0], mh[1] + gc, mh[2]];
        const Rth_hip = mat3_mulVec3(R_thigh, hip[0], hip[1], hip[2]);
        const Rnt_hip = mat3_mulVec3(R_noTwist, hip[0], hip[1], hip[2]);
        const lkn = P.l_knee;
        const thighVec = [lkn[0]-hip[0], lkn[1]-hip[1], lkn[2]-hip[2]];
        const thighLen = Math.sqrt(thighVec[0]**2 + thighVec[1]**2 + thighVec[2]**2);
        transforms[1] = {
          R: R_thigh,
          t: [hipWorld[0]-Rth_hip[0], hipWorld[1]-Rth_hip[1], hipWorld[2]-Rth_hip[2]],
          R_noTwist: R_noTwist,
          t_noTwist: [hipWorld[0]-Rnt_hip[0], hipWorld[1]-Rnt_hip[1], hipWorld[2]-Rnt_hip[2]],
          _hip: hip, _thighDir: [thighVec[0]/thighLen, thighVec[1]/thighLen, thighVec[2]/thighLen], _thighLen: thighLen,
          cRest: transforms[0].cRest, cCurr: transforms[0].cCurr
        };

        // Left shin: prefer Kabsch (computed above). Analytical fallback only if Kabsch failed.
        if (!transforms[3]) {
          const Rb = mat3_rotAxis(legState.lKneeBendAxis[0], legState.lKneeBendAxis[1], legState.lKneeBendAxis[2], legState.lKneeBendAngle);
          const R_shin = mat3_mul(Rb, R_thigh);
          const t_thigh = transforms[1].t;
          const knee = P.l_knee;
          const kneeT = mat3_mulVec3(R_thigh, knee[0], knee[1], knee[2]);
          const kneeWorld = [kneeT[0]+t_thigh[0], kneeT[1]+t_thigh[1], kneeT[2]+t_thigh[2]];
          const Rsh_knee = mat3_mulVec3(R_shin, knee[0], knee[1], knee[2]);
          transforms[3] = {
            R: R_shin,
            t: [kneeWorld[0]-Rsh_knee[0], kneeWorld[1]-Rsh_knee[1], kneeWorld[2]-Rsh_knee[2]],
            cRest: transforms[0].cRest, cCurr: transforms[0].cCurr
          };
        }
      }

      // Right thigh: same pattern with stored pivot
      {
        const R_thigh = legState.rThighR;
        const R_noTwist = legState.rSwingOnlyR;
        const hip = P.r_hip;
        const mh = legState.movedHipR || hip;
        const hipWorld = [mh[0], mh[1] + gc, mh[2]];
        const Rth_hip = mat3_mulVec3(R_thigh, hip[0], hip[1], hip[2]);
        const Rnt_hip = mat3_mulVec3(R_noTwist, hip[0], hip[1], hip[2]);
        const rkn = P.r_knee;
        const thighVec = [rkn[0]-hip[0], rkn[1]-hip[1], rkn[2]-hip[2]];
        const thighLen = Math.sqrt(thighVec[0]**2 + thighVec[1]**2 + thighVec[2]**2);
        transforms[2] = {
          R: R_thigh,
          t: [hipWorld[0]-Rth_hip[0], hipWorld[1]-Rth_hip[1], hipWorld[2]-Rth_hip[2]],
          R_noTwist: R_noTwist,
          t_noTwist: [hipWorld[0]-Rnt_hip[0], hipWorld[1]-Rnt_hip[1], hipWorld[2]-Rnt_hip[2]],
          _hip: hip, _thighDir: [thighVec[0]/thighLen, thighVec[1]/thighLen, thighVec[2]/thighLen], _thighLen: thighLen,
          cRest: transforms[0].cRest, cCurr: transforms[0].cCurr
        };

        // Right shin: prefer Kabsch (computed above). Analytical fallback only if Kabsch failed.
        if (!transforms[4]) {
          const Rb = mat3_rotAxis(legState.rKneeBendAxis[0], legState.rKneeBendAxis[1], legState.rKneeBendAxis[2], legState.rKneeBendAngle);
          const R_shin = mat3_mul(Rb, R_thigh);
          const t_thigh = transforms[2].t;
          const knee = P.r_knee;
          const kneeT = mat3_mulVec3(R_thigh, knee[0], knee[1], knee[2]);
          const kneeWorld = [kneeT[0]+t_thigh[0], kneeT[1]+t_thigh[1], kneeT[2]+t_thigh[2]];
          const Rsh_knee = mat3_mulVec3(R_shin, knee[0], knee[1], knee[2]);
          transforms[4] = {
            R: R_shin,
            t: [kneeWorld[0]-Rsh_knee[0], kneeWorld[1]-Rsh_knee[1], kneeWorld[2]-Rsh_knee[2]],
            cRest: transforms[0].cRest, cCurr: transforms[0].cCurr
          };
        }
      }
    }

    // Feet: analytical ankle rotation (position mode with toe data) or copy shin
    if (legAnalytical && legState.lAnkleBendAngle > 1e-4 && transforms[3]) {
      const Ra = mat3_rotAxis(legState.lAnkleBendAxis[0], legState.lAnkleBendAxis[1], legState.lAnkleBendAxis[2], legState.lAnkleBendAngle);
      const R_foot = mat3_mul(Ra, transforms[3].R);
      const ankle = P.l_ankle;
      const ankleT = mat3_mulVec3(transforms[3].R, ankle[0], ankle[1], ankle[2]);
      const ankleWorld = [ankleT[0]+transforms[3].t[0], ankleT[1]+transforms[3].t[1], ankleT[2]+transforms[3].t[2]];
      const Rf_ankle = mat3_mulVec3(R_foot, ankle[0], ankle[1], ankle[2]);
      transforms[5] = {
        R: R_foot,
        t: [ankleWorld[0]-Rf_ankle[0], ankleWorld[1]-Rf_ankle[1], ankleWorld[2]-Rf_ankle[2]],
        cRest: transforms[0].cRest, cCurr: transforms[0].cCurr
      };
    } else if (transforms[3]) {
      transforms[5] = transforms[3];
    }
    if (legAnalytical && legState.rAnkleBendAngle > 1e-4 && transforms[4]) {
      const Ra = mat3_rotAxis(legState.rAnkleBendAxis[0], legState.rAnkleBendAxis[1], legState.rAnkleBendAxis[2], legState.rAnkleBendAngle);
      const R_foot = mat3_mul(Ra, transforms[4].R);
      const ankle = P.r_ankle;
      const ankleT = mat3_mulVec3(transforms[4].R, ankle[0], ankle[1], ankle[2]);
      const ankleWorld = [ankleT[0]+transforms[4].t[0], ankleT[1]+transforms[4].t[1], ankleT[2]+transforms[4].t[2]];
      const Rf_ankle = mat3_mulVec3(R_foot, ankle[0], ankle[1], ankle[2]);
      transforms[6] = {
        R: R_foot,
        t: [ankleWorld[0]-Rf_ankle[0], ankleWorld[1]-Rf_ankle[1], ankleWorld[2]-Rf_ankle[2]],
        cRest: transforms[0].cRest, cCurr: transforms[0].cCurr
      };
    } else if (transforms[4]) {
      transforms[6] = transforms[4];
    }

    // Head: analytical in position mode (R_headIK * R_torso, pinned at neck)
    if (headState && transforms[0]) {
      const R_t = transforms[0].R, t_t = transforms[0].t;
      // Head IK directly (no R_t) — R_head includes body lean via R_bodyUpper composition
      const R_head = headState;
      const neck = P.neck;
      const neckT = mat3_mulVec3(R_t, neck[0], neck[1], neck[2]);
      const neckWorld = [neckT[0]+t_t[0], neckT[1]+t_t[1], neckT[2]+t_t[2]];
      const Rh_neck = mat3_mulVec3(R_head, neck[0], neck[1], neck[2]);
      transforms[11] = {
        R: R_head,
        t: [neckWorld[0]-Rh_neck[0], neckWorld[1]-Rh_neck[1], neckWorld[2]-Rh_neck[2]],
        cRest: transforms[0].cRest, cCurr: transforms[0].cCurr
      };
    }

    // Upper arms: compose from analytical arm rotations, pinned at body-rotated shoulder.
    // Position mode: uses stored body-rotated shoulder (movedShoulderL/R) for exact pivot.
    // Euler mode: falls back to torso Kabsch-derived shoulder.
    if (armState && transforms[0] && cageShoulderL && cageArmDirL) {
      const R_t = transforms[0].R, t_t = transforms[0].t;
      let R_u;
      if (armState.positionBased) {
        R_u = armState.lUpperR;
      } else {
        const Rz = mat3_rotZ(armState.lTposeZ);
        const Rx = mat3_rotX(armState.lArmSwing);
        R_u = mat3_mul(Rx, mat3_mul(Rz, R_t));
      }
      const sh = cageShoulderL;
      // Use stored body-rotated shoulder in position mode (exact match to cage vert path)
      let actualSh;
      if (armState.positionBased && armState.movedShoulderL) {
        const gc = lastGroundCorrection;
        const ms = armState.movedShoulderL;
        actualSh = [ms[0], ms[1] + gc, ms[2]];
      } else {
        const as = mat3_mulVec3(R_t, sh[0], sh[1], sh[2]);
        actualSh = [as[0]+t_t[0], as[1]+t_t[1], as[2]+t_t[2]];
      }
      const Ru_sh = mat3_mulVec3(R_u, sh[0], sh[1], sh[2]);
      transforms[7] = {
        R: R_u,
        t: [actualSh[0]-Ru_sh[0], actualSh[1]-Ru_sh[1], actualSh[2]-Ru_sh[2]],
        cRest: transforms[0].cRest,
        cCurr: transforms[0].cCurr
      };
    }
    if (armState && transforms[0] && cageShoulderR && cageArmDirR) {
      const R_t = transforms[0].R, t_t = transforms[0].t;
      let R_u;
      if (armState.positionBased) {
        R_u = armState.rUpperR;
      } else {
        const Rz = mat3_rotZ(armState.rTposeZ);
        const Rx = mat3_rotX(armState.rArmSwing);
        R_u = mat3_mul(Rx, mat3_mul(Rz, R_t));
      }
      const sh = cageShoulderR;
      let actualSh;
      if (armState.positionBased && armState.movedShoulderR) {
        const gc = lastGroundCorrection;
        const ms = armState.movedShoulderR;
        actualSh = [ms[0], ms[1] + gc, ms[2]];
      } else {
        const as = mat3_mulVec3(R_t, sh[0], sh[1], sh[2]);
        actualSh = [as[0]+t_t[0], as[1]+t_t[1], as[2]+t_t[2]];
      }
      const Ru_sh = mat3_mulVec3(R_u, sh[0], sh[1], sh[2]);
      transforms[8] = {
        R: R_u,
        t: [actualSh[0]-Ru_sh[0], actualSh[1]-Ru_sh[1], actualSh[2]-Ru_sh[2]],
        cRest: transforms[0].cRest,
        cCurr: transforms[0].cCurr
      };
    }

    // Forearms: upper arm analytical + elbow bend + forearm twist.
    // R_forearm = [R_forearmTwist *] R_bend * R_upper, pinned at elbow.
    if (armState && transforms[7] && cageShoulderL && cageArmDirL && armState.lBendAxis) {
      const ba = armState.lBendAxis;
      const lBendAng = armState.positionBased ? armState.lBendAngle : armState.lElbowAngle;
      const Rb = mat3_rotAxis(ba[0], ba[1], ba[2], lBendAng);
      const R_u = transforms[7].R;
      const t_u = transforms[7].t;
      const uaLen = +$('upperArmLen').value;
      const erx = cageShoulderL[0] + cageArmDirL[0]*uaLen;
      const ery = cageShoulderL[1] + cageArmDirL[1]*uaLen;
      const erz = cageShoulderL[2] + cageArmDirL[2]*uaLen;
      const ae = mat3_mulVec3(R_u, erx, ery, erz);
      const ep = [ae[0]+t_u[0], ae[1]+t_u[1], ae[2]+t_u[2]];
      let R_f = mat3_mul(Rb, R_u);
      // Forearm twist (pronation/supination) — quaternion LOCAL extraction
      if (armState.positionBased && Math.abs(armState.lForearmTwist || 0) > 1e-4) {
        const fd = vnorm(mat3_mulVec3(R_f, cageArmDirL[0], cageArmDirL[1], cageArmDirL[2]));
        R_f = mat3_mul(mat3_rotAxis(fd[0], fd[1], fd[2], armState.lForearmTwist), R_f);
      }
      const Rf_re = mat3_mulVec3(R_f, erx, ery, erz);
      transforms[9] = {
        R: R_f,
        t: [ep[0] - Rf_re[0], ep[1] - Rf_re[1], ep[2] - Rf_re[2]],
        cRest: transforms[7].cRest,
        cCurr: transforms[7].cCurr
      };
    }
    if (armState && transforms[8] && cageShoulderR && cageArmDirR && armState.rBendAxis) {
      const ba = armState.rBendAxis;
      const rBendAng = armState.positionBased ? armState.rBendAngle : armState.rElbowAngle;
      const Rb = mat3_rotAxis(ba[0], ba[1], ba[2], rBendAng);
      const R_u = transforms[8].R;
      const t_u = transforms[8].t;
      const uaLen = +$('upperArmLen').value;
      const erx = cageShoulderR[0] + cageArmDirR[0]*uaLen;
      const ery = cageShoulderR[1] + cageArmDirR[1]*uaLen;
      const erz = cageShoulderR[2] + cageArmDirR[2]*uaLen;
      const ae = mat3_mulVec3(R_u, erx, ery, erz);
      const ep = [ae[0]+t_u[0], ae[1]+t_u[1], ae[2]+t_u[2]];
      let R_f = mat3_mul(Rb, R_u);
      // Forearm twist (right — pronation/supination) — quaternion LOCAL extraction
      if (armState.positionBased && Math.abs(armState.rForearmTwist || 0) > 1e-4) {
        const fd = vnorm(mat3_mulVec3(R_f, cageArmDirR[0], cageArmDirR[1], cageArmDirR[2]));
        R_f = mat3_mul(mat3_rotAxis(fd[0], fd[1], fd[2], armState.rForearmTwist), R_f);
      }
      const Rf_re = mat3_mulVec3(R_f, erx, ery, erz);
      transforms[10] = {
        R: R_f,
        t: [ep[0] - Rf_re[0], ep[1] - Rf_re[1], ep[2] - Rf_re[2]],
        cRest: transforms[8].cRest,
        cCurr: transforms[8].cCurr
      };
    }
    // Hands: forearm transform + analytical wrist rotation around bone-local axis.
    if (armState && transforms[9] && cageShoulderL && cageArmDirL && armState.lBendAxis) {
      // Position mode: use real wrist flex from FBX finger data, or fallback to elbow counter-rotation
      let ba, lWristAng;
      if (armState.positionBased && armState.lWristFlexAxis && armState.lWristFlexAngle > 1e-4) {
        ba = armState.lWristFlexAxis;
        lWristAng = armState.lWristFlexAngle;
      } else if (armState.positionBased) {
        ba = armState.lBendAxis;
        lWristAng = -armState.lBendAngle * 0.3;
      } else {
        ba = armState.lBendAxis;
        lWristAng = armState.lWristAngle;
      }
      const Rb = mat3_rotAxis(ba[0], ba[1], ba[2], lWristAng);
      const R_f = transforms[9].R;
      const t_f = transforms[9].t;
      const uaLen = +$('upperArmLen').value;
      const faLen = +$('forearmLen').value;
      const wrx = cageShoulderL[0] + cageArmDirL[0]*(uaLen+faLen);
      const wry = cageShoulderL[1] + cageArmDirL[1]*(uaLen+faLen);
      const wrz = cageShoulderL[2] + cageArmDirL[2]*(uaLen+faLen);
      const aw = mat3_mulVec3(R_f, wrx, wry, wrz);
      const wp = [aw[0]+t_f[0], aw[1]+t_f[1], aw[2]+t_f[2]];
      const R_h = mat3_mul(Rb, R_f);
      const Rb_tf = mat3_mulVec3(Rb, t_f[0], t_f[1], t_f[2]);
      const Rb_wp = mat3_mulVec3(Rb, wp[0], wp[1], wp[2]);
      transforms[12] = {
        R: R_h,
        t: [Rb_tf[0] + wp[0] - Rb_wp[0], Rb_tf[1] + wp[1] - Rb_wp[1], Rb_tf[2] + wp[2] - Rb_wp[2]],
        cRest: transforms[9].cRest,
        cCurr: transforms[9].cCurr
      };
    }
    if (armState && transforms[10] && cageShoulderR && cageArmDirR && armState.rBendAxis) {
      let ba, rWristAng;
      if (armState.positionBased && armState.rWristFlexAxis && armState.rWristFlexAngle > 1e-4) {
        ba = armState.rWristFlexAxis;
        rWristAng = armState.rWristFlexAngle;
      } else if (armState.positionBased) {
        ba = armState.rBendAxis;
        rWristAng = -armState.rBendAngle * 0.3;
      } else {
        ba = armState.rBendAxis;
        rWristAng = armState.rWristAngle;
      }
      const Rb = mat3_rotAxis(ba[0], ba[1], ba[2], rWristAng);
      const R_f = transforms[10].R;
      const t_f = transforms[10].t;
      const uaLen = +$('upperArmLen').value;
      const faLen = +$('forearmLen').value;
      const wrx = cageShoulderR[0] + cageArmDirR[0]*(uaLen+faLen);
      const wry = cageShoulderR[1] + cageArmDirR[1]*(uaLen+faLen);
      const wrz = cageShoulderR[2] + cageArmDirR[2]*(uaLen+faLen);
      const aw = mat3_mulVec3(R_f, wrx, wry, wrz);
      const wp = [aw[0]+t_f[0], aw[1]+t_f[1], aw[2]+t_f[2]];
      const R_h = mat3_mul(Rb, R_f);
      const Rb_tf = mat3_mulVec3(Rb, t_f[0], t_f[1], t_f[2]);
      const Rb_wp = mat3_mulVec3(Rb, wp[0], wp[1], wp[2]);
      transforms[13] = {
        R: R_h,
        t: [Rb_tf[0] + wp[0] - Rb_wp[0], Rb_tf[1] + wp[1] - Rb_wp[1], Rb_tf[2] + wp[2] - Rb_wp[2]],
        cRest: transforms[10].cRest,
        cCurr: transforms[10].cCurr
      };
    }
    return transforms;
  }

  // Precompute per-mesh-vertex region weights from bind data
  // Returns array of flat pairs: [rid1, w1, rid2, w2, ...] per vertex
  function precomputeMeshRegionWeights(meshRestPos, bindIdx, bindW, K, regionId) {
    const nMesh = meshRestPos.length / 3;
    const result = new Array(nMesh);

    for (let vi = 0; vi < nMesh; vi++) {
      const rw = new Map();
      const base = vi * K;
      for (let j = 0; j < K; j++) {
        const ci = bindIdx[base + j];
        const w = bindW[base + j];
        const rid = regionId[ci];
        rw.set(rid, (rw.get(rid) || 0) + w);
      }

      // Smoothstep elbow weight override: replace KNN-derived upper/forearm split
      // with smooth projection-based interpolation centered on the elbow.
      if (cageShoulderL && cageArmDirL && cageShoulderR && cageArmDirR) {
        const vx = meshRestPos[vi*3], vy = meshRestPos[vi*3+1], vz = meshRestPos[vi*3+2];
        const uaLen = +$('upperArmLen').value;
        const EBAND = 0.12 * uaLen;

        // Left arm: regions 7 (upper) + 9 (forearm)
        if (rw.has(7) && rw.has(9)) {
          const sh = cageShoulderL, d = cageArmDirL;
          const tp = (vx-sh[0])*d[0] + (vy-sh[1])*d[1] + (vz-sh[2])*d[2];
          if (Math.abs(tp - uaLen) < EBAND * 2) {
            const totalW = rw.get(7) + rw.get(9);
            const f = clamp((tp - (uaLen - EBAND)) / (2 * EBAND), 0, 1);
            const sf = f * f * (3 - 2 * f); // smoothstep
            rw.set(7, totalW * (1 - sf));
            rw.set(9, totalW * sf);
          }
        }
        // Right arm: regions 8 (upper) + 10 (forearm)
        if (rw.has(8) && rw.has(10)) {
          const sh = cageShoulderR, d = cageArmDirR;
          const tp = (vx-sh[0])*d[0] + (vy-sh[1])*d[1] + (vz-sh[2])*d[2];
          if (Math.abs(tp - uaLen) < EBAND * 2) {
            const totalW = rw.get(8) + rw.get(10);
            const f = clamp((tp - (uaLen - EBAND)) / (2 * EBAND), 0, 1);
            const sf = f * f * (3 - 2 * f); // smoothstep
            rw.set(8, totalW * (1 - sf));
            rw.set(10, totalW * sf);
          }
        }

        // Shoulder smoothstep: torso (0) ↔ upper arm (7/8)
        // Same idea as elbow — smooth projection-based weight at shoulder boundary.
        const SBAND = 0.08 * uaLen;

        // Left shoulder: regions 0 (torso) + 7 (upper arm)
        if (rw.has(0) && rw.has(7)) {
          const sh = cageShoulderL, d = cageArmDirL;
          const tp = (vx-sh[0])*d[0] + (vy-sh[1])*d[1] + (vz-sh[2])*d[2];
          if (Math.abs(tp) < SBAND * 2) {
            const totalW = rw.get(0) + rw.get(7);
            // t<0 = torso side, t>0 = arm side. Blend centered at t=0 (shoulder).
            const f = clamp((tp + SBAND) / (2 * SBAND), 0, 1);
            const sf = f * f * (3 - 2 * f);
            rw.set(0, totalW * (1 - sf));
            rw.set(7, totalW * sf);
          }
        }
        // Right shoulder: regions 0 (torso) + 8 (upper arm)
        if (rw.has(0) && rw.has(8)) {
          const sh = cageShoulderR, d = cageArmDirR;
          const tp = (vx-sh[0])*d[0] + (vy-sh[1])*d[1] + (vz-sh[2])*d[2];
          if (Math.abs(tp) < SBAND * 2) {
            const totalW = rw.get(0) + rw.get(8);
            const f = clamp((tp + SBAND) / (2 * SBAND), 0, 1);
            const sf = f * f * (3 - 2 * f);
            rw.set(0, totalW * (1 - sf));
            rw.set(8, totalW * sf);
          }
        }
      }

      // Smoothstep knee weight override: thigh (1/2) ↔ shin (3/4)
      if (P.l_hip && P.l_knee && P.l_ankle) {
        const vx = meshRestPos[vi*3], vy = meshRestPos[vi*3+1], vz = meshRestPos[vi*3+2];
        // Left knee: regions 1 (thigh) + 3 (shin)
        if (rw.has(1) && rw.has(3)) {
          const hip = P.l_hip, knee = P.l_knee;
          const thighLen = Math.sqrt((knee[0]-hip[0])**2 + (knee[1]-hip[1])**2 + (knee[2]-hip[2])**2);
          const d = [(knee[0]-hip[0])/thighLen, (knee[1]-hip[1])/thighLen, (knee[2]-hip[2])/thighLen];
          const tp = (vx-hip[0])*d[0] + (vy-hip[1])*d[1] + (vz-hip[2])*d[2];
          const KBAND = 0.15 * thighLen;
          if (Math.abs(tp - thighLen) < KBAND * 2) {
            const totalW = rw.get(1) + rw.get(3);
            const f = clamp((tp - (thighLen - KBAND)) / (2 * KBAND), 0, 1);
            const sf = f * f * (3 - 2 * f);
            rw.set(1, totalW * (1 - sf));
            rw.set(3, totalW * sf);
          }
        }
        // Right knee: regions 2 (thigh) + 4 (shin)
        if (rw.has(2) && rw.has(4) && P.r_hip && P.r_knee) {
          const hip = P.r_hip, knee = P.r_knee;
          const thighLen = Math.sqrt((knee[0]-hip[0])**2 + (knee[1]-hip[1])**2 + (knee[2]-hip[2])**2);
          const d = [(knee[0]-hip[0])/thighLen, (knee[1]-hip[1])/thighLen, (knee[2]-hip[2])/thighLen];
          const tp = (vx-hip[0])*d[0] + (vy-hip[1])*d[1] + (vz-hip[2])*d[2];
          const KBAND = 0.15 * thighLen;
          if (Math.abs(tp - thighLen) < KBAND * 2) {
            const totalW = rw.get(2) + rw.get(4);
            const f = clamp((tp - (thighLen - KBAND)) / (2 * KBAND), 0, 1);
            const sf = f * f * (3 - 2 * f);
            rw.set(2, totalW * (1 - sf));
            rw.set(4, totalW * sf);
          }
        }
      }

      // Crotch weight injection: ensure mesh verts between the hips have SOME
      // thigh weight so the hip smoothstep below can blend them properly.
      // IMPORTANT: inject only the NEARER thigh (based on X position) to avoid
      // opposing-rotation cancellation that causes crotch webbing.
      if (P.l_hip && P.r_hip && P.pelvis && P.l_knee && P.r_knee && rw.has(0)) {
        const vx = meshRestPos[vi*3], vy = meshRestPos[vi*3+1];
        const pelvisY = P.pelvis[1];
        const CROTCH_TOP = pelvisY + 0.02;
        const CROTCH_DEPTH = 0.08;

        if (vy < CROTCH_TOP && vy > pelvisY - CROTCH_DEPTH) {
          const lhx = P.l_hip[0], rhx = P.r_hip[0];
          const midX = (lhx + rhx) / 2;
          const halfSpread = Math.abs(lhx - rhx) / 2;
          const margin = halfSpread * 0.5;

          if (vx > lhx - margin && vx < rhx + margin) {
            const vertFade = clamp((CROTCH_TOP - vy) / CROTCH_DEPTH, 0, 1);
            const vertSS = vertFade * vertFade * (3 - 2 * vertFade);
            const normX = clamp((vx - midX) / halfSpread, -1, 1);

            // Inject only the NEARER thigh based on lateral position
            if (normX <= 0 && !rw.has(1) && vertSS > 0.05) {
              const lateralW = clamp(0.5 - normX * 0.5, 0.15, 0.5);
              const inject = vertSS * lateralW * 0.4;
              if (inject > 0.02) {
                const take = Math.min(inject, rw.get(0) * 0.45);
                rw.set(0, rw.get(0) - take);
                rw.set(1, take);
              }
            } else if (normX > 0 && !rw.has(2) && vertSS > 0.05) {
              const lateralW = clamp(0.5 + normX * 0.5, 0.15, 0.5);
              const inject = vertSS * lateralW * 0.4;
              if (inject > 0.02) {
                const take = Math.min(inject, rw.get(0) * 0.45);
                rw.set(0, rw.get(0) - take);
                rw.set(2, take);
              }
            }
          }
        }
      }

      // Smoothstep hip weight override: torso (0) ↔ thigh (1/2)
      // Without this, the torso-thigh boundary has no smooth transition,
      // causing stretching when hip twist or thigh rotations differ from torso.
      // HBAND = 0.60 * thighLen → gradient spans 60% above hip to 60% below hip,
      // so verts in the upper half of the thigh get graduated torso weight.
      // Previous value (0.30) was too narrow — only 5cm zone left inner-thigh
      // verts with 100% thigh weight, causing the crotch stretching gap.
      if (P.l_hip && P.l_knee) {
        const vx = meshRestPos[vi*3], vy = meshRestPos[vi*3+1], vz = meshRestPos[vi*3+2];
        // Left hip: regions 0 (torso) + 1 (l_thigh)
        if (rw.has(0) && rw.has(1)) {
          const hip = P.l_hip, knee = P.l_knee;
          const thighLen = Math.sqrt((knee[0]-hip[0])**2 + (knee[1]-hip[1])**2 + (knee[2]-hip[2])**2);
          const d = [(knee[0]-hip[0])/thighLen, (knee[1]-hip[1])/thighLen, (knee[2]-hip[2])/thighLen];
          const tp = (vx-hip[0])*d[0] + (vy-hip[1])*d[1] + (vz-hip[2])*d[2];
          const HBAND = 0.60 * thighLen;
          if (Math.abs(tp) < HBAND * 2) {
            const totalW = rw.get(0) + rw.get(1);
            const f = clamp((tp + HBAND) / (2 * HBAND), 0, 1);
            const sf = f * f * (3 - 2 * f); // smoothstep
            rw.set(0, totalW * (1 - sf));
            rw.set(1, totalW * sf);
          }
        }
        // Right hip: regions 0 (torso) + 2 (r_thigh)
        if (rw.has(0) && rw.has(2) && P.r_hip && P.r_knee) {
          const hip = P.r_hip, knee = P.r_knee;
          const thighLen = Math.sqrt((knee[0]-hip[0])**2 + (knee[1]-hip[1])**2 + (knee[2]-hip[2])**2);
          const d = [(knee[0]-hip[0])/thighLen, (knee[1]-hip[1])/thighLen, (knee[2]-hip[2])/thighLen];
          const tp = (vx-hip[0])*d[0] + (vy-hip[1])*d[1] + (vz-hip[2])*d[2];
          const HBAND = 0.60 * thighLen;
          if (Math.abs(tp) < HBAND * 2) {
            const totalW = rw.get(0) + rw.get(2);
            const f = clamp((tp + HBAND) / (2 * HBAND), 0, 1);
            const sf = f * f * (3 - 2 * f); // smoothstep
            rw.set(0, totalW * (1 - sf));
            rw.set(2, totalW * sf);
          }
        }
      }

      // Opposing thigh suppression: if a vert has both L_thigh(1) and R_thigh(2)
      // weight, the opposing rotations cancel when legs spread (one forward, one back),
      // causing the vert to "stay put" = crotch webbing. Fix: keep only the nearer
      // thigh based on lateral position, transfer the other to torso.
      if (P.l_hip && P.r_hip && rw.has(1) && rw.has(2)) {
        const vx = meshRestPos[vi*3];
        const midX = (P.l_hip[0] + P.r_hip[0]) / 2;
        const halfSpread = Math.abs(P.l_hip[0] - P.r_hip[0]) / 2;
        // Narrow cross-fade zone (15% of hip spread) for smooth transition at midline
        const FADE = Math.max(halfSpread * 0.15, 0.005);
        const t = clamp((vx - midX) / FADE, -1, 1);
        const keepR = (t + 1) * 0.5; // 0 at left edge, 1 at right edge
        const keepL = 1 - keepR;
        const lw = rw.get(1), rwv = rw.get(2);
        const newL = lw * keepL;
        const newR = rwv * keepR;
        const toTorso = (lw - newL) + (rwv - newR);
        if (toTorso > 0.001) rw.set(0, (rw.get(0) || 0) + toTorso);
        if (newL > 0.001) rw.set(1, newL); else rw.delete(1);
        if (newR > 0.001) rw.set(2, newR); else rw.delete(2);
      }

      // Smoothstep ankle weight override: shin (3/4) ↔ foot (5/6)
      if (P.l_knee && P.l_ankle) {
        const vx = meshRestPos[vi*3], vy = meshRestPos[vi*3+1], vz = meshRestPos[vi*3+2];
        // Left ankle: regions 3 (shin) + 5 (foot)
        if (rw.has(3) && rw.has(5)) {
          const knee = P.l_knee, ankle = P.l_ankle;
          const shinLen = Math.sqrt((ankle[0]-knee[0])**2 + (ankle[1]-knee[1])**2 + (ankle[2]-knee[2])**2);
          const d = [(ankle[0]-knee[0])/shinLen, (ankle[1]-knee[1])/shinLen, (ankle[2]-knee[2])/shinLen];
          const tp = (vx-knee[0])*d[0] + (vy-knee[1])*d[1] + (vz-knee[2])*d[2];
          const ABAND = 0.15 * shinLen;
          if (Math.abs(tp - shinLen) < ABAND * 2) {
            const totalW = rw.get(3) + rw.get(5);
            const f = clamp((tp - (shinLen - ABAND)) / (2 * ABAND), 0, 1);
            const sf = f * f * (3 - 2 * f);
            rw.set(3, totalW * (1 - sf));
            rw.set(5, totalW * sf);
          }
        }
        // Right ankle: regions 4 (shin) + 6 (foot)
        if (rw.has(4) && rw.has(6) && P.r_knee && P.r_ankle) {
          const knee = P.r_knee, ankle = P.r_ankle;
          const shinLen = Math.sqrt((ankle[0]-knee[0])**2 + (ankle[1]-knee[1])**2 + (ankle[2]-knee[2])**2);
          const d = [(ankle[0]-knee[0])/shinLen, (ankle[1]-knee[1])/shinLen, (ankle[2]-knee[2])/shinLen];
          const tp = (vx-knee[0])*d[0] + (vy-knee[1])*d[1] + (vz-knee[2])*d[2];
          const ABAND = 0.15 * shinLen;
          if (Math.abs(tp - shinLen) < ABAND * 2) {
            const totalW = rw.get(4) + rw.get(6);
            const f = clamp((tp - (shinLen - ABAND)) / (2 * ABAND), 0, 1);
            const sf = f * f * (3 - 2 * f);
            rw.set(4, totalW * (1 - sf));
            rw.set(6, totalW * sf);
          }
        }
      }

      // Neck weight injection: head verts near/below the neck need some torso
      // weight so the neck smoothstep can create a smooth transition. Without this,
      // head verts whose K nearest cage verts are all head (e.g. chin/jaw) get 100%
      // head transform, causing visible deformation when head and torso differ.
      if (P.neck && P.pelvis && rw.has(11) && !rw.has(0)) {
        const vy = meshRestPos[vi*3+1];
        const neckY = P.neck[1];
        const NECK_INJECT_TOP = 0.02;   // start injecting just above neck
        const NECK_INJECT_DEPTH = 0.08; // blend over 8cm below neck
        if (vy < neckY + NECK_INJECT_TOP && vy > neckY - NECK_INJECT_DEPTH) {
          const f = clamp((neckY + NECK_INJECT_TOP - vy) / NECK_INJECT_DEPTH, 0, 1);
          const sf = f * f * (3 - 2 * f); // smoothstep
          const inject = sf * 0.5; // up to 50% torso at bottom of zone
          if (inject > 0.03) {
            const headW = rw.get(11);
            const take = headW * inject;
            rw.set(11, headW - take);
            rw.set(0, take);
          }
        }
      }

      // Smoothstep neck weight override: torso (0) ↔ head (11)
      if (P.pelvis && P.neck) {
        const vx = meshRestPos[vi*3], vy = meshRestPos[vi*3+1], vz = meshRestPos[vi*3+2];
        if (rw.has(0) && rw.has(11)) {
          const pelvis = P.pelvis, neck = P.neck;
          const spineLen = Math.sqrt((neck[0]-pelvis[0])**2 + (neck[1]-pelvis[1])**2 + (neck[2]-pelvis[2])**2);
          const d = [(neck[0]-pelvis[0])/spineLen, (neck[1]-pelvis[1])/spineLen, (neck[2]-pelvis[2])/spineLen];
          const tp = (vx-pelvis[0])*d[0] + (vy-pelvis[1])*d[1] + (vz-pelvis[2])*d[2];
          const NBAND = 0.12 * spineLen;
          if (Math.abs(tp - spineLen) < NBAND * 2) {
            const totalW = rw.get(0) + rw.get(11);
            const f = clamp((tp - (spineLen - NBAND)) / (2 * NBAND), 0, 1);
            const sf = f * f * (3 - 2 * f);
            rw.set(0, totalW * (1 - sf));
            rw.set(11, totalW * sf);
          }
        }
      }

      // Flatten to [rid, weight, rid, weight, ...] and normalize
      const entries = [];
      let sum = 0;
      for (const [rid, w] of rw) {
        entries.push(rid, w);
        sum += w;
      }
      if (sum > 0) {
        for (let i = 1; i < entries.length; i += 2) entries[i] /= sum;
      }
      result[vi] = entries;
    }

    // One-shot crotch weight diagnostic
    if (P.l_hip && P.r_hip && P.pelvis) {
      const pelvisY = P.pelvis[1];
      let crotchVerts = 0, torsoOnly = 0, hasThigh = 0;
      let minThighW = 1, maxThighW = 0, sumThighW = 0;
      for (let vi = 0; vi < nMesh; vi++) {
        const vy = meshRestPos[vi*3+1], vx = meshRestPos[vi*3];
        if (vy > pelvisY - 0.08 && vy < pelvisY + 0.02 &&
            vx > P.l_hip[0] * 0.8 && vx < P.r_hip[0] * 0.8) {
          crotchVerts++;
          const e = result[vi];
          let thighW = 0, torsoW = 0;
          for (let i = 0; i < e.length; i += 2) {
            if (e[i] === 1 || e[i] === 2) thighW += e[i+1];
            if (e[i] === 0) torsoW += e[i+1];
          }
          if (thighW < 0.01) torsoOnly++;
          else {
            hasThigh++;
            sumThighW += thighW;
            if (thighW < minThighW) minThighW = thighW;
            if (thighW > maxThighW) maxThighW = thighW;
          }
        }
      }
      console.log('=== CROTCH WEIGHT DIAGNOSTIC ===');
      console.log('  Crotch verts: ' + crotchVerts + ' (torso-only: ' + torsoOnly + ', has thigh: ' + hasThigh + ')');
      if (hasThigh > 0) console.log('  Thigh weight: min=' + minThighW.toFixed(3) + ' max=' + maxThighW.toFixed(3) + ' avg=' + (sumThighW/hasThigh).toFixed(3));
      if (torsoOnly > 0) console.log('  WARNING: ' + torsoOnly + ' crotch verts have NO thigh weight — they will stretch as webbing!');
    }

    return result;
  }

  // ---------------- Motion interpolation (from your old code) ----------------
  function getRotationAtTime(boneName, t, axis = 'x_rot') {
    if (!motionData || !motionData[boneName]) return 0;
    const data = motionData[boneName];
    const times = data.times;
    const values = data[axis];
    if (!values || values.length === 0) return 0;

    const cycleT = t % cycleDuration;
    const firstTime = times[0];
    const lastTime = times[times.length - 1];

    if (cycleT < firstTime || cycleT > lastTime) {
      const v0 = values[values.length - 1];
      const v1 = values[0];
      const t0 = lastTime;
      const t1 = firstTime + cycleDuration;
      const adjustedT = cycleT < firstTime ? cycleT + cycleDuration : cycleT;
      const frac = (adjustedT - t0) / (t1 - t0);
      const degrees = v0 + (v1 - v0) * clamp(frac,0,1);
      return degrees * Math.PI / 180;
    }

    let i = 0;
    while (i < times.length - 1 && times[i + 1] < cycleT) i++;
    const t0 = times[i], t1 = times[i + 1];
    const v0 = values[i], v1 = values[i + 1];
    const frac = (cycleT - t0) / (t1 - t0);
    const degrees = v0 + (v1 - v0) * frac;
    return degrees * Math.PI / 180;
  }

  function getArmRotationAtTime(boneName, t, axis = 'x_rot') {
    if (!armMotionData || !armMotionData[boneName]) return 0;
    const data = armMotionData[boneName];
    const times = data.times;
    const values = data[axis];
    if (!values || values.length === 0) return 0;

    const cycleT = t % cycleDuration;
    const firstTime = times[0];
    const lastTime = times[times.length - 1];
    const n = values.length;

    if (cycleT < firstTime || cycleT > lastTime) {
      const v0 = values[n - 1];
      const v1 = values[0];
      const t0 = lastTime;
      const t1 = firstTime + cycleDuration;
      const adjustedT = cycleT < firstTime ? cycleT + cycleDuration : cycleT;
      const frac = clamp((adjustedT - t0) / (t1 - t0), 0, 1);
      // Catmull-Rom at cycle boundary
      const p0 = values[n - 2];
      const p3 = values[1];
      const tt = frac, tt2 = tt*tt, tt3 = tt2*tt;
      const degrees = 0.5*((2*v0)+(-p0+v1)*tt+(2*p0-5*v0+4*v1-p3)*tt2+(-p0+3*v0-3*v1+p3)*tt3);
      return degrees * Math.PI / 180;
    }

    let i = 0;
    while (i < times.length - 1 && times[i + 1] < cycleT) i++;
    const t0 = times[i], t1 = times[i + 1];
    const frac = (cycleT - t0) / (t1 - t0);

    // Catmull-Rom spline: smooth velocity at each keyframe (no abrupt direction changes)
    const p0 = values[i > 0 ? i - 1 : n - 1];
    const p1 = values[i];
    const p2 = values[i + 1 < n ? i + 1 : 0];
    const p3 = values[i + 2 < n ? i + 2 : (i + 2) % n];
    const tt = frac, tt2 = tt*tt, tt3 = tt2*tt;
    const degrees = 0.5*((2*p1)+(-p0+p2)*tt+(2*p0-5*p1+4*p2-p3)*tt2+(-p0+3*p1-3*p2+p3)*tt3);
    return degrees * Math.PI / 180;
  }

  function getSpineDataAtTime(boneName, t, axis = 'x_rot') {
    if (!spineMotionData || !spineMotionData[boneName]) return 0;
    const data = spineMotionData[boneName];
    let times = data.times;
    const values = data[axis];
    if (!values || values.length === 0) return 0;

    // Some bones (spine, neck, head) have only 2 time entries but 32 values.
    // Generate uniform times from first to last when there's a mismatch.
    if (values.length > times.length) {
      const t0 = times[0], tN = times[times.length - 1];
      times = [];
      for (let k = 0; k < values.length; k++) {
        times.push(t0 + (tN - t0) * k / (values.length - 1));
      }
    }

    const cycleT = t % cycleDuration;
    const firstTime = times[0];
    const lastTime = times[times.length - 1];

    if (cycleT < firstTime || cycleT > lastTime) {
      const v0 = values[values.length - 1];
      const v1 = values[0];
      const t0 = lastTime;
      const t1 = firstTime + cycleDuration;
      const adjustedT = cycleT < firstTime ? cycleT + cycleDuration : cycleT;
      const frac = (adjustedT - t0) / (t1 - t0);
      return v0 + (v1 - v0) * clamp(frac,0,1);
    }

    let i = 0;
    while (i < times.length - 1 && times[i + 1] < cycleT) i++;
    const t0 = times[i], t1 = times[i + 1];
    const v0 = values[i], v1 = values[i + 1];
    const frac = (cycleT - t0) / (t1 - t0);
    return v0 + (v1 - v0) * frac;
  }

  // ---------------- Position-based interpolation & limb IK ----------------

  // Get hips-local joint position at time t (Catmull-Rom, cyclic)
  // Returns [x, y, z] in hips-local cage-scale coordinates
  function getJointPositionAtTime(jointName, t) {
    if (!jointPosData || !jointPosData.joints[jointName]) return null;
    const jd = jointPosData.joints[jointName];
    const times = jd.times;
    const positions = jd.positions;
    const n = positions.length;
    if (n === 0) return null;
    if (n === 1) return positions[0].slice();

    const duration = jointPosData.duration;
    const cycleT = t % duration;

    // Find bracketing frames
    let i = 0;
    while (i < n - 1 && times[i + 1] < cycleT) i++;

    // Catmull-Rom with 4 control points (cyclic wrapping)
    const i0 = (i - 1 + n) % n;
    const i1 = i;
    const i2 = (i + 1) % n;
    const i3 = (i + 2) % n;

    const t0 = times[i1];
    const t1 = (i2 === 0) ? duration : times[i2];
    let frac;
    if (cycleT < t0) {
      // Before first frame — interpolate from last to first
      const tLast = times[n - 1];
      frac = (cycleT + duration - tLast) / (times[0] + duration - tLast);
      frac = Math.max(0, Math.min(1, frac));
      // Use last→first segment
      const p0 = positions[(n - 2 + n) % n];
      const p1 = positions[n - 1];
      const p2 = positions[0];
      const p3 = positions[Math.min(1, n - 1)];
      return catmullRomVec3(p0, p1, p2, p3, frac);
    }

    frac = (t1 > t0) ? (cycleT - t0) / (t1 - t0) : 0;
    frac = Math.max(0, Math.min(1, frac));

    return catmullRomVec3(positions[i0], positions[i1], positions[i2], positions[i3], frac);
  }

  function catmullRomVec3(p0, p1, p2, p3, t) {
    const t2 = t * t, t3 = t2 * t;
    const result = [0, 0, 0];
    for (let c = 0; c < 3; c++) {
      result[c] = 0.5 * (
        (2 * p1[c]) +
        (-p0[c] + p2[c]) * t +
        (2*p0[c] - 5*p1[c] + 4*p2[c] - p3[c]) * t2 +
        (-p0[c] + 3*p1[c] - 3*p2[c] + p3[c]) * t3
      );
    }
    return result;
  }

  // Interpolate joint quaternion at time t using SLERP (cyclic)
  function getJointQuaternionAtTime(jointName, t) {
    if (!jointPosData || !jointPosData.joints[jointName]) return null;
    const jd = jointPosData.joints[jointName];
    const quats = jd.quaternions;
    if (!quats || quats.length === 0) return null;
    if (quats.length === 1) return quats[0].slice();

    const times = jd.times;
    const n = quats.length;
    const duration = jointPosData.duration;
    const cycleT = t % duration;

    // Find bracketing frames
    let i = 0;
    while (i < n - 1 && times[i + 1] < cycleT) i++;

    const i1 = i;
    const i2 = (i + 1) % n;

    const t0 = times[i1];
    const t1 = (i2 === 0) ? duration : times[i2];
    let frac;
    if (cycleT < t0) {
      const tLast = times[n - 1];
      frac = (cycleT + duration - tLast) / (times[0] + duration - tLast);
      frac = Math.max(0, Math.min(1, frac));
      return quat_slerp(quats[n - 1], quats[0], frac);
    }
    frac = (t1 > t0) ? (cycleT - t0) / (t1 - t0) : 0;
    frac = Math.max(0, Math.min(1, frac));
    return quat_slerp(quats[i1], quats[i2], frac);
  }

  // Get twist angle for a bone at time t using quaternion swing-twist decomposition.
  // boneName: the joint whose twist we want (e.g. 'l_shoulder')
  // childName: the child joint, used to compute rest bone direction
  function getJointTwistAtTime(boneName, childName, t) {
    const curQ = getJointQuaternionAtTime(boneName, t);
    if (!curQ || !jointPosData.rest_quats[boneName]) return 0;
    const restQ = jointPosData.rest_quats[boneName];
    // Bone direction from rest-pose positions (already X-mirrored & scaled)
    const rp = jointPosData.rest_pose;
    if (!rp[boneName] || !rp[childName]) return 0;
    const boneDir = vnorm(vsub(rp[childName], rp[boneName]));
    return extractQuatTwist(restQ, curQ, boneDir);
  }

  // Transform hips-local position to world via torso Kabsch transform
  function hipsLocalToWorld(localPos, torsoR, torsoT) {
    const w = mat3_mulVec3(torsoR, localPos[0], localPos[1], localPos[2]);
    return [w[0] + torsoT[0], w[1] + torsoT[1], w[2] + torsoT[2]];
  }

  // Interpolate root (hips world) position at time t — for body bob/sway
  function getRootPositionAtTime(t) {
    if (!jointPosData || !jointPosData.root_positions) return null;
    const rp = jointPosData.root_positions;
    const n = rp.length;
    if (n === 0) return null;
    if (n === 1) return rp[0].slice();

    const duration = jointPosData.duration;
    const cycleT = t % duration;
    const fps = jointPosData.fps;

    const fExact = cycleT * fps;
    const i1 = Math.min(Math.floor(fExact), n - 1);
    const i2 = Math.min(i1 + 1, n - 1);
    const frac = fExact - Math.floor(fExact);

    return [
      rp[i1][0] + (rp[i2][0] - rp[i1][0]) * frac,
      rp[i1][1] + (rp[i2][1] - rp[i1][1]) * frac,
      rp[i1][2] + (rp[i2][2] - rp[i1][2]) * frac
    ];
  }

  // Helper: cross product
  function vcross(a, b) {
    return [
      a[1]*b[2] - a[2]*b[1],
      a[2]*b[0] - a[0]*b[2],
      a[0]*b[1] - a[1]*b[0]
    ];
  }

  // Stable bend axis: returns normalized cross product of two directions,
  // falling back to previous frame's axis when the cross product is degenerate
  // (near-straight or near-180° limb). Updates prevBendAxis[key] in-place.
  function stableBendAxis(dir1, dir2, key, fallback) {
    const c = vcross(dir1, dir2);
    const len = vlen(c);
    if (len > 1e-4) {
      const axis = vnorm(c);
      // Ensure consistent sign with previous frame (prevent 180° flips)
      if (prevBendAxis[key] && vdot(axis, prevBendAxis[key]) < 0) {
        const flipped = [-axis[0], -axis[1], -axis[2]];
        prevBendAxis[key] = flipped;
        return flipped;
      }
      prevBendAxis[key] = axis;
      return axis;
    }
    // Degenerate: use previous frame or fallback
    if (prevBendAxis[key]) return prevBendAxis[key];
    return fallback || [0, 1, 0];
  }

  // Extract twist around a bone axis by comparing actual vs expected bend plane normals.
  // The bend plane normal (cross product of parent and child bone directions) encodes
  // the axial rotation that position-only retargeting misses (e.g. forearm pronation,
  // thigh internal rotation). Returns signed twist angle in radians.
  function extractBoneTwist(restBoneDir, curBoneDir, restChildDir, curChildDir) {
    const restCross = vcross(restBoneDir, restChildDir);
    const curCross = vcross(curBoneDir, curChildDir);

    // Need both normals to be non-degenerate (bend > ~0.3°)
    // T-pose always has a tiny natural bend, giving a clean reference
    if (vlen(restCross) < 0.002 || vlen(curCross) < 0.05) return 0;

    const restBendN = vnorm(restCross);
    const curBendN = vnorm(curCross);

    // Expected bend normal if NO twist occurred (just swing from rest to current)
    const expectedBendN = applyDeltaRotation(restBoneDir, curBoneDir, restBendN);

    // Project both onto plane perpendicular to current bone direction
    const projE = vsub(expectedBendN, vscale(curBoneDir, vdot(expectedBendN, curBoneDir)));
    const projC = vsub(curBendN, vscale(curBoneDir, vdot(curBendN, curBoneDir)));

    if (vlen(projE) < 1e-6 || vlen(projC) < 1e-6) return 0;

    const pE = vnorm(projE), pC = vnorm(projC);
    const dotTw = Math.max(-1, Math.min(1, vdot(pE, pC)));
    let twist = Math.acos(dotTw);
    const crossTw = vcross(pE, pC);
    if (vdot(crossTw, curBoneDir) < 0) twist = -twist;
    return twist;
  }

  // Position-based twist extraction from bend-axis comparison.
  // Compares actual joint bend axis (from positions, character-independent) with
  // the expected "no-twist" bend axis (swing rotation applied to a canonical reference).
  // Returns signed twist angle in radians. Returns 0 if bend is too small for stable axis.
  function extractPositionTwist(swingR, targetBoneDir, refPerp, actualBendAxis, bendAngle) {
    if (Math.abs(bendAngle) < 0.1) return 0; // need ~6° bend for stable axis
    const expected = vnorm(mat3_mulVec3(swingR, refPerp[0], refPerp[1], refPerp[2]));
    // Project both onto plane perpendicular to target bone direction (ensure orthogonality)
    const projE = vsub(expected, vscale(targetBoneDir, vdot(expected, targetBoneDir)));
    const projA = vsub(actualBendAxis, vscale(targetBoneDir, vdot(actualBendAxis, targetBoneDir)));
    if (vlen(projE) < 1e-6 || vlen(projA) < 1e-6) return 0;
    const eN = vnorm(projE), aN = vnorm(projA);
    const cosT = Math.max(-1, Math.min(1, vdot(eN, aN)));
    const sinT = vdot(vcross(eN, aN), targetBoneDir);
    return Math.atan2(sinT, cosT);
  }

  // Compute a canonical reference perpendicular for a bone rest direction.
  // rawRef is the anatomical "natural bend" direction (e.g., [0,1,0] for elbow up).
  // Orthogonalizes rawRef to restBoneDir to ensure perpendicularity.
  function canonicalBendRef(rawRef, restBoneDir) {
    const d = vdot(rawRef, restBoneDir);
    const orth = vsub(rawRef, vscale(restBoneDir, d));
    return (vlen(orth) > 1e-6) ? vnorm(orth) : rawRef;
  }

  // For position-based driving: given target world positions of 3 joints,
  // rotate cage vertices so the limb matches. Works by computing the full
  // rotation chain: rest→T-pose→target for each segment.
  //
  // This replaces the Euler-based applyRotZ(tPose)+applyRotX(swing)+applyRotAxis(bend) chain.
  function driveArmFromPositions_world(
    shoulderW, elbowW, wristW,    // target world positions
    shoulderRest,                   // rest-pose shoulder position [x,y,z]
    restArmDir,                     // rest-pose arm direction (±X normalized)
    upperLen, foreLen,              // segment lengths
    upperIdx, foreIdx, handIdx      // cage vert index arrays
  ) {
    const allIdx = [].concat(upperIdx, foreIdx, handIdx || []);
    const foreHandIdx = [].concat(foreIdx, handIdx || []);

    // Target upper arm direction
    const targetUpperDir = vnorm(vsub(elbowW, shoulderW));

    // Rest-pose upper arm direction (from cage)
    // The rest-pose arm points along restArmDir from shoulderRest
    const restDir = restArmDir;

    // Rotation from rest direction to target direction (axis-angle)
    let axis1 = vcross(restDir, targetUpperDir);
    const axis1Len = vlen(axis1);
    const dot1 = Math.max(-1, Math.min(1, vdot(restDir, targetUpperDir)));
    const angle1 = Math.acos(dot1);

    if (axis1Len > 1e-6 && Math.abs(angle1) > 1e-4) {
      axis1 = vnorm(axis1);
      applyRotAxis(allIdx, shoulderRest, axis1[0], axis1[1], axis1[2], angle1);
    }

    // Elbow pivot after upper rotation
    const elbowPivot = [
      shoulderRest[0] + targetUpperDir[0] * upperLen,
      shoulderRest[1] + targetUpperDir[1] * upperLen,
      shoulderRest[2] + targetUpperDir[2] * upperLen
    ];

    // Forearm bend: angle between upper and lower segments
    const targetLowerDir = vnorm(vsub(wristW, elbowW));
    const dotBend = Math.max(-1, Math.min(1, vdot(targetUpperDir, targetLowerDir)));
    const bendAngle = Math.acos(dotBend);

    if (Math.abs(bendAngle) > 1e-4) {
      let bendAxis = vcross(targetUpperDir, targetLowerDir);
      if (vlen(bendAxis) < 1e-6) {
        // Try two perpendicular directions; at least one won't be parallel
        bendAxis = vcross(targetUpperDir, [0, 1, 0]);
        if (vlen(bendAxis) < 1e-6) bendAxis = vcross(targetUpperDir, [1, 0, 0]);
        if (vlen(bendAxis) < 1e-6) bendAxis = vcross(targetUpperDir, [0, 0, 1]);
      }
      bendAxis = vnorm(bendAxis);
      applyRotAxis(foreHandIdx, elbowPivot, bendAxis[0], bendAxis[1], bendAxis[2], bendAngle);
    }

    // Wrist pivot after all rotations
    const wristPivot = [
      elbowPivot[0] + targetLowerDir[0] * foreLen,
      elbowPivot[1] + targetLowerDir[1] * foreLen,
      elbowPivot[2] + targetLowerDir[2] * foreLen
    ];

    return {
      shoulderPivot: shoulderRest,
      elbowPivot,
      wristPivot,
      upperDir: targetUpperDir,
      lowerDir: targetLowerDir,
      bendAngle,
      bendAxis: (Math.abs(bendAngle) > 1e-4) ? vnorm(vcross(targetUpperDir, targetLowerDir)) : [0,1,0]
    };
  }

  // Drive a leg from positions: hip→knee→ankle, plus foot
  function driveLegFromPositions_world(
    hipW, kneeW, ankleW,
    hipRest, restLegDir,
    thighLen, shinLen,
    thighIdx, shinIdx, footIdx
  ) {
    const allIdx = [].concat(thighIdx, shinIdx, footIdx);
    const shinFootIdx = [].concat(shinIdx, footIdx);

    // Target thigh direction
    const targetThighDir = vnorm(vsub(kneeW, hipW));
    const restDir = restLegDir; // should be [0, -1, 0] (straight down)

    // Rotation from rest to target for whole leg
    let axis1 = vcross(restDir, targetThighDir);
    const axis1Len = vlen(axis1);
    const dot1 = Math.max(-1, Math.min(1, vdot(restDir, targetThighDir)));
    const angle1 = Math.acos(dot1);

    if (axis1Len > 1e-6 && Math.abs(angle1) > 1e-4) {
      axis1 = vnorm(axis1);
      applyRotAxis(allIdx, hipRest, axis1[0], axis1[1], axis1[2], angle1);
    }

    // Knee pivot
    const kneePivot = [
      hipRest[0] + targetThighDir[0] * thighLen,
      hipRest[1] + targetThighDir[1] * thighLen,
      hipRest[2] + targetThighDir[2] * thighLen
    ];

    // Shin bend
    const targetShinDir = vnorm(vsub(ankleW, kneeW));
    const dotBend = Math.max(-1, Math.min(1, vdot(targetThighDir, targetShinDir)));
    const bendAngle = Math.acos(dotBend);

    if (Math.abs(bendAngle) > 1e-4) {
      let bendAxis = vcross(targetThighDir, targetShinDir);
      if (vlen(bendAxis) < 1e-6) {
        bendAxis = [1, 0, 0]; // default: bend forward
      }
      bendAxis = vnorm(bendAxis);
      applyRotAxis(shinFootIdx, kneePivot, bendAxis[0], bendAxis[1], bendAxis[2], bendAngle);
    }

    const anklePivot = [
      kneePivot[0] + targetShinDir[0] * shinLen,
      kneePivot[1] + targetShinDir[1] * shinLen,
      kneePivot[2] + targetShinDir[2] * shinLen
    ];

    return {
      hipPivot: hipRest, kneePivot, anklePivot,
      thighDir: targetThighDir, shinDir: targetShinDir,
      bendAngle
    };
  }

  function deformMesh(){
    if (!meshVertRegions) return;
    const alpha = +$('alpha').value;
    const pos = meshGeo.attributes.position.array;
    const nMesh = meshRestPos.length / 3;

    // Debug region coloring

    if (meshObj) {

      meshObj.material.needsUpdate = true;
    }

    const transforms = computeAllRegionTransforms();
    lastRegionTransforms = transforms;

    if (window._posDiagDone && !window._kabschDiagDone) {
      window._kabschDiagDone = true;

      console.log('=== REGION TRANSFORMS (posMode=' + posMode + ', arms=analytical) ===');
      const regionNames = {0:'torso',1:'l_thigh',2:'r_thigh',3:'l_shin',4:'r_shin',5:'l_foot',6:'r_foot',7:'l_uarm',8:'r_uarm',9:'l_farm',10:'r_farm',11:'head',12:'l_hand',13:'r_hand'};
      for (let r = 0; r <= 13; r++) {

        const det = mat3_det(tf.R);
        const tMag = Math.sqrt(tf.t[0]*tf.t[0]+tf.t[1]*tf.t[1]+tf.t[2]*tf.t[2]);
        const hasNaN = Array.from(tf.R).some(v=>isNaN(v)) || tf.t.some(v=>isNaN(v));
        console.log('  ' + (regionNames[r]||r) + ': det=' + det.toFixed(4) + ' |t|=' + tMag.toFixed(4) + ' t=[' + tf.t.map(v=>v.toFixed(4)).join(',') + ']' + (hasNaN?' NaN!':''));
      }
    }

    const useDQS = $('useDQS').checked;
    const hardRegions = $('hardRegions').checked;

    // Pre-convert each region's (R, t) to dual quaternion once per frame (DQS mode)
    const regionDQs = {};
    if (useDQS && !hardRegions) {
      for (const rid in transforms) {
        const tf = transforms[rid];
        if (!tf) continue;
        const q = mat3_to_quat(tf.R);
        regionDQs[rid] = rigid_to_dq(q, tf.t);
      }
    }

    // ── Pre-compute hip-anchor slerp data (once per frame) ──
    // The analytical thigh transform is a RIGID rotation around the hip.
    // This rotates ALL thigh verts by the same angle — but groin verts should
    // barely move (like in a real body / skeletal rig with pelvis bone weights).
    // Fix: for each thigh mesh vert, slerp between torso and thigh quaternions
    // based on bone-projection fraction (0 at hip → 1 at knee). This replaces
    // DQS/LBS for the thigh and provides a proper joint gradient.
    let q_torso = null, hipAnchorL = null, hipAnchorR = null;
    if (transforms[0] && P.pelvis) {
      q_torso = mat3_to_quat(transforms[0].R);
      const R0 = transforms[0].R, t0 = transforms[0].t;
      if (transforms[1] && transforms[1]._hip && transforms[1]._thighDir) {
        const hip = transforms[1]._hip;
        // Use FULL thigh rotation (not R_noTwist) for correct positions
        const Rt = transforms[1].R;
        const tt = transforms[1].t;
        hipAnchorL = {
          hip: hip,
          q_thigh: mat3_to_quat(Rt),
          thighLen: transforms[1]._thighLen,
          thighDir: transforms[1]._thighDir,
          hipTorso: [R0[0]*hip[0]+R0[1]*hip[1]+R0[2]*hip[2]+t0[0],
                     R0[3]*hip[0]+R0[4]*hip[1]+R0[5]*hip[2]+t0[1],
                     R0[6]*hip[0]+R0[7]*hip[1]+R0[8]*hip[2]+t0[2]],
          hipThigh: [Rt[0]*hip[0]+Rt[1]*hip[1]+Rt[2]*hip[2]+tt[0],
                     Rt[3]*hip[0]+Rt[4]*hip[1]+Rt[5]*hip[2]+tt[1],
                     Rt[6]*hip[0]+Rt[7]*hip[1]+Rt[8]*hip[2]+tt[2]]
        };
      }
      if (transforms[2] && transforms[2]._hip && transforms[2]._thighDir) {
        const hip = transforms[2]._hip;
        const Rt = transforms[2].R;
        const tt = transforms[2].t;
        hipAnchorR = {
          hip: hip,
          q_thigh: mat3_to_quat(Rt),
          thighLen: transforms[2]._thighLen,
          thighDir: transforms[2]._thighDir,
          hipTorso: [R0[0]*hip[0]+R0[1]*hip[1]+R0[2]*hip[2]+t0[0],
                     R0[3]*hip[0]+R0[4]*hip[1]+R0[5]*hip[2]+t0[1],
                     R0[6]*hip[0]+R0[7]*hip[1]+R0[8]*hip[2]+t0[2]],
          hipThigh: [Rt[0]*hip[0]+Rt[1]*hip[1]+Rt[2]*hip[2]+tt[0],
                     Rt[3]*hip[0]+Rt[4]*hip[1]+Rt[5]*hip[2]+tt[1],
                     Rt[6]*hip[0]+Rt[7]*hip[1]+Rt[8]*hip[2]+tt[2]]
        };
      }
    }

    let _hipAnchorCount = 0; // diagnostic counter

    for (let i = 0; i < nMesh; i++) {
      const rx = meshRestPos[i*3];
      const ry = meshRestPos[i*3+1];
      const rz = meshRestPos[i*3+2];

      const entries = meshVertRegions[i]; // flat pairs: [rid, w, rid, w, ...]
      let dominantRid = 0, dominantW = -1;
      // Find dominant region first (needed for hard mode and debug color)
      for (let e = 0; e < entries.length; e += 2) {
        if (entries[e + 1] > dominantW) { dominantRid = entries[e]; dominantW = entries[e + 1]; }
      }
      let px, py, pz;

      if (hardRegions) {
        // --- Hard assignment: dominant region only, zero blending ---
        const tf = transforms[dominantRid];
        if (tf) {
          const R = tf.R, t = tf.t;
          px = R[0]*rx + R[1]*ry + R[2]*rz + t[0];
          py = R[3]*rx + R[4]*ry + R[5]*rz + t[1];
          pz = R[6]*rx + R[7]*ry + R[8]*rz + t[2];
        } else { px = rx; py = ry; pz = rz; }
      } else if (useDQS) {
        // --- DQS blend with hip-slerp for thigh verts ---
        // Thigh verts get a position-based slerp (torso→thigh along the bone)
        // instead of DQS weight-blending. This eliminates both the torso/thigh
        // blend artifacts AND the L/R thigh seam stretch, because:
        // - At frac=0 (hip): slerp = torso → matches torso region seamlessly
        // - At frac=1 (knee): slerp = thigh → matches shin region seamlessly
        // - L and R thigh both converge to torso at hip → no L/R seam

        // First pass: identify thigh weight
        let thighW1 = 0, thighW2 = 0, totalTorsoW = 0, totalOtherW = 0;
        for (let e = 0; e < entries.length; e += 2) {
          const rid = entries[e], w = entries[e + 1];
          if (rid === 0) totalTorsoW += w;
          else if (rid === 1) thighW1 += w;
          else if (rid === 2) thighW2 += w;
          else totalOtherW += w;
        }
        const totalThighW = thighW1 + thighW2;
        const thighRid = (thighW1 >= thighW2) ? 1 : 2;
        const slerpAnchor = (thighRid === 1) ? hipAnchorL : hipAnchorR;

        if (slerpAnchor && totalThighW > 0.01 && q_torso) {
          // --- Hip-slerp path ---
          const sd = slerpAnchor.thighDir;
          const ship = slerpAnchor.hip;
          const frac = Math.max(0, Math.min(1,
            ((rx-ship[0])*sd[0]+(ry-ship[1])*sd[1]+(rz-ship[2])*sd[2]) / slerpAnchor.thighLen));
          const sf = frac * frac * (3 - 2 * frac); // smoothstep

          const q = quat_slerp(q_torso, slerpAnchor.q_thigh, sf);

          // Interpolated pivot (torso-mapped hip → thigh-mapped hip)
          const hpx = slerpAnchor.hipTorso[0]*(1-sf) + slerpAnchor.hipThigh[0]*sf;
          const hpy = slerpAnchor.hipTorso[1]*(1-sf) + slerpAnchor.hipThigh[1]*sf;
          const hpz = slerpAnchor.hipTorso[2]*(1-sf) + slerpAnchor.hipThigh[2]*sf;

          // Rotate rest offset from hip by slerped quaternion
          const rot = quat_rotate_vec(q, [rx - ship[0], ry - ship[1], rz - ship[2]]);
          const slerpPx = hpx + rot[0];
          const slerpPy = hpy + rot[1];
          const slerpPz = hpz + rot[2];

          if (totalOtherW < 0.01) {
            // Pure torso+thigh: slerp directly
            px = slerpPx; py = slerpPy; pz = slerpPz;
          } else {
            // Has other regions (shin, etc.): DQS for those, LBS blend with slerp
            let bqr0 = 0, bqr1 = 0, bqr2 = 0, bqr3 = 0;
            let bqd0 = 0, bqd1 = 0, bqd2 = 0, bqd3 = 0;
            let firstQr = null;
            for (let e = 0; e < entries.length; e += 2) {
              const rid = entries[e];
              if (rid === 0 || rid === 1 || rid === 2) continue; // handled by slerp
              const w = entries[e + 1];
              const dq = regionDQs[rid];
              if (!dq) continue;
              let qr = dq.qr, qd = dq.qd;
              if (!firstQr) { firstQr = qr; }
              else {
                const dot = firstQr[0]*qr[0]+firstQr[1]*qr[1]+firstQr[2]*qr[2]+firstQr[3]*qr[3];
                if (dot < 0) { qr = [-qr[0],-qr[1],-qr[2],-qr[3]]; qd = [-qd[0],-qd[1],-qd[2],-qd[3]]; }
              }
              bqr0 += w*qr[0]; bqr1 += w*qr[1]; bqr2 += w*qr[2]; bqr3 += w*qr[3];
              bqd0 += w*qd[0]; bqd1 += w*qd[1]; bqd2 += w*qd[2]; bqd3 += w*qd[3];
            }
            const dqLen = Math.sqrt(bqr0*bqr0+bqr1*bqr1+bqr2*bqr2+bqr3*bqr3);
            if (dqLen > 1e-8) {
              const inv2 = 1/dqLen;
              const nqr = [bqr0*inv2, bqr1*inv2, bqr2*inv2, bqr3*inv2];
              const nqd = [bqd0*inv2, bqd1*inv2, bqd2*inv2, bqd3*inv2];
              const p = dq_apply(nqr, nqd, rx, ry, rz);
              const ttw = totalTorsoW + totalThighW;
              px = ttw * slerpPx + totalOtherW * p[0];
              py = ttw * slerpPy + totalOtherW * p[1];
              pz = ttw * slerpPz + totalOtherW * p[2];
            } else {
              px = slerpPx; py = slerpPy; pz = slerpPz;
            }
          }
        } else {
          // --- Normal DQS blend (no thigh weight) ---
          let bqr0 = 0, bqr1 = 0, bqr2 = 0, bqr3 = 0;
          let bqd0 = 0, bqd1 = 0, bqd2 = 0, bqd3 = 0;
          let firstQr = null;

          for (let e = 0; e < entries.length; e += 2) {
            const rid = entries[e];
            const w   = entries[e + 1];
            const dq  = regionDQs[rid];
            if (!dq) continue;
            let qr = dq.qr, qd = dq.qd;

            if (!firstQr) { firstQr = qr; }
            else {
              const dot = firstQr[0]*qr[0] + firstQr[1]*qr[1] + firstQr[2]*qr[2] + firstQr[3]*qr[3];
              if (dot < 0) {
                qr = [-qr[0], -qr[1], -qr[2], -qr[3]];
                qd = [-qd[0], -qd[1], -qd[2], -qd[3]];
              }
            }

            bqr0 += w * qr[0]; bqr1 += w * qr[1]; bqr2 += w * qr[2]; bqr3 += w * qr[3];
            bqd0 += w * qd[0]; bqd1 += w * qd[1]; bqd2 += w * qd[2]; bqd3 += w * qd[3];
          }

          const len = Math.sqrt(bqr0*bqr0 + bqr1*bqr1 + bqr2*bqr2 + bqr3*bqr3);
          if (len < 1e-8) {
            px = rx; py = ry; pz = rz;
          } else {
            const inv = 1.0 / len;
            const nqr = [bqr0*inv, bqr1*inv, bqr2*inv, bqr3*inv];
            const nqd = [bqd0*inv, bqd1*inv, bqd2*inv, bqd3*inv];
            const p = dq_apply(nqr, nqd, rx, ry, rz);
            px = p[0]; py = p[1]; pz = p[2];
          }
        }
      } else {
        // --- LBS blend ---
        px = 0; py = 0; pz = 0;
        for (let e = 0; e < entries.length; e += 2) {
          const rid = entries[e];
          const w   = entries[e + 1];
          if (w > dominantW) { dominantRid = rid; dominantW = w; }
          const tf  = transforms[rid];
          if (!tf) continue;
          // Twist gradient: blend from no-twist at hip to full twist at knee.
          // Acts as virtual twist bone for thighs (1/2).
          // Eliminates LBS candy-wrapper artifacts from bone-axis twist.
          if (tf.R_noTwist && (rid === 1 || rid === 2)) {
            const piv = tf._hip;
            const d = tf._thighDir;
            const tLen = tf._thighLen;
            const tp = ((rx-piv[0])*d[0] + (ry-piv[1])*d[1] + (rz-piv[2])*d[2]) / tLen;
            const f = Math.max(0, Math.min(1, tp));
            const sf = f * f * (3 - 2 * f); // smoothstep
            const R = tf.R, t = tf.t;
            const Rn = tf.R_noTwist, tn = tf.t_noTwist;
            const fx = Rn[0]*rx+Rn[1]*ry+Rn[2]*rz+tn[0], fy = Rn[3]*rx+Rn[4]*ry+Rn[5]*rz+tn[1], fz = Rn[6]*rx+Rn[7]*ry+Rn[8]*rz+tn[2];
            const tx2 = R[0]*rx+R[1]*ry+R[2]*rz+t[0], ty2 = R[3]*rx+R[4]*ry+R[5]*rz+t[1], tz2 = R[6]*rx+R[7]*ry+R[8]*rz+t[2];
            px += w * (fx + sf * (tx2 - fx));
            py += w * (fy + sf * (ty2 - fy));
            pz += w * (fz + sf * (tz2 - fz));
          } else {
            const R = tf.R, t = tf.t;
            px += w * (R[0]*rx + R[1]*ry + R[2]*rz + t[0]);
            py += w * (R[3]*rx + R[4]*ry + R[5]*rz + t[1]);
            pz += w * (R[6]*rx + R[7]*ry + R[8]*rz + t[2]);
          }
        }
      }

      // ── THIGH JOINT GRADIENT (DISABLED for diagnostic baseline) ──
      // All hip-anchor slerp corrections removed to establish clean baseline.

      // Lerp with alpha stiffness (0 = rest, 1 = fully deformed)
      pos[i*3]   = rx + alpha * (px - rx);
      pos[i*3+1] = ry + alpha * (py - ry);
      pos[i*3+2] = rz + alpha * (pz - rz);

      // Set debug color from dominant region
      if (debugRegions && colors) {
        const c = REGION_COLORS[dominantRid] || [0.5, 0.5, 0.5];
        colors[i*3]   = c[0];
        colors[i*3+1] = c[1];
        colors[i*3+2] = c[2];
      }
    }

    // ═══════════════════════════════════════════════════════════════
    // ONE-SHOT STRETCH DIAGNOSTIC: find WHY crotch verts stretch
    // ═══════════════════════════════════════════════════════════════
    if (!window._stretchDiagDone && transforms[0] && (transforms[1] || transforms[2]) && P.pelvis) {
      window._stretchDiagDone = true;
      const R0 = transforms[0].R, t0 = transforms[0].t;
      const midX = P.pelvis[0]; // body midline X

      // Collect inner thigh verts (near midline, between hip and mid-thigh height)
      const diagVerts = [];
      for (let i = 0; i < nMesh; i++) {
        const rrx = meshRestPos[i*3], rry = meshRestPos[i*3+1], rrz = meshRestPos[i*3+2];
        const entries2 = meshVertRegions[i];
        let tw = 0, tRid = 0, toW = 0;
        for (let e = 0; e < entries2.length; e += 2) {
          if (entries2[e] === 0) toW = entries2[e+1];
          if (entries2[e] === 1 || entries2[e] === 2) { tw = entries2[e+1]; tRid = entries2[e]; }
        }
        if (tw < 0.01) continue; // skip verts with no thigh weight
        // Inner thigh: close to midline (within 3cm of center X)
        if (Math.abs(rrx - midX) > 0.03) continue;
        // Between hip and mid-thigh height
        const hipY = P.l_hip ? P.l_hip[1] : P.r_hip[1];
        const kneeY = P.l_knee ? P.l_knee[1] : P.r_knee[1];
        const midThighY = (hipY + kneeY) / 2;
        if (rry > hipY + 0.02 || rry < midThighY) continue;

        // Compute individual transform results
        const torsoX = R0[0]*rrx + R0[1]*rry + R0[2]*rrz + t0[0];
        const torsoY = R0[3]*rrx + R0[4]*rry + R0[5]*rrz + t0[1];
        const torsoZ = R0[6]*rrx + R0[7]*rry + R0[8]*rrz + t0[2];
        let thighX = rrx, thighY = rry, thighZ = rrz;
        if (transforms[tRid]) {
          const Rt = transforms[tRid].R, tt = transforms[tRid].t;
          thighX = Rt[0]*rrx + Rt[1]*rry + Rt[2]*rrz + tt[0];
          thighY = Rt[3]*rrx + Rt[4]*rry + Rt[5]*rrz + tt[1];
          thighZ = Rt[6]*rrx + Rt[7]*rry + Rt[8]*rrz + tt[2];
        }
        // DQS/LBS result (already in pos array)
        const fx = pos[i*3], fy = pos[i*3+1], fz = pos[i*3+2];

        diagVerts.push({
          i, rest: [rrx, rry, rrz],
          thighW: tw, torsoW: toW, thighRid: tRid,
          torsoPos: [torsoX, torsoY, torsoZ],
          thighPos: [thighX, thighY, thighZ],
          finalPos: [fx, fy, fz],
          // Distance from torso and thigh individual results
          distFromTorso: Math.sqrt((fx-torsoX)**2+(fy-torsoY)**2+(fz-torsoZ)**2),
          distFromThigh: Math.sqrt((fx-thighX)**2+(fy-thighY)**2+(fz-thighZ)**2)
        });
      }

      // Sort by thigh weight descending to see the worst offenders
      diagVerts.sort((a, b) => b.thighW - a.thighW);

      console.log('═══ CROTCH STRETCH DIAGNOSTIC (hip-slerp replaces DQS for thigh) ═══');
      console.log('Inner thigh verts found (near midline, hip-to-mid-thigh): ' + diagVerts.length);

      // Show the top 15 highest thigh-weight crotch verts
      const show = diagVerts.slice(0, 15);
      for (const v of show) {
        // Also compute bone-projection frac for this vert
        let frac = -1;
        if (transforms[v.thighRid] && transforms[v.thighRid]._hip) {
          const hip = transforms[v.thighRid]._hip;
          const d = transforms[v.thighRid]._thighDir;
          const tLen = transforms[v.thighRid]._thighLen;
          frac = ((v.rest[0]-hip[0])*d[0] + (v.rest[1]-hip[1])*d[1] + (v.rest[2]-hip[2])*d[2]) / tLen;
        }
        console.log('  v' + v.i +
          ' thighW=' + v.thighW.toFixed(3) + ' torsoW=' + v.torsoW.toFixed(3) +
          ' rid=' + v.thighRid + ' frac=' + frac.toFixed(3) +
          ' rest=[' + v.rest.map(x=>x.toFixed(4)).join(',') + ']' +
          ' final=[' + v.finalPos.map(x=>x.toFixed(4)).join(',') + ']' +
          ' torso=[' + v.torsoPos.map(x=>x.toFixed(4)).join(',') + ']' +
          ' thigh=[' + v.thighPos.map(x=>x.toFixed(4)).join(',') + ']' +
          ' dTorso=' + v.distFromTorso.toFixed(4) +
          ' dThigh=' + v.distFromThigh.toFixed(4));
      }

      // Summary: weight distribution of crotch verts
      const w100 = diagVerts.filter(v => v.thighW > 0.99).length;
      const w80 = diagVerts.filter(v => v.thighW > 0.80 && v.thighW <= 0.99).length;
      const w60 = diagVerts.filter(v => v.thighW > 0.60 && v.thighW <= 0.80).length;
      const wLow = diagVerts.filter(v => v.thighW <= 0.60).length;
      console.log('  Weight distribution: >99%=' + w100 + ' 80-99%=' + w80 + ' 60-80%=' + w60 + ' <60%=' + wLow);

      // Find worst stretch: max distance between final positions of adjacent crotch verts
      // that are close in rest pose but far in deformed pose
      let worstStretch = 0, worstPair = null;
      for (let a = 0; a < diagVerts.length; a++) {
        for (let b = a + 1; b < Math.min(diagVerts.length, a + 50); b++) {
          const restDist = Math.sqrt(
            (diagVerts[a].rest[0]-diagVerts[b].rest[0])**2 +
            (diagVerts[a].rest[1]-diagVerts[b].rest[1])**2 +
            (diagVerts[a].rest[2]-diagVerts[b].rest[2])**2);
          if (restDist > 0.02) continue; // only check verts close in rest
          const defDist = Math.sqrt(
            (diagVerts[a].finalPos[0]-diagVerts[b].finalPos[0])**2 +
            (diagVerts[a].finalPos[1]-diagVerts[b].finalPos[1])**2 +
            (diagVerts[a].finalPos[2]-diagVerts[b].finalPos[2])**2);
          const stretch = defDist / Math.max(restDist, 0.001);
          if (stretch > worstStretch) {
            worstStretch = stretch;
            worstPair = [diagVerts[a], diagVerts[b], restDist, defDist];
          }
        }
      }
      if (worstPair) {
        console.log('  WORST STRETCH: ' + worstStretch.toFixed(1) + 'x' +
          ' (rest=' + worstPair[2].toFixed(4) + ' def=' + worstPair[3].toFixed(4) + ')');
        console.log('    vertA: v' + worstPair[0].i +
          ' thighW=' + worstPair[0].thighW.toFixed(3) +
          ' torsoW=' + worstPair[0].torsoW.toFixed(3) +
          ' rid=' + worstPair[0].thighRid);
        console.log('    vertB: v' + worstPair[1].i +
          ' thighW=' + worstPair[1].thighW.toFixed(3) +
          ' torsoW=' + worstPair[1].torsoW.toFixed(3) +
          ' rid=' + worstPair[1].thighRid);
      }
      console.log('═══ END STRETCH DIAGNOSTIC ═══');
    }

    meshGeo.attributes.position.needsUpdate = true;
    if (debugRegions && colors) meshGeo.attributes.color.needsUpdate = true;
    meshGeo.computeVertexNormals();
  }

  function updateCageMesh(){
    if(!cageMesh) return;
    const arr = cageMesh.geometry.attributes.position.array;
    for(let i=0;i<cageRest.length;i++){
      arr[i*3]   = cagePos[i*3];
      arr[i*3+1] = cagePos[i*3+1];
      arr[i*3+2] = cagePos[i*3+2];
    }
    cageMesh.geometry.attributes.position.needsUpdate = true;
  }

  function applyCageSmoothing(){
    if(!ENABLE_CAGE_SMOOTHING || CAGE_SMOOTHING <= 0) return;
    if(!cageSmoothed || cageSmoothed.length !== cagePos.length){
      cageSmoothed = new Float32Array(cagePos.length);
      cageSmoothed.set(cagePos);
    }
    const s = clamp(CAGE_SMOOTHING, 0, 1);
    for(let i=0;i<cagePos.length;i++){
      cageSmoothed[i] = cageSmoothed[i] * (1 - s) + cagePos[i] * s;
      cagePos[i] = cageSmoothed[i];
    }
  }

  // ---------------- Foot plant correction ----------------
  function footLockCorrection(t){
    const plantStrength = +$('plant').value;
    if(plantStrength <= 0) { lockL=null; lockR=null; return; }

    const stanceThresh = +$('stanceKnee').value; // degrees
    const lKdeg = Math.abs(getRotationAtTime('l_knee', t, 'x_rot') * 180/Math.PI);
    const rKdeg = Math.abs(getRotationAtTime('r_knee', t, 'x_rot') * 180/Math.PI);

    const lStance = lKdeg < stanceThresh;
    const rStance = rKdeg < stanceThresh;

    const lFootC = regionCenter(R.l_foot);
    const rFootC = regionCenter(R.r_foot);

    if(lStance && !lockL) lockL = lFootC.slice();
    if(!lStance) lockL = null;

    if(rStance && !lockR) lockR = rFootC.slice();
    if(!rStance) lockR = null;

    if(lockL){
      const dx = (lockL[0]-lFootC[0]) * plantStrength;
      const dy = (lockL[1]-lFootC[1]) * plantStrength;
      const dz = (lockL[2]-lFootC[2]) * plantStrength;
      translate([].concat(R.l_thigh, R.l_shin, R.l_foot), dx,dy,dz);
    }
    if(lockR){
      const dx = (lockR[0]-rFootC[0]) * plantStrength;
      const dy = (lockR[1]-rFootC[1]) * plantStrength;
      const dz = (lockR[2]-rFootC[2]) * plantStrength;
      translate([].concat(R.r_thigh, R.r_shin, R.r_foot), dx,dy,dz);
    }
  }

  // ---------------- Foot plant correction (FBX position mode) ----------------
  // Detects planted feet via world-space ankle velocity from FBX joint data.
  // When a foot has near-zero velocity, locks it in place to prevent sliding.
  function footLockCorrectionFBX(t) {
    const plantStrength = +$('plant').value;
    if (plantStrength <= 0 || !jointPosData) { lockL = null; lockR = null; return; }
    if (!R.l_foot || !R.r_foot) return;

    const dur = jointPosData.duration;
    const dt = Math.min(0.02, dur * 0.01); // sample offset for velocity estimation

    // World-space ankle: root_position + hips-centered ankle
    const root0 = getRootPositionAtTime(t);
    const root1 = getRootPositionAtTime(t - dt);
    const lAnk0 = getJointPositionAtTime('l_ankle', t);
    const lAnk1 = getJointPositionAtTime('l_ankle', t - dt);
    const rAnk0 = getJointPositionAtTime('r_ankle', t);
    const rAnk1 = getJointPositionAtTime('r_ankle', t - dt);
    if (!root0 || !root1 || !lAnk0 || !lAnk1 || !rAnk0 || !rAnk1) return;

    // World-space velocities (per second)
    const lVx = ((root0[0]+lAnk0[0])-(root1[0]+lAnk1[0])) / dt;
    const lVy = ((root0[1]+lAnk0[1])-(root1[1]+lAnk1[1])) / dt;
    const lVz = ((root0[2]+lAnk0[2])-(root1[2]+lAnk1[2])) / dt;
    const rVx = ((root0[0]+rAnk0[0])-(root1[0]+rAnk1[0])) / dt;
    const rVy = ((root0[1]+rAnk0[1])-(root1[1]+rAnk1[1])) / dt;
    const rVz = ((root0[2]+rAnk0[2])-(root1[2]+rAnk1[2])) / dt;
    const lSpeed = Math.sqrt(lVx*lVx + lVy*lVy + lVz*lVz);
    const rSpeed = Math.sqrt(rVx*rVx + rVy*rVy + rVz*rVz);

    // Threshold: 5% of mesh height per second = practically stationary
    const velThresh = meshHeight * 0.05;
    const lStance = lSpeed < velThresh;
    const rStance = rSpeed < velThresh;

    const lFootC = regionCenter(R.l_foot);
    const rFootC = regionCenter(R.r_foot);

    if (lStance && !lockL) lockL = lFootC.slice();
    if (!lStance) lockL = null;
    if (rStance && !lockR) lockR = rFootC.slice();
    if (!rStance) lockR = null;

    if (lockL) {
      const dx = (lockL[0]-lFootC[0]) * plantStrength;
      const dy = (lockL[1]-lFootC[1]) * plantStrength;
      const dz = (lockL[2]-lFootC[2]) * plantStrength;
      translate([].concat(R.l_thigh, R.l_shin, R.l_foot), dx, dy, dz);
    }
    if (lockR) {
      const dx = (lockR[0]-rFootC[0]) * plantStrength;
      const dy = (lockR[1]-rFootC[1]) * plantStrength;
      const dz = (lockR[2]-rFootC[2]) * plantStrength;
      translate([].concat(R.r_thigh, R.r_shin, R.r_foot), dx, dy, dz);
    }
  }

  // ---------------- Cage driver: Mixamo -> cage ----------------
  function driveCageFromMixamo(t){
    // Spine direction (hoisted for head correction — set in position-mode body rotation)
    let restSpineDir = [0, 1, 0], curSpineDir = [0, 1, 0];
    // reset cage to rest
    for(let i=0;i<cageRest.length;i++){
      cagePos[i*3]=cageRest[i][0];
      cagePos[i*3+1]=cageRest[i][1];
      cagePos[i*3+2]=cageRest[i][2];
    }

    // All cage vertex indices for body-wide transforms
    const allIdx = [].concat(
      R.torso,
      R.l_thigh, R.r_thigh,
      R.l_shin, R.r_shin,
      R.l_foot, R.r_foot,
      R.l_upperarm, R.r_upperarm,
      R.l_forearm, R.r_forearm,
      R.l_hand || [], R.r_hand || [],
      R.head
    );

    // Pelvis translation delta (stored for analytical pivot placement)
    var pelvisDelta = [0, 0, 0];

    // ----- BODY motion (affects everything) -----
    if ($('enableBody').checked && P.pelvis) {
      if (jointPosData) {
        // Position mode: body ROTATIONS are already embedded in the hips-centered
        // joint positions (we only subtracted translation, not rotation).
        // Only apply pelvis TRANSLATION delta (bob/sway) from FBX root positions.
        const rootPos = getRootPositionAtTime(t);
        const restRoot = jointPosData.root_positions[0];
        if (rootPos && restRoot) {
          const hipBobScale = +$('hipBob').value;
          const hipSwayScale = +$('hipSway').value;
          // Scale by leg proportion ratio: if cage legs are shorter than FBX legs,
          // reduce pelvis bob/sway proportionally so the movement fits the cage's
          // actual leg reach and doesn't drive feet through the floor.
          const pr = fbxLegPropRatio;
          const dy = (rootPos[1] - restRoot[1]) * hipBobScale * pr;
          const dx = (rootPos[0] - restRoot[0]) * hipSwayScale * pr;
          const dz = (rootPos[2] - restRoot[2]) * hipSwayScale * pr;
          pelvisDelta = [dx, dy, dz];
          translate(allIdx, dx, dy, dz);
        }
      } else if (spineMotionData) {
        // Euler fallback: body motion from spine_keyframes.json
        const hipBobScale = +$('hipBob').value;
        const hipSwayScale = +$('hipSway').value;
        const forwardLean = +$('forwardLean').value * Math.PI/180;
        const spineTwistScale = +$('spineTwist').value;

        const rawBob = getSpineDataAtTime('hips', t, 'y_trans');
        const hipBob = (rawBob - 0.4) * hipBobScale;

        const hipSwayZ = getSpineDataAtTime('hips', t, 'z_rot') * Math.PI/180 * hipSwayScale;
        const hipTwistY = getSpineDataAtTime('hips', t, 'y_rot') * Math.PI/180 * hipSwayScale;
        const spineTwist = getSpineDataAtTime('spine1', t, 'y_rot') * Math.PI/180 * spineTwistScale;

        applyRotX(allIdx, P.pelvis, forwardLean);
        applyRotZ(allIdx, P.pelvis, hipSwayZ);
        applyRotY(allIdx, P.pelvis, hipTwistY);
        translate(allIdx, 0, hipBob, 0);

        // upper body counter twist (torso + head + arms)
        const upperIdx = [].concat(R.torso, R.head, R.l_upperarm, R.r_upperarm, R.l_forearm, R.r_forearm, R.l_hand || [], R.r_hand || []);
        const spineP = [P.pelvis[0], P.pelvis[1]+0.10, P.pelvis[2]];
        applyRotY(upperIdx, spineP, spineTwist);
      }
    }

    // ═══════════════════════════════════════════════════════════
    // POSITION-BASED DRIVING (when joint_positions.json is loaded)
    // Bypasses all Euler decomposition — uses hips-local joint positions
    // directly, with 2-segment IK for each limb.
    // ═══════════════════════════════════════════════════════════
    if (jointPosData) {
      // Get hips-local positions for all joints at time t
      const jp = {};
      for (const jn of Object.keys(jointPosData.joints)) {
        jp[jn] = getJointPositionAtTime(jn, t);
      }

      // --- Body rotation from FBX spine direction ---
      // Compute rotation from rest spine direction to current spine direction.
      // Apply to ALL cage verts so torso Kabsch shares body rotation with limbs,
      // preventing transform discontinuity at body-limb boundaries.
      let bodyAxis = [0, 1, 0], bodyAngle = 0;
      const rpChest = jointPosData.rest_pose.chest;
      const rpPelvis = jointPosData.rest_pose.pelvis || [0, 0, 0];
      if (rpChest && jp.chest) {
        restSpineDir = vnorm(vsub(rpChest, rpPelvis));
        curSpineDir = vnorm(vsub(jp.chest, jp.pelvis || [0, 0, 0]));
        const bAxis = vcross(restSpineDir, curSpineDir);
        const bAxisLen = vlen(bAxis);
        const bDot = Math.max(-1, Math.min(1, vdot(restSpineDir, curSpineDir)));
        bodyAngle = Math.acos(bDot);
        if (bAxisLen > 1e-6 && Math.abs(bodyAngle) > 1e-4) {
          bodyAxis = vnorm(bAxis);
          applyRotAxis(allIdx, P.pelvis, bodyAxis[0], bodyAxis[1], bodyAxis[2], bodyAngle);
        } else {
          bodyAngle = 0;
        }
      }

      // --- Spine twist from shoulder axis ---
      // After lean, extract the twist component: rotation around the spine axis
      // detected from the change in shoulder-to-shoulder direction.
      let twistAngle = 0;
      const rpLS_tw = jointPosData.rest_pose.l_shoulder;
      const rpRS_tw = jointPosData.rest_pose.r_shoulder;
      if (rpLS_tw && rpRS_tw && jp.l_shoulder && jp.r_shoulder) {
        const restShAxis = vnorm(vsub(rpRS_tw, rpLS_tw));
        const curShAxis = vnorm(vsub(jp.r_shoulder, jp.l_shoulder));
        // Lean-rotate the rest shoulder axis
        let leanedRestSh = restShAxis;
        if (Math.abs(bodyAngle) > 1e-4) {
          const r = rotateAroundAxis(restShAxis[0], restShAxis[1], restShAxis[2], 0, 0, 0,
            bodyAxis[0], bodyAxis[1], bodyAxis[2], bodyAngle);
          leanedRestSh = vnorm(r);
        }
        // Project both onto plane perpendicular to current spine direction
        const dotR = vdot(leanedRestSh, curSpineDir);
        const dotC = vdot(curShAxis, curSpineDir);
        let projR = vsub(leanedRestSh, vscale(curSpineDir, dotR));
        let projC = vsub(curShAxis, vscale(curSpineDir, dotC));
        if (vlen(projR) > 1e-6 && vlen(projC) > 1e-6) {
          projR = vnorm(projR);
          projC = vnorm(projC);
          const dotP = Math.max(-1, Math.min(1, vdot(projR, projC)));
          twistAngle = Math.acos(dotP);
          const cross = vcross(projR, projC);
          if (vdot(cross, curSpineDir) < 0) twistAngle = -twistAngle;
          // Apply twist scale from UI slider
          const twistScale = +$('spineTwist').value;
          twistAngle *= twistScale;
        }
      }
      // Apply spine twist to upper body cage verts only (not legs)
      if (Math.abs(twistAngle) > 1e-4) {
        const upperIdx = [].concat(R.torso, R.head,
          R.l_upperarm, R.r_upperarm, R.l_forearm, R.r_forearm,
          R.l_hand || [], R.r_hand || []);
        applyRotAxis(upperIdx, P.pelvis, curSpineDir[0], curSpineDir[1], curSpineDir[2], twistAngle);
      }

      // --- Hip twist from hip-to-hip axis ---
      // Same approach as spine twist but from pelvis rotation detected via hip joint axis.
      // Applied to lower body only (thighs, shins, feet).
      let hipTwistAngle = 0;
      const rpLH_tw = jointPosData.rest_pose.l_hip;
      const rpRH_tw = jointPosData.rest_pose.r_hip;
      if (rpLH_tw && rpRH_tw && jp.l_hip && jp.r_hip) {
        const restHipAxis = vnorm(vsub(rpRH_tw, rpLH_tw));
        const curHipAxis = vnorm(vsub(jp.r_hip, jp.l_hip));
        let leanedRestHip = restHipAxis;
        if (Math.abs(bodyAngle) > 1e-4) {
          const r = rotateAroundAxis(restHipAxis[0], restHipAxis[1], restHipAxis[2], 0, 0, 0,
            bodyAxis[0], bodyAxis[1], bodyAxis[2], bodyAngle);
          leanedRestHip = vnorm(r);
        }
        const dotR = vdot(leanedRestHip, curSpineDir);
        const dotC = vdot(curHipAxis, curSpineDir);
        let projR = vsub(leanedRestHip, vscale(curSpineDir, dotR));
        let projC = vsub(curHipAxis, vscale(curSpineDir, dotC));
        if (vlen(projR) > 1e-6 && vlen(projC) > 1e-6) {
          projR = vnorm(projR);
          projC = vnorm(projC);
          const dotP = Math.max(-1, Math.min(1, vdot(projR, projC)));
          hipTwistAngle = Math.acos(dotP);
          const cross = vcross(projR, projC);
          if (vdot(cross, curSpineDir) < 0) hipTwistAngle = -hipTwistAngle;
          const twistScale = +$('spineTwist').value;
          hipTwistAngle *= twistScale;
        }
      }
      // Apply hip twist to lower body cage verts
      if (Math.abs(hipTwistAngle) > 1e-4) {
        const lowerIdx = [].concat(R.l_thigh, R.r_thigh, R.l_shin, R.r_shin, R.l_foot, R.r_foot);
        applyRotAxis(lowerIdx, P.pelvis, curSpineDir[0], curSpineDir[1], curSpineDir[2], hipTwistAngle);
      }

      // Smooth hip-spine twist transition on lower torso cage verts.
      // Without this, there's a hard twist discontinuity between torso (spine twist
      // only) and legs (hip twist). For Capoeira with -0.15° spine vs -7° hip twist,
      // the 6.9° gap causes mesh tearing at the butt/crotch boundary.
      // Fix: blend the hip-spine difference into lower torso cage verts using a
      // height-based gradient, so torso Kabsch naturally captures some hip twist.
      const hipSpineDelta = hipTwistAngle - twistAngle;
      if (Math.abs(hipSpineDelta) > 1e-4 && R.torso) {
        const pelvisRestY = P.pelvis[1];
        const gradientH = 0.12; // blend over ~12cm above pelvis
        for (let i = 0; i < R.torso.length; i++) {
          const ci = R.torso[i];
          const restY = cageRest[ci][1];
          if (restY < pelvisRestY + gradientH) {
            const f = clamp(1 - (restY - pelvisRestY) / gradientH, 0, 1);
            const sf = f * f * (3 - 2 * f); // smoothstep
            const partialTwist = hipSpineDelta * sf;
            if (Math.abs(partialTwist) > 1e-5) {
              const r = rotateAroundAxis(cagePos[ci*3], cagePos[ci*3+1], cagePos[ci*3+2],
                P.pelvis[0], P.pelvis[1], P.pelvis[2],
                curSpineDir[0], curSpineDir[1], curSpineDir[2], partialTwist);
              cagePos[ci*3] = r[0]; cagePos[ci*3+1] = r[1]; cagePos[ci*3+2] = r[2];
            }
          }
        }
      }

      // Helper: rotate a direction vector by body rotation (lean + twist).
      // Uses spine twist for upper body (arms), hip twist for lower body (legs).
      function bodyRotDir(dir, isLeg) {
        let d = dir;
        if (Math.abs(bodyAngle) > 1e-4) {
          const r = rotateAroundAxis(d[0], d[1], d[2], 0, 0, 0,
            bodyAxis[0], bodyAxis[1], bodyAxis[2], bodyAngle);
          d = vnorm(r);
        }
        const tw = isLeg ? hipTwistAngle : twistAngle;
        if (Math.abs(tw) > 1e-4) {
          const r = rotateAroundAxis(d[0], d[1], d[2], 0, 0, 0,
            curSpineDir[0], curSpineDir[1], curSpineDir[2], tw);
          d = vnorm(r);
        }
        return d;
      }

      // One-shot diagnostic: log sample FBX positions at first frame
      if (!window._posDiagDone) {
        window._posDiagDone = true;
        console.log('=== POS-DRIVE DIAGNOSTIC at t=' + t.toFixed(3) + ' ===');
        console.log('Body rotation: angle=' + (bodyAngle * 180 / Math.PI).toFixed(2) + '° axis=[' + bodyAxis.map(v=>v.toFixed(3)).join(',') + ']');
        console.log('Spine twist: angle=' + (twistAngle * 180 / Math.PI).toFixed(2) + '°');
        console.log('Hip twist: angle=' + (hipTwistAngle * 180 / Math.PI).toFixed(2) + '°');
        console.log('FBX joint positions (hips-centered, X-mirrored):');
        for (const jn of ['pelvis','l_hip','l_knee','l_ankle','r_hip','r_knee','r_ankle','l_shoulder','l_elbow','l_wrist','r_shoulder','r_elbow','r_wrist','chest','neck','head']) {
          if (jp[jn]) console.log('  ' + jn + ': [' + jp[jn].map(v => v.toFixed(4)).join(', ') + ']');
        }
        console.log('Cage pivots (rest pose):');
        for (const pn of ['pelvis','l_hip','l_knee','l_ankle','r_hip','r_knee','r_ankle','l_shoulder','l_elbow','l_wrist','r_shoulder','r_elbow','r_wrist','neck','head']) {
          if (P[pn]) console.log('  P.' + pn + ': [' + P[pn].map(v => v.toFixed(4)).join(', ') + ']');
        }
        // Root position delta
        const rootPos = getRootPositionAtTime(t);
        const restRoot = jointPosData.root_positions[0];
        if (rootPos && restRoot) {
          console.log('Root pos at t: [' + rootPos.map(v=>v.toFixed(4)).join(', ') + ']');
          console.log('Root pos rest: [' + restRoot.map(v=>v.toFixed(4)).join(', ') + ']');
          console.log('Root delta: [' + [rootPos[0]-restRoot[0], rootPos[1]-restRoot[1], rootPos[2]-restRoot[2]].map(v=>v.toFixed(4)).join(', ') + ']');
        }
        // Leg directions + delta retargeting info (now using T-pose as rest)
        if (jp.l_hip && jp.l_knee) {
          const fbxCur = vnorm(vsub(jp.l_knee, jp.l_hip));
          const fpLH = jointPosData.rest_pose.l_hip, fpLK = jointPosData.rest_pose.l_knee;
          const fbxTpose = fpLH && fpLK ? vnorm(vsub(fpLK, fpLH)) : null;
          console.log('FBX l_thigh T-pose: [' + (fbxTpose ? fbxTpose.map(v=>v.toFixed(4)).join(', ') : 'N/A') + ']');
          console.log('FBX l_thigh cur:    [' + fbxCur.map(v=>v.toFixed(4)).join(', ') + ']');
          if (fbxTpose) {
            const deltaAng = Math.acos(Math.max(-1, Math.min(1, vdot(fbxTpose, fbxCur)))) * 180/Math.PI;
            console.log('FBX l_thigh delta (from T-pose): ' + deltaAng.toFixed(1) + '°');
          }
        }
        const restLegDirL = P.l_knee ? vnorm(vsub(P.l_knee, P.l_hip)) : null;
        if (restLegDirL) console.log('Cage rest leg dir L: [' + restLegDirL.map(v=>v.toFixed(4)).join(', ') + ']');
        // Show retargeted result
        if (jp.l_hip && jp.l_knee && restLegDirL) {
          const fpLH = jointPosData.rest_pose.l_hip, fpLK = jointPosData.rest_pose.l_knee;
          if (fpLH && fpLK) {
            const fbxTpose = vnorm(vsub(fpLK, fpLH));
            const fbxCur = vnorm(vsub(jp.l_knee, jp.l_hip));
            const retargeted = applyDeltaRotation(fbxTpose, fbxCur, restLegDirL);
            const retargetAng = Math.acos(Math.max(-1, Math.min(1, vdot(restLegDirL, retargeted)))) * 180/Math.PI;
            console.log('Retargeted l_thigh dir: [' + retargeted.map(v=>v.toFixed(4)).join(', ') + '] (delta=' + retargetAng.toFixed(1) + '° from cage rest)');
          }
        }
        // Show knee bend angles from FBX and cage rest (to verify fix)
        if (jp.l_hip && jp.l_knee && jp.l_ankle) {
          const thighD = vnorm(vsub(jp.l_knee, jp.l_hip));
          const shinD = vnorm(vsub(jp.l_ankle, jp.l_knee));
          const kneeBend = Math.acos(Math.max(-1, Math.min(1, vdot(thighD, shinD)))) * 180/Math.PI;
          console.log('FBX l_knee bend angle: ' + kneeBend.toFixed(1) + '°');
          // Cage rest knee angle (should be used as reference, NOT added to animation bend)
          const cThighD = vnorm(vsub(P.l_knee, P.l_hip));
          const cShinD = vnorm(vsub(P.l_ankle, P.l_knee));
          const cageKnee = Math.acos(Math.max(-1, Math.min(1, vdot(cThighD, cShinD)))) * 180/Math.PI;
          console.log('Cage rest L knee pre-bend: ' + cageKnee.toFixed(1) + '° → R_bend delta = ' + (kneeBend - cageKnee).toFixed(1) + '°');
          const cThighDR = vnorm(vsub(P.r_knee, P.r_hip));
          const cShinDR = vnorm(vsub(P.r_ankle, P.r_knee));
          const cageKneeR = Math.acos(Math.max(-1, Math.min(1, vdot(cThighDR, cShinDR)))) * 180/Math.PI;
          console.log('Cage rest R knee pre-bend: ' + cageKneeR.toFixed(1) + '°');
        }
        // RIGHT leg diagnostics (to match left leg above)
        if (jp.r_hip && jp.r_knee) {
          const fbxCurR = vnorm(vsub(jp.r_knee, jp.r_hip));
          const fpRH = jointPosData.rest_pose.r_hip, fpRK = jointPosData.rest_pose.r_knee;
          const fbxTposeR = fpRH && fpRK ? vnorm(vsub(fpRK, fpRH)) : null;
          console.log('FBX r_thigh T-pose: [' + (fbxTposeR ? fbxTposeR.map(v=>v.toFixed(4)).join(', ') : 'N/A') + ']');
          console.log('FBX r_thigh cur:    [' + fbxCurR.map(v=>v.toFixed(4)).join(', ') + ']');
          if (fbxTposeR) {
            const deltaAngR = Math.acos(Math.max(-1, Math.min(1, vdot(fbxTposeR, fbxCurR)))) * 180/Math.PI;
            console.log('FBX r_thigh delta (from T-pose): ' + deltaAngR.toFixed(1) + '°');
          }
        }
        const restLegDirR_diag = P.r_knee ? vnorm(vsub(P.r_knee, P.r_hip)) : null;
        if (restLegDirR_diag) console.log('Cage rest leg dir R: [' + restLegDirR_diag.map(v=>v.toFixed(4)).join(', ') + ']');
        if (jp.r_hip && jp.r_knee && restLegDirR_diag) {
          const fpRH = jointPosData.rest_pose.r_hip, fpRK = jointPosData.rest_pose.r_knee;
          if (fpRH && fpRK) {
            const fbxTposeR = vnorm(vsub(fpRK, fpRH));
            const fbxCurR = vnorm(vsub(jp.r_knee, jp.r_hip));
            const retargetedR = applyDeltaRotation(fbxTposeR, fbxCurR, restLegDirR_diag);
            const retargetAngR = Math.acos(Math.max(-1, Math.min(1, vdot(restLegDirR_diag, retargetedR)))) * 180/Math.PI;
            console.log('Retargeted r_thigh dir: [' + retargetedR.map(v=>v.toFixed(4)).join(', ') + '] (delta=' + retargetAngR.toFixed(1) + '° from cage rest)');
          }
        }
        if (jp.r_hip && jp.r_knee && jp.r_ankle) {
          const thighDR = vnorm(vsub(jp.r_knee, jp.r_hip));
          const shinDR = vnorm(vsub(jp.r_ankle, jp.r_knee));
          const kneeBendR = Math.acos(Math.max(-1, Math.min(1, vdot(thighDR, shinDR)))) * 180/Math.PI;
          const cThDR2 = vnorm(vsub(P.r_knee, P.r_hip));
          const cShDR2 = vnorm(vsub(P.r_ankle, P.r_knee));
          const cageKneeR2 = Math.acos(Math.max(-1, Math.min(1, vdot(cThDR2, cShDR2)))) * 180/Math.PI;
          console.log('FBX r_knee bend angle: ' + kneeBendR.toFixed(1) + '° → R_bend delta = ' + (kneeBendR - cageKneeR2).toFixed(1) + '°');
        }
        // Leg length comparison: asymmetric cage legs cause foot height differences
        const lThighLen = vlen(vsub(P.l_knee, P.l_hip));
        const lShinLen  = vlen(vsub(P.l_ankle, P.l_knee));
        const rThighLen = vlen(vsub(P.r_knee, P.r_hip));
        const rShinLen  = vlen(vsub(P.r_ankle, P.r_knee));
        console.log('Cage leg lengths: L=' + (lThighLen+lShinLen).toFixed(4) +
          ' (thigh=' + lThighLen.toFixed(4) + ' shin=' + lShinLen.toFixed(4) +
          ') R=' + (rThighLen+rShinLen).toFixed(4) +
          ' (thigh=' + rThighLen.toFixed(4) + ' shin=' + rShinLen.toFixed(4) +
          ') diff=' + ((lThighLen+lShinLen)-(rThighLen+rShinLen)).toFixed(4));
      }

      // Update FBX joint debug spheres: show FBX skeleton mapped to cage space.
      // Each FBX joint is hips-centered, so we place it at P.pelvis + jp[joint].
      // If the skeleton matches the cage, spheres align with mesh joints.
      const showFBXJoints = $('debugFBXJoints').checked;
      const pelvis = P.pelvis || [0, 0, 0];
      for (const jn of fbxJointNames) {
        const s = fbxJointSpheres[jn];
        if (!s) continue;
        if (showFBXJoints && jp[jn]) {
          s.position.set(
            pelvis[0] + jp[jn][0],
            pelvis[1] + jp[jn][1],
            pelvis[2] + jp[jn][2]
          );
          s.visible = true;
        } else {
          s.visible = false;
        }
      }

      // ----- LEGS from positions (DELTA RETARGETED) -----
      // Uses T-pose bind pose as FBX rest reference. Computes animation deltas
      // (T-pose → current frame) and applies them to cage rest directions.
      // This preserves the cage's own proportions while transferring the animation.
      const rp = jointPosData.rest_pose;  // T-pose (bind pose) positions

      let lThighTwist = 0, rThighTwist = 0;
      // Include pelvis translation in initial hip positions (cage verts had translation applied)
      let movedHipL = vadd(P.l_hip, pelvisDelta), movedHipR = vadd(P.r_hip, pelvisDelta);

      if (jp.l_hip && jp.l_knee && jp.l_ankle
          && vlen(vsub(P.l_knee, P.l_hip)) > 1e-6 && vlen(vsub(P.l_ankle, P.l_knee)) > 1e-6) {
        // Compute hip position after pelvis translation + body lean + hip twist
        movedHipL = vadd(P.l_hip, pelvisDelta);
        if (Math.abs(bodyAngle) > 1e-4 || Math.abs(hipTwistAngle) > 1e-4) {
          movedHipL = [P.l_hip[0]+pelvisDelta[0], P.l_hip[1]+pelvisDelta[1], P.l_hip[2]+pelvisDelta[2]];
          if (Math.abs(bodyAngle) > 1e-4) {
            movedHipL = rotateAroundAxis(movedHipL[0], movedHipL[1], movedHipL[2],
              P.pelvis[0], P.pelvis[1], P.pelvis[2], bodyAxis[0], bodyAxis[1], bodyAxis[2], bodyAngle);
          }
          if (Math.abs(hipTwistAngle) > 1e-4) {
            movedHipL = rotateAroundAxis(movedHipL[0], movedHipL[1], movedHipL[2],
              P.pelvis[0], P.pelvis[1], P.pelvis[2], curSpineDir[0], curSpineDir[1], curSpineDir[2], hipTwistAngle);
          }
        }
        const restLegDirL = bodyRotDir(vnorm(vsub(P.l_knee, P.l_hip)), true);
        const lThighLen = vlen(vsub(P.l_knee, P.l_hip));
        const lShinLen  = vlen(vsub(P.l_ankle, P.l_knee));

        // FBX bind-pose thigh direction (legs keep FBX bind reference because
        // cage and FBX have structurally different thigh angles)
        const fbxTPoseThighL = vnorm(vsub(rp.l_knee, rp.l_hip));

        // FBX current (animated) directions
        const fbxCurThighL = vnorm(vsub(jp.l_knee, jp.l_hip));
        const fbxCurShinL  = vnorm(vsub(jp.l_ankle, jp.l_knee));

        // Thigh twist computed later (position-based, in analytical section)

        // Delta-retarget THIGH: apply (T-pose → current) rotation to cage rest thigh
        // Body-rotate FBX T-pose so delta captures only pelvis-LOCAL leg motion.
        // Without this, body lean is double-counted: once in restLegDirL (via bodyRotDir)
        // and again in the FBX world-space delta (which includes pelvis tilt).
        const fbxTPoseThighL_rot = bodyRotDir(fbxTPoseThighL, true);
        const targetThighDirL = applyDeltaRotation(fbxTPoseThighL_rot, fbxCurThighL, restLegDirL);

        // KNEE BEND: use FBX animation's knee bend angle directly (NOT delta-retarget shin).
        // The cage has pre-existing knee angles in rest pose (up to 48° on left leg!).
        // Independently retargeting shin would ADD FBX bend on top of cage rest angle.
        // Instead: take FBX knee bend angle + map bend axis to cage frame.
        const fbxKneeBendL = Math.acos(Math.max(-1, Math.min(1, vdot(fbxCurThighL, fbxCurShinL))));
        const fbxKneeBendAxisL = stableBendAxis(fbxCurThighL, fbxCurShinL, 'lKneeV', [1, 0, 0]);

        // Map FBX bend axis to cage thigh frame
        const cageKneeBendAxisL = applyDeltaRotation(fbxCurThighL, targetThighDirL, fbxKneeBendAxisL);

        // Compute shin direction by bending from retargeted thigh direction
        let targetShinDirL = targetThighDirL;
        if (Math.abs(fbxKneeBendL) > 1e-4) {
          const rf = rotateAroundAxis(targetThighDirL[0], targetThighDirL[1], targetThighDirL[2],
            0, 0, 0, cageKneeBendAxisL[0], cageKneeBendAxisL[1], cageKneeBendAxisL[2], fbxKneeBendL);
          targetShinDirL = vnorm(rf);
        }

        // Construct target positions using cage segment lengths
        const lKneeW  = vadd(movedHipL, vscale(targetThighDirL, lThighLen));
        const lAnkleW = vadd(lKneeW, vscale(targetShinDirL, lShinLen));

        const lLegResult = driveLegFromPositions_world(
          movedHipL, lKneeW, lAnkleW,
          movedHipL, restLegDirL,
          lThighLen, lShinLen,
          R.l_thigh, R.l_shin, R.l_foot
        );

        // Ankle/foot rotation DELTA from T-pose (not absolute angle!)
        // In T-pose, shin is vertical and foot is horizontal (~60-90° angle).
        // We only want the CHANGE from T-pose ankle orientation.
        if (lLegResult && jp.l_toe && rp.l_ankle && rp.l_knee && rp.l_toe) {
          const fbxRestShinL = vnorm(vsub(rp.l_ankle, rp.l_knee));
          const fbxRestFootL = vnorm(vsub(rp.l_toe, rp.l_ankle));
          const fbxCurFootL = vnorm(vsub(jp.l_toe, jp.l_ankle));
          // Expected foot dir if ankle didn't move (foot follows shin rotation)
          const expectedFootL = applyDeltaRotation(fbxRestShinL, fbxCurShinL, fbxRestFootL);
          // Ankle-LOCAL delta: expected → actual
          const ankleDeltaL = Math.acos(Math.max(-1, Math.min(1, vdot(expectedFootL, fbxCurFootL))));
          if (Math.abs(ankleDeltaL) > 1e-4) {
            // Use raw cross product (NOT stableBendAxis) — ankle flex direction
            // legitimately reverses between dorsi/plantarflexion, sign must be preserved
            let ankleAxisL = vcross(expectedFootL, fbxCurFootL);
            if (vlen(ankleAxisL) > 1e-6) {
              ankleAxisL = vnorm(ankleAxisL);
              const cageAnkleAxisL = applyDeltaRotation(fbxCurShinL, targetShinDirL, ankleAxisL);
              applyRotAxis(R.l_foot, lLegResult.anklePivot,
                cageAnkleAxisL[0], cageAnkleAxisL[1], cageAnkleAxisL[2], ankleDeltaL);
            }
          }
        }

        // Thigh twist applied via analytical lThighR in legState (not cage verts)
      }

      if (jp.r_hip && jp.r_knee && jp.r_ankle
          && vlen(vsub(P.r_knee, P.r_hip)) > 1e-6 && vlen(vsub(P.r_ankle, P.r_knee)) > 1e-6) {
        // Compute hip position after pelvis translation + body lean + hip twist
        movedHipR = vadd(P.r_hip, pelvisDelta);
        if (Math.abs(bodyAngle) > 1e-4 || Math.abs(hipTwistAngle) > 1e-4) {
          movedHipR = [P.r_hip[0]+pelvisDelta[0], P.r_hip[1]+pelvisDelta[1], P.r_hip[2]+pelvisDelta[2]];
          if (Math.abs(bodyAngle) > 1e-4) {
            movedHipR = rotateAroundAxis(movedHipR[0], movedHipR[1], movedHipR[2],
              P.pelvis[0], P.pelvis[1], P.pelvis[2], bodyAxis[0], bodyAxis[1], bodyAxis[2], bodyAngle);
          }
          if (Math.abs(hipTwistAngle) > 1e-4) {
            movedHipR = rotateAroundAxis(movedHipR[0], movedHipR[1], movedHipR[2],
              P.pelvis[0], P.pelvis[1], P.pelvis[2], curSpineDir[0], curSpineDir[1], curSpineDir[2], hipTwistAngle);
          }
        }
        const restLegDirR = bodyRotDir(vnorm(vsub(P.r_knee, P.r_hip)), true);
        const rThighLen = vlen(vsub(P.r_knee, P.r_hip));
        const rShinLen  = vlen(vsub(P.r_ankle, P.r_knee));

        const fbxTPoseThighR = vnorm(vsub(rp.r_knee, rp.r_hip));

        const fbxCurThighR = vnorm(vsub(jp.r_knee, jp.r_hip));
        const fbxCurShinR  = vnorm(vsub(jp.r_ankle, jp.r_knee));

        // Thigh twist computed later (position-based, in analytical section)

        const fbxTPoseThighR_rot = bodyRotDir(fbxTPoseThighR, true);
        const targetThighDirR = applyDeltaRotation(fbxTPoseThighR_rot, fbxCurThighR, restLegDirR);

        const fbxKneeBendR = Math.acos(Math.max(-1, Math.min(1, vdot(fbxCurThighR, fbxCurShinR))));
        const fbxKneeBendAxisR = stableBendAxis(fbxCurThighR, fbxCurShinR, 'rKneeV', [1, 0, 0]);

        const cageKneeBendAxisR = applyDeltaRotation(fbxCurThighR, targetThighDirR, fbxKneeBendAxisR);

        let targetShinDirR = targetThighDirR;
        if (Math.abs(fbxKneeBendR) > 1e-4) {
          const rf = rotateAroundAxis(targetThighDirR[0], targetThighDirR[1], targetThighDirR[2],
            0, 0, 0, cageKneeBendAxisR[0], cageKneeBendAxisR[1], cageKneeBendAxisR[2], fbxKneeBendR);
          targetShinDirR = vnorm(rf);
        }

        const rKneeW  = vadd(movedHipR, vscale(targetThighDirR, rThighLen));
        const rAnkleW = vadd(rKneeW, vscale(targetShinDirR, rShinLen));

        const rLegResult = driveLegFromPositions_world(
          movedHipR, rKneeW, rAnkleW,
          movedHipR, restLegDirR,
          rThighLen, rShinLen,
          R.r_thigh, R.r_shin, R.r_foot
        );

        // Ankle/foot rotation DELTA from T-pose
        if (rLegResult && jp.r_toe && rp.r_ankle && rp.r_knee && rp.r_toe) {
          const fbxRestShinR = vnorm(vsub(rp.r_ankle, rp.r_knee));
          const fbxRestFootR = vnorm(vsub(rp.r_toe, rp.r_ankle));
          const fbxCurFootR = vnorm(vsub(jp.r_toe, jp.r_ankle));
          const expectedFootR = applyDeltaRotation(fbxRestShinR, fbxCurShinR, fbxRestFootR);
          const ankleDeltaR = Math.acos(Math.max(-1, Math.min(1, vdot(expectedFootR, fbxCurFootR))));
          if (Math.abs(ankleDeltaR) > 1e-4) {
            let ankleAxisR = vcross(expectedFootR, fbxCurFootR);
            if (vlen(ankleAxisR) > 1e-6) {
              ankleAxisR = vnorm(ankleAxisR);
              const cageAnkleAxisR = applyDeltaRotation(fbxCurShinR, targetShinDirR, ankleAxisR);
              applyRotAxis(R.r_foot, rLegResult.anklePivot,
                cageAnkleAxisR[0], cageAnkleAxisR[1], cageAnkleAxisR[2], ankleDeltaR);
            }
          }
        }

        // Thigh twist applied via analytical rThighR in legState (not cage verts)
      }

      // (hip boundary smoothing moved to after legState — needs lSwingOnlyR matrices)

      // Store leg IK state for analytical transforms in computeAllRegionTransforms.
      // bodyRotDir provides world-space body rotation (lean + hip twist).
      // FBX positions are hips-local, so FBX delta only has limb motion relative to pelvis.
      // bodyRotDir + FBX delta = full world-space leg direction (same as vert-level code).
      // R_thigh is NOT composed with R_t — R_t only places the hip pivot point.

      // Body rotation matrices for analytical transforms.
      // Composed into R_thigh/R_upper/R_head so REST-SPACE vert offsets perpendicular
      // to the bone axis are correctly body-rotated. Without this, a 30° lean only
      // changes the bone direction but verts in front/back/sides don't follow.
      let R_bodyLeg = [1,0,0, 0,1,0, 0,0,1];   // lean + hip twist (lower body)
      let R_bodyUpper = [1,0,0, 0,1,0, 0,0,1]; // lean + spine twist (upper body)
      {
        const R_lean = (Math.abs(bodyAngle) > 1e-4)
          ? mat3_rotAxis(bodyAxis[0], bodyAxis[1], bodyAxis[2], bodyAngle)
          : [1,0,0, 0,1,0, 0,0,1];
        if (Math.abs(bodyAngle) > 1e-4 || Math.abs(hipTwistAngle) > 1e-4) {
          const R_hipTw = (Math.abs(hipTwistAngle) > 1e-4)
            ? mat3_rotAxis(curSpineDir[0], curSpineDir[1], curSpineDir[2], hipTwistAngle)
            : [1,0,0, 0,1,0, 0,0,1];
          R_bodyLeg = mat3_mul(R_hipTw, R_lean);
        }
        if (Math.abs(bodyAngle) > 1e-4 || Math.abs(twistAngle) > 1e-4) {
          const R_spineTw = (Math.abs(twistAngle) > 1e-4)
            ? mat3_rotAxis(curSpineDir[0], curSpineDir[1], curSpineDir[2], twistAngle)
            : [1,0,0, 0,1,0, 0,0,1];
          R_bodyUpper = mat3_mul(R_spineTw, R_lean);
        }
      }

      {
        const rpL = jointPosData.rest_pose;

        // Left leg
        let lThighR = [1,0,0, 0,1,0, 0,0,1]; // identity
        let lSwingOnlyR = [1,0,0, 0,1,0, 0,0,1]; // swing without twist
        let lKneeBendAxisC = [1, 0, 0], lKneeBendAngleC = 0;
        let lAnkleBendAxisC = [1, 0, 0], lAnkleBendAngleC = 0;
        if (jp.l_hip && jp.l_knee && jp.l_ankle) {
          const restLegDirL = bodyRotDir(vnorm(vsub(P.l_knee, P.l_hip)), true);
          const fbxTPoseThighL = vnorm(vsub(rpL.l_knee, rpL.l_hip));
          const fbxCurThighL = vnorm(vsub(jp.l_knee, jp.l_hip));
          const fbxCurShinL  = vnorm(vsub(jp.l_ankle, jp.l_knee));
          const fbxTPoseThighL_rot = bodyRotDir(fbxTPoseThighL, true);
          const targetThighDirL = applyDeltaRotation(fbxTPoseThighL_rot, fbxCurThighL, restLegDirL);
          // Compose R_swing with R_bodyLeg: first body-rotate vert offsets, then swing.
          // This ensures perpendicular vert offsets follow body lean correctly.
          lSwingOnlyR = mat3_mul(computeIKRotMatrix(restLegDirL, targetThighDirL), R_bodyLeg);
          // Thigh twist: LOCAL quaternion twist relative to pelvis.
          // Must be LOCAL (not world) because bodyRotDir already applies body rotation
          // to the rest direction. World twist double-counts body rotation (~15-20° extra
          // for Capoeira's 31° lean), causing over-rotation and mesh tearing at crotch.
          lThighTwist = getLocalJointTwistAtTime('l_hip', 'pelvis', 'l_knee', t);
          lThighR = (Math.abs(lThighTwist) > 1e-4)
            ? mat3_mul(mat3_rotAxis(targetThighDirL[0], targetThighDirL[1], targetThighDirL[2], lThighTwist), lSwingOnlyR)
            : lSwingOnlyR;

          // Knee R_bend: shin-to-shin delta retargeting (bodyRotDir for consistency with thigh)
          const cageRestShinDirL = bodyRotDir(vnorm(vsub(P.l_ankle, P.l_knee)), true);
          const fbxTPoseShinL = vnorm(vsub(rpL.l_ankle, rpL.l_knee));
          const fbxTPoseShinL_rot = bodyRotDir(fbxTPoseShinL, true);
          const targetShinDirL = applyDeltaRotation(fbxTPoseShinL_rot, fbxCurShinL, cageRestShinDirL);
          // Use raw rest shin dir (not bodyRotDir'd) since R_thigh now includes R_bodyLeg
          const rawRestShinDirL = vnorm(vsub(P.l_ankle, P.l_knee));
          const shinAfterThighL = vnorm(mat3_mulVec3(lThighR, rawRestShinDirL[0], rawRestShinDirL[1], rawRestShinDirL[2]));
          lKneeBendAxisC = stableBendAxis(shinAfterThighL, targetShinDirL, 'lKnee', [1, 0, 0]);
          lKneeBendAngleC = Math.acos(Math.max(-1, Math.min(1, vdot(shinAfterThighL, targetShinDirL))));

          // Ankle DELTA from T-pose
          if (jp.l_toe && rpL.l_ankle && rpL.l_knee && rpL.l_toe) {
            const fbxTPoseShinL = vnorm(vsub(rpL.l_ankle, rpL.l_knee));
            const fbxRestFootL = vnorm(vsub(rpL.l_toe, rpL.l_ankle));
            const fbxCurFootL = vnorm(vsub(jp.l_toe, jp.l_ankle));
            const expectedFootL = applyDeltaRotation(fbxTPoseShinL, fbxCurShinL, fbxRestFootL);
            lAnkleBendAngleC = Math.acos(Math.max(-1, Math.min(1, vdot(expectedFootL, fbxCurFootL))));
            if (lAnkleBendAngleC > 1e-4) {
              let fbxABA = vcross(expectedFootL, fbxCurFootL);
              if (vlen(fbxABA) > 1e-6) {
                fbxABA = vnorm(fbxABA);
                lAnkleBendAxisC = applyDeltaRotation(fbxCurShinL, targetShinDirL, fbxABA);
              }
            }
          }
        }
        // Right leg
        let rThighR = [1,0,0, 0,1,0, 0,0,1];
        let rSwingOnlyR = [1,0,0, 0,1,0, 0,0,1]; // swing without twist
        let rKneeBendAxisC = [1, 0, 0], rKneeBendAngleC = 0;
        let rAnkleBendAxisC = [1, 0, 0], rAnkleBendAngleC = 0;
        if (jp.r_hip && jp.r_knee && jp.r_ankle) {
          // bodyRotDir adds body lean + hip twist (same as vert-level cage driving)
          const restLegDirR = bodyRotDir(vnorm(vsub(P.r_knee, P.r_hip)), true);
          const fbxTPoseThighR = vnorm(vsub(rpL.r_knee, rpL.r_hip));
          const fbxCurThighR = vnorm(vsub(jp.r_knee, jp.r_hip));
          const fbxCurShinR  = vnorm(vsub(jp.r_ankle, jp.r_knee));
          const fbxTPoseThighR_rot = bodyRotDir(fbxTPoseThighR, true);
          const targetThighDirR = applyDeltaRotation(fbxTPoseThighR_rot, fbxCurThighR, restLegDirR);
          // Compose R_swing with R_bodyLeg (same as left leg)
          rSwingOnlyR = mat3_mul(computeIKRotMatrix(restLegDirR, targetThighDirR), R_bodyLeg);
          // Right thigh twist: LOCAL quaternion twist relative to pelvis (same as left)
          rThighTwist = getLocalJointTwistAtTime('r_hip', 'pelvis', 'r_knee', t);
          rThighR = (Math.abs(rThighTwist) > 1e-4)
            ? mat3_mul(mat3_rotAxis(targetThighDirR[0], targetThighDirR[1], targetThighDirR[2], rThighTwist), rSwingOnlyR)
            : rSwingOnlyR;

          // Knee R_bend: shin-to-shin delta retargeting (bodyRotDir for consistency with thigh)
          const cageRestShinDirR = bodyRotDir(vnorm(vsub(P.r_ankle, P.r_knee)), true);
          const fbxTPoseShinR = vnorm(vsub(rpL.r_ankle, rpL.r_knee));
          const fbxTPoseShinR_rot = bodyRotDir(fbxTPoseShinR, true);
          const targetShinDirR = applyDeltaRotation(fbxTPoseShinR_rot, fbxCurShinR, cageRestShinDirR);
          // Use raw rest shin dir (not bodyRotDir'd) since R_thigh now includes R_bodyLeg
          const rawRestShinDirR = vnorm(vsub(P.r_ankle, P.r_knee));
          const shinAfterThighR = vnorm(mat3_mulVec3(rThighR, rawRestShinDirR[0], rawRestShinDirR[1], rawRestShinDirR[2]));
          rKneeBendAxisC = stableBendAxis(shinAfterThighR, targetShinDirR, 'rKnee', [1, 0, 0]);
          rKneeBendAngleC = Math.acos(Math.max(-1, Math.min(1, vdot(shinAfterThighR, targetShinDirR))));

          // Ankle DELTA from T-pose
          if (jp.r_toe && rpL.r_ankle && rpL.r_knee && rpL.r_toe) {
            const fbxTPoseShinR = vnorm(vsub(rpL.r_ankle, rpL.r_knee));
            const fbxRestFootR = vnorm(vsub(rpL.r_toe, rpL.r_ankle));
            const fbxCurFootR = vnorm(vsub(jp.r_toe, jp.r_ankle));
            const expectedFootR = applyDeltaRotation(fbxTPoseShinR, fbxCurShinR, fbxRestFootR);
            rAnkleBendAngleC = Math.acos(Math.max(-1, Math.min(1, vdot(expectedFootR, fbxCurFootR))));
            if (rAnkleBendAngleC > 1e-4) {
              let fbxABA = vcross(expectedFootR, fbxCurFootR);
              if (vlen(fbxABA) > 1e-6) {
                fbxABA = vnorm(fbxABA);
                rAnkleBendAxisC = applyDeltaRotation(fbxCurShinR, targetShinDirR, fbxABA);
              }
            }
          }
        }

        // One-shot knee diagnostic (shin-to-shin approach)
        if (!window._ankleDiagDone) {
          window._ankleDiagDone = true;
          console.log('=== KNEE BEND (shin-to-shin delta retarget) ===');
          console.log('  L: angle=' + (lKneeBendAngleC*180/Math.PI).toFixed(1) +
            '° axis=[' + lKneeBendAxisC.map(v=>v.toFixed(3)).join(',') + ']');
          console.log('  R: angle=' + (rKneeBendAngleC*180/Math.PI).toFixed(1) +
            '° axis=[' + rKneeBendAxisC.map(v=>v.toFixed(3)).join(',') + ']');
          console.log('  L ankle=' + (lAnkleBendAngleC*180/Math.PI).toFixed(1) +
            '° R ankle=' + (rAnkleBendAngleC*180/Math.PI).toFixed(1) + '°');
        }

        legState = {
          positionBased: true,
          lThighR, rThighR,
          lSwingOnlyR, rSwingOnlyR,
          lKneeBendAxis: lKneeBendAxisC, rKneeBendAxis: rKneeBendAxisC,
          lKneeBendAngle: lKneeBendAngleC, rKneeBendAngle: rKneeBendAngleC,
          lAnkleBendAxis: lAnkleBendAxisC, rAnkleBendAxis: rAnkleBendAxisC,
          lAnkleBendAngle: lAnkleBendAngleC, rAnkleBendAngle: rAnkleBendAngleC,
          movedHipL, movedHipR
        };
      }

      // ── HIP BOUNDARY SMOOTHING (distance-based) ──
      // Blend lower torso cage verts toward the thigh swing transform.
      // Uses distance from hip (not Y-range) because pelvis and hips can be at the same Y.
      if (R.torso && legState.positionBased) {
        const lSwR = legState.lSwingOnlyR;
        const rSwR = legState.rSwingOnlyR;
        const mhL = legState.movedHipL;
        const mhR = legState.movedHipR;
        const hipL = P.l_hip, hipR = P.r_hip;
        const lThighLen = vlen(vsub(P.l_knee, P.l_hip));
        const rThighLen = vlen(vsub(P.r_knee, P.r_hip));

        if (lSwR && rSwR && mhL && mhR) {
          let smoothCount = 0;
          for (let ti = 0; ti < R.torso.length; ti++) {
            const ci = R.torso[ti];
            const vr = cageRest[ci];

            // Choose left or right thigh based on rest X position
            const isLeft = vr[0] < P.pelvis[0];
            const hip = isLeft ? hipL : hipR;
            const thighLen = isLeft ? lThighLen : rThighLen;
            const blendRadius = thighLen * 0.5; // cage verts within 50% of thigh length

            // Distance from hip in rest pose
            const hdx = vr[0] - hip[0], hdy = vr[1] - hip[1], hdz = vr[2] - hip[2];
            const hipDist = Math.sqrt(hdx*hdx + hdy*hdy + hdz*hdz);
            if (hipDist > blendRadius) continue;

            // Blend factor: 1 at hip, 0 at edge of radius
            const rawF = 1 - hipDist / blendRadius;
            const sf = rawF * rawF * (3 - 2 * rawF) * 0.8; // smoothstep, max 80%
            if (sf < 0.005) continue;

            const R_sw = isLeft ? lSwR : rSwR;
            const mh = isLeft ? mhL : mhR;

            // Where the thigh swing transform would place this vert:
            const dx = vr[0] - hip[0], dy = vr[1] - hip[1], dz = vr[2] - hip[2];
            const tx = R_sw[0]*dx + R_sw[1]*dy + R_sw[2]*dz + mh[0];
            const ty = R_sw[3]*dx + R_sw[4]*dy + R_sw[5]*dz + mh[1];
            const tz = R_sw[6]*dx + R_sw[7]*dy + R_sw[8]*dz + mh[2];

            const cx = cagePos[ci*3], cy = cagePos[ci*3+1], cz = cagePos[ci*3+2];
            cagePos[ci*3]   = cx + sf * (tx - cx);
            cagePos[ci*3+1] = cy + sf * (ty - cy);
            cagePos[ci*3+2] = cz + sf * (tz - cz);
            smoothCount++;
          }

          // One-shot diagnostic
          if (!window._hipSmoothDiag) {
            window._hipSmoothDiag = true;
            console.log('=== HIP BOUNDARY SMOOTHING (distance-based) ===');
            console.log('  lThighLen=' + lThighLen.toFixed(4) + ' blendRadius=' + (lThighLen*0.5).toFixed(4));
            console.log('  torso verts: ' + R.torso.length + ', smoothed: ' + smoothCount);
            console.log('  HBAND=0.60*thighLen (weight gradient covers upper thigh)');
          }
        }
      }

      // ----- ARMS from positions (RETARGETED) -----
      if ($('enableArms').checked && P.l_shoulder && P.r_shoulder) {
        if (cageShoulderL) P.l_shoulder = cageShoulderL.slice();
        if (cageShoulderR) P.r_shoulder = cageShoulderR.slice();

        // Body-rotated shoulder positions (lean + spine twist, matching cage vert transforms)
        // Arm cage verts are body-rotated at lines 2460/2500, so the IK pivot must match.
        var movedShoulderL = vadd(P.l_shoulder, pelvisDelta);
        var movedShoulderR = vadd(P.r_shoulder, pelvisDelta);
        if (Math.abs(bodyAngle) > 1e-4) {
          movedShoulderL = rotateAroundAxis(movedShoulderL[0], movedShoulderL[1], movedShoulderL[2],
            P.pelvis[0], P.pelvis[1], P.pelvis[2], bodyAxis[0], bodyAxis[1], bodyAxis[2], bodyAngle);
          movedShoulderR = rotateAroundAxis(movedShoulderR[0], movedShoulderR[1], movedShoulderR[2],
            P.pelvis[0], P.pelvis[1], P.pelvis[2], bodyAxis[0], bodyAxis[1], bodyAxis[2], bodyAngle);
        }
        if (Math.abs(twistAngle) > 1e-4) {
          movedShoulderL = rotateAroundAxis(movedShoulderL[0], movedShoulderL[1], movedShoulderL[2],
            P.pelvis[0], P.pelvis[1], P.pelvis[2], curSpineDir[0], curSpineDir[1], curSpineDir[2], twistAngle);
          movedShoulderR = rotateAroundAxis(movedShoulderR[0], movedShoulderR[1], movedShoulderR[2],
            P.pelvis[0], P.pelvis[1], P.pelvis[2], curSpineDir[0], curSpineDir[1], curSpineDir[2], twistAngle);
        }

        // Arm rest directions: body-rotated cage arm directions
        const lDir = bodyRotDir(cageArmDirL || [-1, 0, 0]);
        const rDir = bodyRotDir(cageArmDirR || [ 1, 0, 0]);
        const upperArmLenL = (P.l_shoulder && P.l_elbow) ? vlen(vsub(P.l_elbow, P.l_shoulder)) : +$('upperArmLen').value;
        const forearmLenL  = (P.l_elbow && P.l_wrist) ? vlen(vsub(P.l_wrist, P.l_elbow)) : +$('forearmLen').value;
        const upperArmLenR = (P.r_shoulder && P.r_elbow) ? vlen(vsub(P.r_elbow, P.r_shoulder)) : +$('upperArmLen').value;
        const forearmLenR  = (P.r_elbow && P.r_wrist) ? vlen(vsub(P.r_wrist, P.r_elbow)) : +$('forearmLen').value;

        // Left arm — retargeted with animation deltas
        let lResult = null, lElbowPivot = null, lWristPivot = null, lBendAxis = null;
        let lUpperTwist = 0, lTargetArmDir = null;
        let lFbxArmBendL = 0, lFbxArmBendAxisL = null;
        let rFbxArmBendR = 0, rFbxArmBendAxisR = null;
        if (jp.l_shoulder && jp.l_elbow && jp.l_wrist && upperArmLenL > 1e-6 && forearmLenL > 1e-6) {
          // Use CAGE rest arm as delta reference (not FBX bind pose).
          // This makes retargeting work for ANY bind pose (T-pose, A-pose, etc.)
          // because the "from" direction always matches the cage's own rest.
          const fbxCurArmL = vnorm(vsub(jp.l_elbow, jp.l_shoulder));
          const targetArmDirL = applyDeltaRotation(cageArmDirL || [-1,0,0], fbxCurArmL, lDir);
          lTargetArmDir = targetArmDirL;

          // FBX current forearm and bend
          const fbxCurForeL = vnorm(vsub(jp.l_wrist, jp.l_elbow));
          const fbxArmBendL = Math.acos(Math.max(-1, Math.min(1, vdot(fbxCurArmL, fbxCurForeL))));
          const fbxArmBendAxisL = stableBendAxis(fbxCurArmL, fbxCurForeL, 'lArm', [0, 1, 0]);
          lFbxArmBendL = fbxArmBendL;
          lFbxArmBendAxisL = fbxArmBendAxisL;

          // Upper arm twist computed later (position-based, after swing matrix)

          // Map bend axis to cage target arm frame
          const cageArmBendAxisL = applyDeltaRotation(fbxCurArmL, targetArmDirL, fbxArmBendAxisL);

          // Target forearm direction
          let targetForeDirL = targetArmDirL;
          if (Math.abs(fbxArmBendL) > 1e-4) {
            const rf = rotateAroundAxis(targetArmDirL[0], targetArmDirL[1], targetArmDirL[2],
              0, 0, 0, cageArmBendAxisL[0], cageArmBendAxisL[1], cageArmBendAxisL[2], fbxArmBendL);
            targetForeDirL = vnorm(rf);
          }

          // Construct target positions with cage segment lengths
          // Use body-rotated shoulder as anchor (matches body-rotated cage verts)
          const lElbowW = vadd(movedShoulderL, vscale(targetArmDirL, upperArmLenL));
          const lWristW = vadd(lElbowW, vscale(targetForeDirL, forearmLenL));

          lResult = driveArmFromPositions_world(
            movedShoulderL, lElbowW, lWristW,
            movedShoulderL, lDir, upperArmLenL, forearmLenL,
            R.l_upperarm, R.l_forearm, R.l_hand || []
          );

          if (lResult) {
            lElbowPivot = lResult.elbowPivot;
            lWristPivot = lResult.wristPivot;
            lBendAxis = lResult.bendAxis;
          }

          // Upper arm twist applied via analytical lUpperR in armState (not cage verts)
        }

        // Right arm — retargeted with animation deltas
        let rResult = null, rElbowPivot = null, rWristPivot = null, rBendAxis = null;
        let rUpperTwist = 0, rTargetArmDir = null;
        if (jp.r_shoulder && jp.r_elbow && jp.r_wrist && upperArmLenR > 1e-6 && forearmLenR > 1e-6) {
          const fbxCurArmR = vnorm(vsub(jp.r_elbow, jp.r_shoulder));
          const targetArmDirR = applyDeltaRotation(cageArmDirR || [1,0,0], fbxCurArmR, rDir);
          rTargetArmDir = targetArmDirR;

          const fbxCurForeR = vnorm(vsub(jp.r_wrist, jp.r_elbow));
          const fbxArmBendR = Math.acos(Math.max(-1, Math.min(1, vdot(fbxCurArmR, fbxCurForeR))));
          const fbxArmBendAxisR = stableBendAxis(fbxCurArmR, fbxCurForeR, 'rArm', [0, -1, 0]);
          rFbxArmBendR = fbxArmBendR;
          rFbxArmBendAxisR = fbxArmBendAxisR;

          // Upper arm twist computed later (position-based, after swing matrix)

          const cageArmBendAxisR = applyDeltaRotation(fbxCurArmR, targetArmDirR, fbxArmBendAxisR);

          let targetForeDirR = targetArmDirR;
          if (Math.abs(fbxArmBendR) > 1e-4) {
            const rf = rotateAroundAxis(targetArmDirR[0], targetArmDirR[1], targetArmDirR[2],
              0, 0, 0, cageArmBendAxisR[0], cageArmBendAxisR[1], cageArmBendAxisR[2], fbxArmBendR);
            targetForeDirR = vnorm(rf);
          }

          // Use body-rotated shoulder as anchor (matches body-rotated cage verts)
          const rElbowW = vadd(movedShoulderR, vscale(targetArmDirR, upperArmLenR));
          const rWristW = vadd(rElbowW, vscale(targetForeDirR, forearmLenR));

          rResult = driveArmFromPositions_world(
            movedShoulderR, rElbowW, rWristW,
            movedShoulderR, rDir, upperArmLenR, forearmLenR,
            R.r_upperarm, R.r_forearm, R.r_hand || []
          );

          if (rResult) {
            rElbowPivot = rResult.elbowPivot;
            rWristPivot = rResult.wristPivot;
            rBendAxis = rResult.bendAxis;
          }

          // Upper arm twist applied via analytical rUpperR in armState (not cage verts)
        }

        // Left upper arm: delta-retargeted direction
        // Body-rotate "from" to remove body lean from delta (same fix as legs/head).
        // Compose swing with R_bodyUpper for vert offset body rotation.
        const lArmRest = bodyRotDir(cageArmDirL || [-1, 0, 0]);
        let lTargetUpper = lArmRest;
        if (jp.l_shoulder && jp.l_elbow) {
          const fbxCurL = vnorm(vsub(jp.l_elbow, jp.l_shoulder));
          lTargetUpper = applyDeltaRotation(lArmRest, fbxCurL, lArmRest);
        }
        const lSwingR = mat3_mul(computeIKRotMatrix(lArmRest, lTargetUpper), R_bodyUpper);
        // Upper arm twist: LOCAL quaternion twist relative to chest.
        // Must be LOCAL (not world) because bodyRotDir already applies body rotation
        // to the rest direction — world twist would double-count it.
        lUpperTwist = getLocalJointTwistAtTime('l_shoulder', 'chest', 'l_elbow', t);
        const lUpperR = (Math.abs(lUpperTwist) > 1e-4)
          ? mat3_mul(mat3_rotAxis(lTargetUpper[0], lTargetUpper[1], lTargetUpper[2], lUpperTwist), lSwingR)
          : lSwingR;

        // Right upper arm: delta-retargeted direction (same fix as left)
        const rArmRest = bodyRotDir(cageArmDirR || [1, 0, 0]);
        let rTargetUpper = rArmRest;
        if (jp.r_shoulder && jp.r_elbow) {
          const fbxCurR = vnorm(vsub(jp.r_elbow, jp.r_shoulder));
          rTargetUpper = applyDeltaRotation(rArmRest, fbxCurR, rArmRest);
        }
        const rSwingR = mat3_mul(computeIKRotMatrix(rArmRest, rTargetUpper), R_bodyUpper);
        // Right upper arm twist: LOCAL quaternion twist relative to chest (same as left)
        rUpperTwist = getLocalJointTwistAtTime('r_shoulder', 'chest', 'r_elbow', t);
        const rUpperR = (Math.abs(rUpperTwist) > 1e-4)
          ? mat3_mul(mat3_rotAxis(rTargetUpper[0], rTargetUpper[1], rTargetUpper[2], rUpperTwist), rSwingR)
          : rSwingR;

        // Elbow bend angles
        let lBendAngleVal = 0, rBendAngleVal = 0;
        if (lResult) lBendAngleVal = lResult.bendAngle || 0;
        if (rResult) rBendAngleVal = rResult.bendAngle || 0;

        let lWristFlexAngle = 0, lWristFlexAxis = null;
        let rWristFlexAngle = 0, rWristFlexAxis = null;
        let lForearmTwist = 0, rForearmTwist = 0;

        // Forearm twist: quaternion LOCAL twist (relative to upper arm parent).
        // LOCAL is critical — our analytical composition already propagates upper arm
        // twist to forearm, so we must not double-count it.
        // The position-based method with canonical ref had the same amplification issue
        // as upper arm twist, plus depended on mid_knuckle being available.
        lForearmTwist = getLocalJointTwistAtTime('l_elbow', 'l_shoulder', 'l_wrist', t);
        rForearmTwist = getLocalJointTwistAtTime('r_elbow', 'r_shoulder', 'r_wrist', t);

        const rpLMK = jointPosData.rest_pose.l_mid_knuckle;
        const rpRMK = jointPosData.rest_pose.r_mid_knuckle;

        // Left wrist flex (uses forearm twist computed above)
        if (rpLMK && jp.l_mid_knuckle && jp.l_elbow && jp.l_wrist) {
          const rpLE = jointPosData.rest_pose.l_elbow, rpLW = jointPosData.rest_pose.l_wrist;
          if (rpLE && rpLW) {
            const fbxRestFore = vnorm(vsub(rpLW, rpLE));
            const fbxRestHand = vnorm(vsub(rpLMK, rpLW));
            const fbxCurFore = vnorm(vsub(jp.l_wrist, jp.l_elbow));
            const fbxCurHand = vnorm(vsub(jp.l_mid_knuckle, jp.l_wrist));
            const expectedHand = applyDeltaRotation(fbxRestFore, fbxCurFore, fbxRestHand);

            // Wrist flex: remove forearm twist, remaining is pure flex
            let untwistedHand = fbxCurHand;
            if (Math.abs(lForearmTwist) > 1e-4) {
              const r = rotateAroundAxis(fbxCurHand[0], fbxCurHand[1], fbxCurHand[2],
                0, 0, 0, fbxCurFore[0], fbxCurFore[1], fbxCurFore[2], -lForearmTwist);
              untwistedHand = vnorm(r);
            }
            const wDot = Math.max(-1, Math.min(1, vdot(expectedHand, untwistedHand)));
            lWristFlexAngle = Math.acos(wDot);
            if (lWristFlexAngle > 1e-4) {
              let wAxis = vcross(expectedHand, untwistedHand);
              if (vlen(wAxis) > 1e-6) {
                wAxis = vnorm(wAxis);
                const cageForeL = lResult ? lResult.lowerDir : (lTargetArmDir || lDir);
                lWristFlexAxis = applyDeltaRotation(fbxCurFore, cageForeL, wAxis);
              }
            }
          }
        }

        // Right wrist flex (uses forearm twist computed above)
        if (rpRMK && jp.r_mid_knuckle && jp.r_elbow && jp.r_wrist) {
          const rpRE = jointPosData.rest_pose.r_elbow, rpRW = jointPosData.rest_pose.r_wrist;
          if (rpRE && rpRW) {
            const fbxRestFore = vnorm(vsub(rpRW, rpRE));
            const fbxRestHand = vnorm(vsub(rpRMK, rpRW));
            const fbxCurFore = vnorm(vsub(jp.r_wrist, jp.r_elbow));
            const fbxCurHand = vnorm(vsub(jp.r_mid_knuckle, jp.r_wrist));
            const expectedHand = applyDeltaRotation(fbxRestFore, fbxCurFore, fbxRestHand);

            // Wrist flex: remove forearm twist, remaining is pure flex
            let untwistedHand = fbxCurHand;
            if (Math.abs(rForearmTwist) > 1e-4) {
              const r = rotateAroundAxis(fbxCurHand[0], fbxCurHand[1], fbxCurHand[2],
                0, 0, 0, fbxCurFore[0], fbxCurFore[1], fbxCurFore[2], -rForearmTwist);
              untwistedHand = vnorm(r);
            }
            const wDot = Math.max(-1, Math.min(1, vdot(expectedHand, untwistedHand)));
            rWristFlexAngle = Math.acos(wDot);
            if (rWristFlexAngle > 1e-4) {
              let wAxis = vcross(expectedHand, untwistedHand);
              if (vlen(wAxis) > 1e-6) {
                wAxis = vnorm(wAxis);
                const cageForeR = rResult ? rResult.lowerDir : (rTargetArmDir || rDir);
                rWristFlexAxis = applyDeltaRotation(fbxCurFore, cageForeR, wAxis);
              }
            }
          }
        }

        // Twist diagnostic (quaternion-based swing-twist decomposition)
        if (!window._twistDiagCount) window._twistDiagCount = 0;
        if (window._twistDiagCount < 5) {
          window._twistDiagCount++;
          console.log('[TWIST DIAG (quat) t=' + t.toFixed(3) + '] upperArm L=' +
            (lUpperTwist*180/Math.PI).toFixed(1) + '° R=' + (rUpperTwist*180/Math.PI).toFixed(1) +
            '° | forearm L=' + (lForearmTwist*180/Math.PI).toFixed(1) +
            '° R=' + (rForearmTwist*180/Math.PI).toFixed(1) +
            '° | thigh L=' + (lThighTwist*180/Math.PI).toFixed(1) + '° R=' + (rThighTwist*180/Math.PI).toFixed(1) +
            '° | hip=' + (hipTwistAngle*180/Math.PI).toFixed(1) + '°');
        }

        // One-shot diagnostic for analytical arm composition (delta from T-pose)
        if (!window._armAnalyticalDiagDone) {
          window._armAnalyticalDiagDone = true;
          console.log('=== ARM ANALYTICAL (bind-pose independent) ===');
          // Show FBX bind-pose arm direction (for reference only — NOT used in retargeting)
          const rpLS = jointPosData.rest_pose.l_shoulder, rpLE = jointPosData.rest_pose.l_elbow;
          const rpRS = jointPosData.rest_pose.r_shoulder, rpRE = jointPosData.rest_pose.r_elbow;
          if (rpLS && rpLE) console.log('  FBX bind L arm: [' + vnorm(vsub(rpLE, rpLS)).map(v=>v.toFixed(4)).join(', ') + ']');
          if (rpRS && rpRE) console.log('  FBX bind R arm: [' + vnorm(vsub(rpRE, rpRS)).map(v=>v.toFixed(4)).join(', ') + ']');
          console.log('  cage rest L arm: [' + (cageArmDirL||[-1,0,0]).map(v=>v.toFixed(4)).join(', ') + ']');
          console.log('  cage rest R arm: [' + (cageArmDirR||[1,0,0]).map(v=>v.toFixed(4)).join(', ') + ']');
          console.log('  lArmRest (unrotated): [' + lArmRest.map(v=>v.toFixed(4)).join(', ') + ']');
          console.log('  lTargetUpper (retargeted): [' + lTargetUpper.map(v=>v.toFixed(4)).join(', ') + ']');
          const lArmAng = Math.acos(Math.max(-1, Math.min(1, vdot(lArmRest, lTargetUpper)))) * 180/Math.PI;
          console.log('  lArm swing angle: ' + lArmAng.toFixed(1) + '°');
          console.log('  rArmRest (unrotated): [' + rArmRest.map(v=>v.toFixed(4)).join(', ') + ']');
          console.log('  rTargetUpper (retargeted): [' + rTargetUpper.map(v=>v.toFixed(4)).join(', ') + ']');
          const rArmAng = Math.acos(Math.max(-1, Math.min(1, vdot(rArmRest, rTargetUpper)))) * 180/Math.PI;
          console.log('  rArm swing angle: ' + rArmAng.toFixed(1) + '°');
          console.log('  lBendAngle=' + (lBendAngleVal*180/Math.PI).toFixed(1) + '° rBendAngle=' + (rBendAngleVal*180/Math.PI).toFixed(1) + '°');
          if (lBendAxis) console.log('  lBendAxis: [' + lBendAxis.map(v=>v.toFixed(4)).join(', ') + ']');
          if (rBendAxis) console.log('  rBendAxis: [' + rBendAxis.map(v=>v.toFixed(4)).join(', ') + ']');
        }

        armState = {
          positionBased: true,
          lShoulder: P.l_shoulder.slice(), rShoulder: P.r_shoulder.slice(),
          lElbowPivot: lElbowPivot || P.l_shoulder,
          rElbowPivot: rElbowPivot || P.r_shoulder,
          lWristPivot: lWristPivot || P.l_shoulder,
          rWristPivot: rWristPivot || P.r_shoulder,
          lBendAxis: lBendAxis || [0,1,0],
          rBendAxis: rBendAxis || [0,-1,0],
          // IK rotation matrices for analytical composition
          lUpperR: lUpperR,
          rUpperR: rUpperR,
          lBendAngle: lBendAngleVal,
          rBendAngle: rBendAngleVal,
          // Forearm twist (pronation/supination)
          lForearmTwist: lForearmTwist,
          rForearmTwist: rForearmTwist,
          // Real wrist flex from FBX finger positions (or fallback to 0)
          lWristFlexAngle: lWristFlexAngle,
          rWristFlexAngle: rWristFlexAngle,
          lWristFlexAxis: lWristFlexAxis,
          rWristFlexAxis: rWristFlexAxis,
          // Dummy Euler values (not used in position mode)
          lTposeZ: 0, rTposeZ: 0, lArmSwing: 0, rArmSwing: 0,
          lElbowAngle: 0, rElbowAngle: 0,
          lWristAngle: 0, rWristAngle: 0,
          // Body-rotated shoulder pivots for analytical transforms
          movedShoulderL, movedShoulderR
        };

        // Update debug pivot spheres
        if ($('debugPivots').checked) {
          pivotSpheres.lShoulder.position.set(movedShoulderL[0], movedShoulderL[1], movedShoulderL[2]);
          pivotSpheres.rShoulder.position.set(movedShoulderR[0], movedShoulderR[1], movedShoulderR[2]);
          if (lElbowPivot) pivotSpheres.lElbow.position.set(lElbowPivot[0], lElbowPivot[1], lElbowPivot[2]);
          if (rElbowPivot) pivotSpheres.rElbow.position.set(rElbowPivot[0], rElbowPivot[1], rElbowPivot[2]);
          if (lWristPivot) pivotSpheres.lWrist.position.set(lWristPivot[0], lWristPivot[1], lWristPivot[2]);
          if (rWristPivot) pivotSpheres.rWrist.position.set(rWristPivot[0], rWristPivot[1], rWristPivot[2]);
        }
        for (const s of Object.values(pivotSpheres)) s.visible = $('debugPivots').checked;
      } else {
        armState = null;
        for (const s of Object.values(pivotSpheres)) s.visible = false;
      }

      // ----- HEAD IK for analytical transform (position mode) -----
      // Same approach as arms/legs: compute IK rotation matrix, compose with torso.
      // Avoids the pivot-mismatch problem from vert-level rotation around rest P.neck.
      headState = null;
      if (P.neck && P.head) {
        const cageRestHead = vnorm(vsub(P.head, P.neck));
        // bodyRotDir adds body lean + spine twist (FBX positions are hips-local)
        const headRestDir = bodyRotDir(cageRestHead);
        const jpNeck = getJointPositionAtTime('neck', t);
        const jpHead = getJointPositionAtTime('head', t);
        const fpNeck = jointPosData.rest_pose.neck, fpHead = jointPosData.rest_pose.head;
        if (jpNeck && jpHead && fpNeck && fpHead) {
          const fbxRestDir = vnorm(vsub(fpHead, fpNeck));
          const fbxCurDir = vnorm(vsub(jpHead, jpNeck));
          // Body-rotate FBX rest dir to remove body lean from delta (same fix as legs).
          // Then compose IK with R_bodyUpper for vert offset body rotation.
          const fbxRestDir_rot = bodyRotDir(fbxRestDir);
          const headTarget = applyDeltaRotation(fbxRestDir_rot, fbxCurDir, headRestDir);
          headState = mat3_mul(computeIKRotMatrix(headRestDir, headTarget), R_bodyUpper);
        }
      }

    } else {
    // ═══════════════════════════════════════════════════════════
    // EULER-BASED DRIVING (fallback when no joint_positions.json)
    legState = null;  // no analytical legs in Euler mode
    headState = null;
    // ═══════════════════════════════════════════════════════════

    // ----- Leg angles from Mixamo -----
    const flipHip   = $('flipHip').checked ? -1 : 1;
    const flipKnee  = $('flipKnee').checked ? -1 : 1;
    const flipAnkle = $('flipAnkleX').checked ? -1 : 1;

    const lHipX   = getRotationAtTime('l_hip',   t, 'x_rot') * flipHip;
    const lHipY   = getRotationAtTime('l_hip',   t, 'y_rot');
    const lKneeX  = getRotationAtTime('l_knee',  t, 'x_rot') * flipKnee;
    const lKneeY  = getRotationAtTime('l_knee',  t, 'y_rot');
    const lAnkleX = getRotationAtTime('l_ankle', t, 'x_rot') * flipAnkle;
    const lAnkleY = getRotationAtTime('l_ankle', t, 'y_rot');
    const lAnkleZ = getRotationAtTime('l_ankle', t, 'z_rot');

    const rHipX   = getRotationAtTime('r_hip',   t, 'x_rot') * flipHip;
    const rHipY   = getRotationAtTime('r_hip',   t, 'y_rot');
    const rKneeX  = getRotationAtTime('r_knee',  t, 'x_rot') * flipKnee;
    const rKneeY  = getRotationAtTime('r_knee',  t, 'y_rot');
    const rAnkleX = getRotationAtTime('r_ankle', t, 'x_rot') * flipAnkle;
    const rAnkleY = getRotationAtTime('r_ankle', t, 'y_rot');
    const rAnkleZ = getRotationAtTime('r_ankle', t, 'z_rot');

    // ----- LEFT LEG with propagated pivots -----
    const lLeg = [].concat(R.l_thigh, R.l_shin, R.l_foot);
    applyRotX(lLeg, P.l_hip, lHipX);
    applyRotY(lLeg, P.l_hip, lHipY);

    let lKneePivot = rotateAroundAxis(
      P.l_knee[0], P.l_knee[1], P.l_knee[2],
      P.l_hip[0], P.l_hip[1], P.l_hip[2], 1,0,0, lHipX);
    lKneePivot = rotateAroundAxis(
      lKneePivot[0], lKneePivot[1], lKneePivot[2],
      P.l_hip[0], P.l_hip[1], P.l_hip[2], 0,1,0, lHipY);
    const lShinFoot = [].concat(R.l_shin, R.l_foot);
    applyRotX(lShinFoot, lKneePivot, lKneeX);
    applyRotY(lShinFoot, lKneePivot, lKneeY);

    let lAnklePivot = rotateAroundAxis(
      P.l_ankle[0], P.l_ankle[1], P.l_ankle[2],
      P.l_hip[0], P.l_hip[1], P.l_hip[2], 1,0,0, lHipX);
    lAnklePivot = rotateAroundAxis(
      lAnklePivot[0], lAnklePivot[1], lAnklePivot[2],
      P.l_hip[0], P.l_hip[1], P.l_hip[2], 0,1,0, lHipY);
    lAnklePivot = rotateAroundAxis(
      lAnklePivot[0], lAnklePivot[1], lAnklePivot[2],
      lKneePivot[0], lKneePivot[1], lKneePivot[2], 1,0,0, lKneeX);
    lAnklePivot = rotateAroundAxis(
      lAnklePivot[0], lAnklePivot[1], lAnklePivot[2],
      lKneePivot[0], lKneePivot[1], lKneePivot[2], 0,1,0, lKneeY);
    const ankleRestL = (+$('ankleOffL').value) * Math.PI/180;
    applyRotX(R.l_foot, lAnklePivot, lAnkleX - ankleRestL);
    applyRotY(R.l_foot, lAnklePivot, lAnkleY);
    applyRotZ(R.l_foot, lAnklePivot, lAnkleZ);

    // ----- RIGHT LEG with propagated pivots -----
    const rLeg = [].concat(R.r_thigh, R.r_shin, R.r_foot);
    applyRotX(rLeg, P.r_hip, rHipX);
    applyRotY(rLeg, P.r_hip, rHipY);

    let rKneePivot = rotateAroundAxis(
      P.r_knee[0], P.r_knee[1], P.r_knee[2],
      P.r_hip[0], P.r_hip[1], P.r_hip[2], 1,0,0, rHipX);
    rKneePivot = rotateAroundAxis(
      rKneePivot[0], rKneePivot[1], rKneePivot[2],
      P.r_hip[0], P.r_hip[1], P.r_hip[2], 0,1,0, rHipY);
    const rShinFoot = [].concat(R.r_shin, R.r_foot);
    applyRotX(rShinFoot, rKneePivot, rKneeX);
    applyRotY(rShinFoot, rKneePivot, rKneeY);

    let rAnklePivot = rotateAroundAxis(
      P.r_ankle[0], P.r_ankle[1], P.r_ankle[2],
      P.r_hip[0], P.r_hip[1], P.r_hip[2], 1,0,0, rHipX);
    rAnklePivot = rotateAroundAxis(
      rAnklePivot[0], rAnklePivot[1], rAnklePivot[2],
      P.r_hip[0], P.r_hip[1], P.r_hip[2], 0,1,0, rHipY);
    rAnklePivot = rotateAroundAxis(
      rAnklePivot[0], rAnklePivot[1], rAnklePivot[2],
      rKneePivot[0], rKneePivot[1], rKneePivot[2], 1,0,0, rKneeX);
    rAnklePivot = rotateAroundAxis(
      rAnklePivot[0], rAnklePivot[1], rAnklePivot[2],
      rKneePivot[0], rKneePivot[1], rKneePivot[2], 0,1,0, rKneeY);
    const ankleRestR = (+$('ankleOffR').value) * Math.PI/180;
    applyRotX(R.r_foot, rAnklePivot, rAnkleX - ankleRestR);
    applyRotY(R.r_foot, rAnklePivot, rAnkleY);
    applyRotZ(R.r_foot, rAnklePivot, rAnkleZ);

    // ----- ARMS (Euler-based, actual Mixamo data + T-pose base) -----
    if($('enableArms').checked && P.l_shoulder && P.r_shoulder){
      const shoulderX = +$('shoulderX').value;
      if (cageShoulderL) { P.l_shoulder = cageShoulderL.slice(); }
      else { P.l_shoulder = [-shoulderX, 0.20, -0.03]; }
      if (cageShoulderR) { P.r_shoulder = cageShoulderR.slice(); }
      else { P.r_shoulder = [ shoulderX, 0.20, -0.03]; }

      const tPoseDown = +$('tposeAngle').value * Math.PI/180;
      const flipShoulder = $('flipShoulder').checked ? -1 : 1;
      const flipElbow = $('flipElbow').checked ? -1 : 1;
      const armSwingScale = +$('armSwing').value * flipShoulder;
      const elbowBendScale = +$('elbowBend').value * flipElbow;
      const upperArmLen = +$('upperArmLen').value;
      const forearmLen = +$('forearmLen').value;
      const lDir = cageArmDirL || [-1, 0, 0];
      const rDir = cageArmDirR || [ 1, 0, 0];

      let lArmSwing, rArmSwing, lElbowBend, rElbowBend, lWristCounter, rWristCounter;
      let lTposeZ, rTposeZ;

      if (armMotionData && armBaseline) {
        const lShX = getArmRotationAtTime('l_shoulder', t, 'x_rot');
        const rShX = getArmRotationAtTime('r_shoulder', t, 'x_rot');
        lArmSwing = (lShX - armBaseline.l_shoulder.x) * armSwingScale;
        rArmSwing = (rShX - armBaseline.r_shoulder.x) * armSwingScale;
        lTposeZ = tPoseDown;
        rTposeZ = -tPoseDown;
        const lElZ = getArmRotationAtTime('l_elbow', t, 'z_rot');
        const rElZ = getArmRotationAtTime('r_elbow', t, 'z_rot');
        lElbowBend = (lElZ - armBaseline.l_elbow.z) * elbowBendScale;
        rElbowBend = -(rElZ - armBaseline.r_elbow.z) * elbowBendScale;
        const lWrZ = getArmRotationAtTime('l_wrist', t, 'z_rot');
        const rWrZ = getArmRotationAtTime('r_wrist', t, 'z_rot');
        lWristCounter = (lWrZ - armBaseline.l_wrist.z);
        rWristCounter = -(rWrZ - armBaseline.r_wrist.z);
      } else {
        lArmSwing = rHipX * armSwingScale;
        rArmSwing = lHipX * armSwingScale;
        lTposeZ = tPoseDown;
        rTposeZ = -tPoseDown;
        lElbowBend = Math.max(0, -lArmSwing) * elbowBendScale;
        rElbowBend = Math.max(0, -rArmSwing) * elbowBendScale;
        lWristCounter = -lElbowBend * 0.5;
        rWristCounter = -rElbowBend * 0.5;
      }

      // LEFT ARM
      const lArmAll = [].concat(R.l_upperarm, R.l_forearm, R.l_hand || []);
      applyRotZ(lArmAll, P.l_shoulder, lTposeZ);
      applyRotX(lArmAll, P.l_shoulder, lArmSwing);
      let lBendAxis = rotVec3(0, 1, 0, 0, 0, 1, lTposeZ);
      lBendAxis = rotVec3(lBendAxis[0], lBendAxis[1], lBendAxis[2], 1, 0, 0, lArmSwing);
      let lElbowPivot = [
        P.l_shoulder[0] + lDir[0] * upperArmLen,
        P.l_shoulder[1] + lDir[1] * upperArmLen,
        P.l_shoulder[2] + lDir[2] * upperArmLen
      ];
      lElbowPivot = rotateAroundAxis(lElbowPivot[0], lElbowPivot[1], lElbowPivot[2],
        P.l_shoulder[0], P.l_shoulder[1], P.l_shoulder[2], 0,0,1, lTposeZ);
      lElbowPivot = rotateAroundAxis(lElbowPivot[0], lElbowPivot[1], lElbowPivot[2],
        P.l_shoulder[0], P.l_shoulder[1], P.l_shoulder[2], 1,0,0, lArmSwing);
      const lForearmHand = [].concat(R.l_forearm, R.l_hand || []);
      if(lForearmHand.length && Math.abs(lElbowBend) > 0.001) {
        applyRotAxis(lForearmHand, lElbowPivot, lBendAxis[0], lBendAxis[1], lBendAxis[2], lElbowBend);
      }
      let lWristPivot = [
        P.l_shoulder[0] + lDir[0] * (upperArmLen + forearmLen),
        P.l_shoulder[1] + lDir[1] * (upperArmLen + forearmLen),
        P.l_shoulder[2] + lDir[2] * (upperArmLen + forearmLen)
      ];
      lWristPivot = rotateAroundAxis(lWristPivot[0], lWristPivot[1], lWristPivot[2],
        P.l_shoulder[0], P.l_shoulder[1], P.l_shoulder[2], 0,0,1, lTposeZ);
      lWristPivot = rotateAroundAxis(lWristPivot[0], lWristPivot[1], lWristPivot[2],
        P.l_shoulder[0], P.l_shoulder[1], P.l_shoulder[2], 1,0,0, lArmSwing);
      lWristPivot = rotateAroundAxis(lWristPivot[0], lWristPivot[1], lWristPivot[2],
        lElbowPivot[0], lElbowPivot[1], lElbowPivot[2], lBendAxis[0], lBendAxis[1], lBendAxis[2], lElbowBend);
      if((R.l_hand||[]).length && Math.abs(lWristCounter) > 0.001) {
        applyRotAxis(R.l_hand, lWristPivot, lBendAxis[0], lBendAxis[1], lBendAxis[2], lWristCounter);
      }

      // RIGHT ARM
      const rArmAll = [].concat(R.r_upperarm, R.r_forearm, R.r_hand || []);
      applyRotZ(rArmAll, P.r_shoulder, rTposeZ);
      applyRotX(rArmAll, P.r_shoulder, rArmSwing);
      let rBendAxis = rotVec3(0, -1, 0, 0, 0, 1, rTposeZ);
      rBendAxis = rotVec3(rBendAxis[0], rBendAxis[1], rBendAxis[2], 1, 0, 0, rArmSwing);
      let rElbowPivot = [
        P.r_shoulder[0] + rDir[0] * upperArmLen,
        P.r_shoulder[1] + rDir[1] * upperArmLen,
        P.r_shoulder[2] + rDir[2] * upperArmLen
      ];
      rElbowPivot = rotateAroundAxis(rElbowPivot[0], rElbowPivot[1], rElbowPivot[2],
        P.r_shoulder[0], P.r_shoulder[1], P.r_shoulder[2], 0,0,1, rTposeZ);
      rElbowPivot = rotateAroundAxis(rElbowPivot[0], rElbowPivot[1], rElbowPivot[2],
        P.r_shoulder[0], P.r_shoulder[1], P.r_shoulder[2], 1,0,0, rArmSwing);
      const rForearmHand = [].concat(R.r_forearm, R.r_hand || []);
      if(rForearmHand.length && Math.abs(rElbowBend) > 0.001) {
        applyRotAxis(rForearmHand, rElbowPivot, rBendAxis[0], rBendAxis[1], rBendAxis[2], rElbowBend);
      }
      let rWristPivot = [
        P.r_shoulder[0] + rDir[0] * (upperArmLen + forearmLen),
        P.r_shoulder[1] + rDir[1] * (upperArmLen + forearmLen),
        P.r_shoulder[2] + rDir[2] * (upperArmLen + forearmLen)
      ];
      rWristPivot = rotateAroundAxis(rWristPivot[0], rWristPivot[1], rWristPivot[2],
        P.r_shoulder[0], P.r_shoulder[1], P.r_shoulder[2], 0,0,1, rTposeZ);
      rWristPivot = rotateAroundAxis(rWristPivot[0], rWristPivot[1], rWristPivot[2],
        P.r_shoulder[0], P.r_shoulder[1], P.r_shoulder[2], 1,0,0, rArmSwing);
      rWristPivot = rotateAroundAxis(rWristPivot[0], rWristPivot[1], rWristPivot[2],
        rElbowPivot[0], rElbowPivot[1], rElbowPivot[2], rBendAxis[0], rBendAxis[1], rBendAxis[2], rElbowBend);
      if((R.r_hand||[]).length && Math.abs(rWristCounter) > 0.001) {
        applyRotAxis(R.r_hand, rWristPivot, rBendAxis[0], rBendAxis[1], rBendAxis[2], rWristCounter);
      }

      armState = {
        lTposeZ, rTposeZ, lArmSwing, rArmSwing,
        lShoulder: P.l_shoulder.slice(), rShoulder: P.r_shoulder.slice(),
        lElbowAngle: lElbowBend, rElbowAngle: rElbowBend,
        lElbowPivot, rElbowPivot,
        lWristAngle: lWristCounter, rWristAngle: rWristCounter,
        lWristPivot, rWristPivot,
        lBendAxis, rBendAxis
      };

      if ($('debugPivots').checked) {
        pivotSpheres.lShoulder.position.set(P.l_shoulder[0], P.l_shoulder[1], P.l_shoulder[2]);
        pivotSpheres.rShoulder.position.set(P.r_shoulder[0], P.r_shoulder[1], P.r_shoulder[2]);
        pivotSpheres.lElbow.position.set(lElbowPivot[0], lElbowPivot[1], lElbowPivot[2]);
        pivotSpheres.rElbow.position.set(rElbowPivot[0], rElbowPivot[1], rElbowPivot[2]);
        pivotSpheres.lWrist.position.set(lWristPivot[0], lWristPivot[1], lWristPivot[2]);
        pivotSpheres.rWrist.position.set(rWristPivot[0], rWristPivot[1], rWristPivot[2]);
      }
      for (const s of Object.values(pivotSpheres)) s.visible = $('debugPivots').checked;
    } else {
      armState = null;
      for (const s of Object.values(pivotSpheres)) s.visible = false;
    }

    } // end Euler fallback

    // ── ARM-TORSO COLLISION AVOIDANCE ──
    // Approximate the torso as a capsule (spine axis + radius) computed from
    // the deformed torso cage verts. Push arm cage verts outward if they penetrate.
    if ((allLeftArmCage.length || allRightArmCage.length) && R.torso && R.torso.length > 10) {
      const neckRY = P.neck ? P.neck[1] : P.pelvis[1] + 0.2;
      const midRY = (P.pelvis[1] + neckRY) / 2;
      // Compute deformed spine axis from torso cage vert centroids
      let bx = 0, by = 0, bz = 0, bn = 0; // lower torso centroid
      let ux = 0, uy = 0, uz = 0, un = 0; // upper torso centroid
      for (const ci of R.torso) {
        const ry = cageRest[ci][1];
        const px = cagePos[ci*3], py = cagePos[ci*3+1], pz = cagePos[ci*3+2];
        if (ry < midRY) { bx += px; by += py; bz += pz; bn++; }
        else { ux += px; uy += py; uz += pz; un++; }
      }
      if (bn > 0 && un > 0) {
        bx /= bn; by /= bn; bz /= bn;
        ux /= un; uy /= un; uz /= un;
        const capAx = ux - bx, capAy = uy - by, capAz = uz - bz;
        const capLen = Math.sqrt(capAx*capAx + capAy*capAy + capAz*capAz);
        if (capLen > 0.005) {
          const nax = capAx/capLen, nay = capAy/capLen, naz = capAz/capLen;
          // Capsule radius: compute from average torso vert distance to spine axis
          let radSum = 0, radN = 0;
          for (const ci of R.torso) {
            const px = cagePos[ci*3]-bx, py = cagePos[ci*3+1]-by, pz = cagePos[ci*3+2]-bz;
            const proj = px*nax + py*nay + pz*naz;
            const perpX = px - proj*nax, perpY = py - proj*nay, perpZ = pz - proj*naz;
            const perpD = Math.sqrt(perpX*perpX + perpY*perpY + perpZ*perpZ);
            radSum += perpD; radN++;
          }
          // Use 85% of average distance as collision radius + 5% margin beyond surface.
          // 60% was too conservative — arms passing through outer torso weren't detected.
          const avgR = (radN > 0) ? (radSum / radN) : 0.05;
          const capR = avgR * 0.85;
          const capRMargin = avgR * 0.90; // push verts slightly beyond surface to avoid clipping
          const capR2 = capRMargin * capRMargin;

          // Push arm verts outward if inside capsule
          const armIdx = [].concat(allLeftArmCage, allRightArmCage,
            R.l_hand || [], R.r_hand || []);
          let pushCount = 0;
          for (const ci of armIdx) {
            const px = cagePos[ci*3], py = cagePos[ci*3+1], pz = cagePos[ci*3+2];
            const dx = px - bx, dy = py - by, dz = pz - bz;
            let proj = dx*nax + dy*nay + dz*naz;
            proj = Math.max(0, Math.min(capLen, proj));
            const closestX = bx + nax*proj, closestY = by + nay*proj, closestZ = bz + naz*proj;
            const rx = px - closestX, ry2 = py - closestY, rz = pz - closestZ;
            const d2 = rx*rx + ry2*ry2 + rz*rz;
            if (d2 < capR2 && d2 > 1e-10) {
              const d = Math.sqrt(d2);
              const push = capRMargin - d;
              const invD = 1 / d;
              cagePos[ci*3]   += rx * invD * push;
              cagePos[ci*3+1] += ry2 * invD * push;
              cagePos[ci*3+2] += rz * invD * push;
              pushCount++;
            }
          }
          // One-shot diagnostic (resets when FBX loaded via window._armCollisionDiag = false)
          if (!window._armCollisionDiag) {
            window._armCollisionDiag = true;
            console.log('=== ARM-TORSO COLLISION ===');
            console.log('  capsule: bottom=['+[bx,by,bz].map(v=>v.toFixed(4))+'] top=['+[ux,uy,uz].map(v=>v.toFixed(4))+'] len='+capLen.toFixed(4));
            console.log('  avg torso radius: '+avgR.toFixed(4)+' collision radius: '+capR.toFixed(4)+' margin: '+capRMargin.toFixed(4));
            console.log('  arm verts checked: '+armIdx.length+' pushed: '+pushCount);
          }
        }
      }
    }

    // ----- HEAD/NECK motion -----
    // Position mode: handled analytically via headState (computed above).
    // No vert-level rotation needed — avoids pivot-mismatch at P.neck after body rotation.
    // Euler fallback: vert-level rotation from spine_keyframes.json.
    if (!jointPosData && spineMotionData && P.neck && P.head && R.head && R.head.length) {
      // Euler fallback: head from spine_keyframes.json
      const neckX = getSpineDataAtTime('neck', t, 'x_rot') * Math.PI/180;
      const neckY = getSpineDataAtTime('neck', t, 'y_rot') * Math.PI/180;
      const neckZ = getSpineDataAtTime('neck', t, 'z_rot') * Math.PI/180;

      applyRotX(R.head, P.neck, neckX);
      applyRotY(R.head, P.neck, neckY);
      applyRotZ(R.head, P.neck, neckZ);

      let headPivot = rotateAroundAxis(
        P.head[0], P.head[1], P.head[2],
        P.neck[0], P.neck[1], P.neck[2], 1,0,0, neckX);
      headPivot = rotateAroundAxis(
        headPivot[0], headPivot[1], headPivot[2],
        P.neck[0], P.neck[1], P.neck[2], 0,1,0, neckY);
      headPivot = rotateAroundAxis(
        headPivot[0], headPivot[1], headPivot[2],
        P.neck[0], P.neck[1], P.neck[2], 0,0,1, neckZ);

      const headX = getSpineDataAtTime('head', t, 'x_rot') * Math.PI/180;
      const headY = getSpineDataAtTime('head', t, 'y_rot') * Math.PI/180;
      const headZ = getSpineDataAtTime('head', t, 'z_rot') * Math.PI/180;

      applyRotX(R.head, headPivot, headX);
      applyRotY(R.head, headPivot, headY);
      applyRotZ(R.head, headPivot, headZ);
    }

    // ----- Foot lock correction LAST (only with Euler data) -----
    // Foot lock: Euler mode uses knee angle detection, FBX mode uses ankle velocity
    if (!jointPosData) footLockCorrection(t);
    else footLockCorrectionFBX(t);

    // ----- Ground compensation (position mode) -----
    // When FBX skeleton has different proportions than the cage (e.g. longer legs),
    // the body lean + shorter cage legs = feet float above ground.
    // Fix: find lowest foot cage vert, compare to rest-pose lowest, and push down.
    if (jointPosData && R.l_foot && R.r_foot) {
      let minFootY = Infinity;
      const footIdx = [].concat(R.l_foot, R.r_foot);
      for (const ci of footIdx) {
        const y = cagePos[ci * 3 + 1];
        if (y < minFootY) minFootY = y;
      }
      // Rest-pose lowest foot vert
      let minFootRestY = Infinity;
      for (const ci of footIdx) {
        if (cageRest[ci][1] < minFootRestY) minFootRestY = cageRest[ci][1];
      }
      // Ground plane = rest-pose lowest foot vert Y.
      // groundCorrection < 0 → feet above ground (floating) → push down
      // groundCorrection > 0 → feet below ground (floor penetration) → push up
      // Both directions are corrected to keep feet at floor level.
      const groundCorrection = minFootRestY - minFootY;
      if (Math.abs(groundCorrection) > 0.001) {
        translate(allIdx, 0, groundCorrection, 0);
        lastGroundCorrection = groundCorrection;
      } else {
        lastGroundCorrection = 0;
      }
    } else {
      lastGroundCorrection = 0;
    }

    // One-shot: log ground comp + pivot alignment diagnostic
    if (!window._groundDiagDone && jointPosData) {
      window._groundDiagDone = true;
      console.log('=== GROUND COMP + PIVOT DIAG ===');
      console.log('  lastGroundCorrection=' + lastGroundCorrection.toFixed(4));
      if (R.l_foot && R.r_foot) {
        const footIdx = [].concat(R.l_foot, R.r_foot);
        let minY = Infinity, minRestY = Infinity;
        for (const ci of footIdx) {
          if (cagePos[ci*3+1] < minY) minY = cagePos[ci*3+1];
          if (cageRest[ci][1] < minRestY) minRestY = cageRest[ci][1];
        }
        console.log('  foot cage minY=' + minY.toFixed(4) + ' rest minY=' + minRestY.toFixed(4));
      }
      const _mhL = legState && legState.movedHipL;
      const _mhR = legState && legState.movedHipR;
      if (_mhL) console.log('  movedHipL=[' + _mhL.map(v=>v.toFixed(4)).join(',') + ']');
      if (_mhR) console.log('  movedHipR=[' + _mhR.map(v=>v.toFixed(4)).join(',') + ']');
      console.log('  P.l_hip=[' + P.l_hip.map(v=>v.toFixed(4)).join(',') + ']');
      console.log('  P.r_hip=[' + P.r_hip.map(v=>v.toFixed(4)).join(',') + ']');
    }

    applyCageSmoothing();
  }


function normalizeRegionId(regionIdInput, cageLen) {
  if (!Array.isArray(regionIdInput)) {
    console.warn("regions.regionId missing or invalid. Filling with zeros.");
    return new Array(cageLen).fill(0);
  }
  if (regionIdInput.length === cageLen) return regionIdInput.slice();
  if (regionIdInput.length > cageLen) {
    console.warn(`regions.regionId longer than cageRest (${regionIdInput.length} > ${cageLen}). Truncating.`);
    return regionIdInput.slice(0, cageLen);
  }
  console.warn(`regions.regionId shorter than cageRest (${regionIdInput.length} < ${cageLen}). Padding.`);
  const padded = regionIdInput.slice();
  const fillVal = regionIdInput.length ? regionIdInput[regionIdInput.length - 1] : 0;
  while (padded.length < cageLen) padded.push(fillVal);
  return padded;
}

  4: [4, 6, 2],         // r_shin -> r_foot + r_thigh
  5: [5, 3],            // l_foot -> l_shin
  6: [6, 4],            // r_foot -> r_shin
  7: [7, 9, 0],         // l_upperarm -> l_forearm + torso
  8: [8, 10, 0],        // r_upperarm -> r_forearm + torso
  9: [9, 7, 12],        // l_forearm -> l_upperarm + l_hand
  10: [10, 8, 13],      // r_forearm -> r_upperarm + r_hand
  11: [11],             // head/neck (if present)
  12: [12, 9],          // l_hand -> l_forearm
  13: [13, 10]          // r_hand -> r_forearm
};

function classifyMeshRegion(meta, x, y, z){
  // Always return null to use nearest-cage-vert classification.
  // The cage regions are properly classified using joint positions (regenerate_cage.py).
  // Assigning each mesh vert the region of its nearest cage vert guarantees consistency.
  return null;
}

function buildBindKNNRegion(meshRestPos, cageRest, regionIdInput, regionMeta, K = 8) {
  const nMesh = meshRestPos.length / 3;
  const nCage = cageRest.length;
  const regionIdLocal = normalizeRegionId(regionIdInput, nCage);

  const regionToIndices = new Map();
  for (let ci = 0; ci < nCage; ci++) {
    const r = regionIdLocal[ci];
    if (!regionToIndices.has(r)) regionToIndices.set(r, []);
    regionToIndices.get(r).push(ci);
  }

  const idx = new Int32Array(nMesh * K);
  const w   = new Float32Array(nMesh * K);
  let reclassCount = 0, clampCount = 0;
  let thighReclassCount = 0, thighClampCount = 0;

  for (let vi = 0; vi < nMesh; vi++) {
    const x = meshRestPos[vi*3], y = meshRestPos[vi*3+1], z = meshRestPos[vi*3+2];

    let meshRegion = classifyMeshRegion(regionMeta, x, y, z);
    if (meshRegion === null) {
      let nearestRegion = 0;
      let nearestD2 = 1e30;
      for (let ci = 0; ci < nCage; ci++) {
        const cx = cageRest[ci][0], cy = cageRest[ci][1], cz = cageRest[ci][2];
        const dx = x - cx, dy = y - cy, dz = z - cz;
        const d2 = dx*dx + dy*dy + dz*dz;
        if (d2 < nearestD2) {
          nearestD2 = d2;
          nearestRegion = regionIdLocal[ci];
        }
      }
      meshRegion = nearestRegion;
    }

    let allowedRegions = REGION_ADJ[meshRegion] || [meshRegion];

    // Geometry-based arm detection + torso clamp (Fix A + Fix B).
    // Catches bicep verts misclassified as torso by nearest-cage-vert.
    // Uses point-to-arm-axis distance + projection range to detect arm verts,
    // then clamps torso influence past a shoulder band.
    if (cageShoulderL && cageShoulderR && cageArmDirL && cageArmDirR) {
      const uaLen = +$('upperArmLen').value;
      const faLen = +$('forearmLen').value;
      const totalArmLen = uaLen + faLen;
      const ARM_RADIUS = 0.06;              // 6cm from axis = "on the arm"
      const SHOULDER_BAND = 0.06 * uaLen;   // torso blending only within this narrow band
      const ELBOW_BAND = 0.12 * uaLen;      // cross-arm blending zone around elbow (smoothstep-weighted)

      // Check left arm
      const lsh = cageShoulderL, ld = cageArmDirL;
      const ltProj = (x-lsh[0])*ld[0] + (y-lsh[1])*ld[1] + (z-lsh[2])*ld[2];
      if (ltProj > 0 && ltProj < totalArmLen) {
        const lpx = lsh[0]+ld[0]*ltProj, lpy = lsh[1]+ld[1]*ltProj, lpz = lsh[2]+ld[2]*ltProj;
        const lAxDist = Math.sqrt((x-lpx)*(x-lpx) + (y-lpy)*(y-lpy) + (z-lpz)*(z-lpz));
        if (lAxDist < ARM_RADIUS) {
          if (meshRegion === 0) {
            meshRegion = (ltProj <= uaLen) ? 7 : 9;
            allowedRegions = REGION_ADJ[meshRegion] || [meshRegion];
            reclassCount++;
          }
          // Shoulder clamp: no torso past shoulder band
          if (ltProj > SHOULDER_BAND) {
            allowedRegions = allowedRegions.filter(r => r !== 0);
            clampCount++;
          }
          // Elbow clamp: upper arm verts far from elbow -> no forearm
          if (meshRegion === 7 && ltProj < uaLen - ELBOW_BAND) {
            allowedRegions = allowedRegions.filter(r => r !== 9);
            clampCount++;
          }
          // Elbow clamp: forearm verts far from elbow -> no upper arm
          if (meshRegion === 9 && ltProj > uaLen + ELBOW_BAND) {
            allowedRegions = allowedRegions.filter(r => r !== 7);
            clampCount++;
          }
        }
      }

      // Check right arm
      const rsh = cageShoulderR, rd = cageArmDirR;
      const rtProj = (x-rsh[0])*rd[0] + (y-rsh[1])*rd[1] + (z-rsh[2])*rd[2];
      if (rtProj > 0 && rtProj < totalArmLen) {
        const rpx = rsh[0]+rd[0]*rtProj, rpy = rsh[1]+rd[1]*rtProj, rpz = rsh[2]+rd[2]*rtProj;
        const rAxDist = Math.sqrt((x-rpx)*(x-rpx) + (y-rpy)*(y-rpy) + (z-rpz)*(z-rpz));
        if (rAxDist < ARM_RADIUS) {
          if (meshRegion === 0) {
            meshRegion = (rtProj <= uaLen) ? 8 : 10;
            allowedRegions = REGION_ADJ[meshRegion] || [meshRegion];
            reclassCount++;
          }
          if (rtProj > SHOULDER_BAND) {
            allowedRegions = allowedRegions.filter(r => r !== 0);
            clampCount++;
          }
          // Elbow clamp: upper arm far from elbow -> no forearm
          if (meshRegion === 8 && rtProj < uaLen - ELBOW_BAND) {
            allowedRegions = allowedRegions.filter(r => r !== 10);
            clampCount++;
          }
          // Elbow clamp: forearm far from elbow -> no upper arm
          if (meshRegion === 10 && rtProj > uaLen + ELBOW_BAND) {
            allowedRegions = allowedRegions.filter(r => r !== 8);
            clampCount++;
          }
        }
      }
    }

    // Geometry-based thigh detection + torso clamp (same pattern as arms).
    // Catches groin/upper-thigh verts misclassified as torso by nearest-cage-vert.
    // Without this, the cage's near-vertical right thigh (3° outward) means r_thigh
    // cage verts are so close to torso that KNN assigns zero thigh weight to groin mesh verts.
    if (P.l_hip && P.l_knee && P.r_hip && P.r_knee) {
      // Left thigh
      const lh = P.l_hip, lk = P.l_knee;
      const lThLen = Math.sqrt((lk[0]-lh[0])**2 + (lk[1]-lh[1])**2 + (lk[2]-lh[2])**2);
      const lThD = [(lk[0]-lh[0])/lThLen, (lk[1]-lh[1])/lThLen, (lk[2]-lh[2])/lThLen];
      const THIGH_RADIUS = 0.25 * lThLen; // proportional to thigh length (~10cm)
      const THIGH_RADIUS_HIP_L = 0.55 * lThLen; // wider near hip for crotch/groin
      const lHipBand = 0.15 * lThLen;
      const HIP_EXTEND_L = 0.12 * lThLen; // detect slightly above hip for crotch
      const lThProj = (x-lh[0])*lThD[0] + (y-lh[1])*lThD[1] + (z-lh[2])*lThD[2];
      if (lThProj > -HIP_EXTEND_L && lThProj < lThLen) {
        const projCL = Math.max(0, lThProj);
        const lpx = lh[0]+lThD[0]*projCL, lpy = lh[1]+lThD[1]*projCL, lpz = lh[2]+lThD[2]*projCL;
        const lThAxDist = Math.sqrt((x-lpx)*(x-lpx) + (y-lpy)*(y-lpy) + (z-lpz)*(z-lpz));
        // Tapered radius: wide near hip (crotch), normal further down thigh
        const hipFracL = Math.min(1, Math.max(0, 1 - lThProj / (0.3 * lThLen)));
        const effectiveRadiusL = THIGH_RADIUS + (THIGH_RADIUS_HIP_L - THIGH_RADIUS) * hipFracL;
        if (lThAxDist < effectiveRadiusL) {
          if (meshRegion === 0) {
            meshRegion = 1;
            allowedRegions = REGION_ADJ[meshRegion] || [meshRegion];
            thighReclassCount++;
          }
          if (lThProj > lHipBand) {
            allowedRegions = allowedRegions.filter(r => r !== 0);
            thighClampCount++;
          }
        }
      }
      // Right thigh
      const rh = P.r_hip, rk = P.r_knee;
      const rThLen = Math.sqrt((rk[0]-rh[0])**2 + (rk[1]-rh[1])**2 + (rk[2]-rh[2])**2);
      const rThD = [(rk[0]-rh[0])/rThLen, (rk[1]-rh[1])/rThLen, (rk[2]-rh[2])/rThLen];
      const THIGH_RADIUS_R = 0.25 * rThLen;
      const THIGH_RADIUS_HIP_R = 0.55 * rThLen; // wider near hip for crotch/groin
      const rHipBand = 0.15 * rThLen;
      const HIP_EXTEND_R = 0.12 * rThLen; // detect slightly above hip for crotch
      const rThProj = (x-rh[0])*rThD[0] + (y-rh[1])*rThD[1] + (z-rh[2])*rThD[2];
      if (rThProj > -HIP_EXTEND_R && rThProj < rThLen) {
        const projCR = Math.max(0, rThProj);
        const rpx = rh[0]+rThD[0]*projCR, rpy = rh[1]+rThD[1]*projCR, rpz = rh[2]+rThD[2]*projCR;
        const rThAxDist = Math.sqrt((x-rpx)*(x-rpx) + (y-rpy)*(y-rpy) + (z-rpz)*(z-rpz));
        // Tapered radius: wide near hip (crotch), normal further down thigh
        const hipFracR = Math.min(1, Math.max(0, 1 - rThProj / (0.3 * rThLen)));
        const effectiveRadiusR = THIGH_RADIUS_R + (THIGH_RADIUS_HIP_R - THIGH_RADIUS_R) * hipFracR;
        if (rThAxDist < effectiveRadiusR) {
          if (meshRegion === 0) {
            meshRegion = 2;
            allowedRegions = REGION_ADJ[meshRegion] || [meshRegion];
            thighReclassCount++;
          }
          if (rThProj > rHipBand) {
            allowedRegions = allowedRegions.filter(r => r !== 0);
            thighClampCount++;
          }
        }
      }
    }

    const candidateIndices = [];
    for (const r of allowedRegions) {
      const list = regionToIndices.get(r);
      if (list) candidateIndices.push(...list);
    }

    if (!candidateIndices.length) {
      for (let ci = 0; ci < nCage; ci++) candidateIndices.push(ci);
    }

    const bestD2 = new Float32Array(K);
    const bestI  = new Int32Array(K);
    for (let k = 0; k < K; k++) { bestD2[k] = 1e30; bestI[k] = candidateIndices[0] ?? 0; }

    for (const ci of candidateIndices) {
      const cx = cageRest[ci][0], cy = cageRest[ci][1], cz = cageRest[ci][2];
      const dx = x - cx, dy = y - cy, dz = z - cz;
      const d2 = dx*dx + dy*dy + dz*dz;

      let worst = 0;
      for (let k = 1; k < K; k++) if (bestD2[k] > bestD2[worst]) worst = k;

      if (d2 < bestD2[worst]) {
        bestD2[worst] = d2;
        bestI[worst]  = ci;
      }
    }

    let sum = 0;
    for (let k = 0; k < K; k++) {
      const d = Math.sqrt(bestD2[k]) + 1e-6;
      const ww = 1.0 / d;
      w[vi*K + k] = ww;
      idx[vi*K + k] = bestI[k];
      sum += ww;
    }
    for (let k = 0; k < K; k++) w[vi*K + k] /= sum;
  }

  if (reclassCount > 0 || clampCount > 0) {
    console.log('Arm geometry clamp: reclassified', reclassCount, 'torso→arm verts, clamped torso on', clampCount, 'arm verts');
  }
  if (thighReclassCount > 0 || thighClampCount > 0) {
    console.log('Thigh geometry clamp: reclassified', thighReclassCount, 'torso→thigh verts, clamped torso on', thighClampCount, 'thigh verts');
  }
  return { K, idx, w, regionId: regionIdLocal };
}

        return d && d[axis] && d[axis].length ? Math.min(...d[axis]) * deg : 0;
      };
      const maxVal = (bone, axis) => {
        const d = armMotionData[bone];
        return d && d[axis] && d[axis].length ? Math.max(...d[axis]) * deg : 0;
      };
      armBaseline = {
        // Shoulder: mean for symmetric swing
        l_shoulder: { x: mean('l_shoulder','x_rot'), y: mean('l_shoulder','y_rot'), z: mean('l_shoulder','z_rot') },
        r_shoulder: { x: mean('r_shoulder','x_rot'), y: mean('r_shoulder','y_rot'), z: mean('r_shoulder','z_rot') },
        // Elbow: min-bend baseline (left=min z_rot, right=max z_rot i.e. closest to 0)
        l_elbow:    { x: first('l_elbow','x_rot'), y: first('l_elbow','y_rot'), z: minVal('l_elbow','z_rot') },
        r_elbow:    { x: first('r_elbow','x_rot'), y: first('r_elbow','y_rot'), z: maxVal('r_elbow','z_rot') },
        // Wrist: same min-bend approach
        l_wrist:    { x: first('l_wrist','x_rot'), y: first('l_wrist','y_rot'), z: minVal('l_wrist','z_rot') },
        r_wrist:    { x: first('r_wrist','x_rot'), y: first('r_wrist','y_rot'), z: maxVal('r_wrist','z_rot') },
      };
    }

    // cage mesh wireframe
    const cgeom = new THREE.BufferGeometry();
    const carr = new Float32Array(cageRest.length*3);
    for(let i=0;i<cageRest.length;i++){
      carr[i*3]=cageRest[i][0];
      carr[i*3+1]=cageRest[i][1];
      carr[i*3+2]=cageRest[i][2];
    }
    cgeom.setAttribute('position', new THREE.BufferAttribute(carr,3));
    const cidx = new Uint32Array(cageFaces.length*3);
    for(let i=0;i<cageFaces.length;i++){
      cidx[i*3]=cageFaces[i][0];
      cidx[i*3+1]=cageFaces[i][1];
      cidx[i*3+2]=cageFaces[i][2];
    }
    cgeom.setIndex(new THREE.BufferAttribute(cidx,1));
    cageMesh = new THREE.Mesh(
      cgeom,
      new THREE.MeshBasicMaterial({color:0x4ecdc4, wireframe:true, transparent:true, opacity:0.5})
    );
    scene.add(cageMesh);

    statusEl.textContent = "Ready.";
  })().catch(err=>{
    console.error(err);
    statusEl.textContent = "Load failed. Make sure you're using a local server (not file://).";

  const FBX_BONE_MAP = [
    { suffix: 'Hips',          name: 'pelvis' },
    { suffix: 'Spine1',        name: 'chest', alt: 'Spine2' },
    { suffix: 'Neck',          name: 'neck' },
    { suffix: 'Head',          name: 'head' },
    { suffix: 'LeftArm',       name: 'l_shoulder' },
    { suffix: 'LeftForeArm',   name: 'l_elbow' },
    { suffix: 'LeftHand',      name: 'l_wrist' },
    { suffix: 'LeftHandMiddle1', name: 'l_mid_knuckle', alt: 'LeftHandMiddle2' },
    { suffix: 'LeftUpLeg',     name: 'l_hip' },
    { suffix: 'LeftLeg',       name: 'l_knee' },
    { suffix: 'LeftFoot',      name: 'l_ankle' },
    { suffix: 'LeftToeBase',   name: 'l_toe' },
    { suffix: 'RightArm',      name: 'r_shoulder' },
    { suffix: 'RightForeArm',  name: 'r_elbow' },
    { suffix: 'RightHand',     name: 'r_wrist' },
    { suffix: 'RightHandMiddle1', name: 'r_mid_knuckle', alt: 'RightHandMiddle2' },
    { suffix: 'RightUpLeg',    name: 'r_hip' },
    { suffix: 'RightLeg',      name: 'r_knee' },
    { suffix: 'RightFoot',     name: 'r_ankle' },
    { suffix: 'RightToeBase',  name: 'r_toe' },
  ];

  function matchFBXBone(boneName) {
    const lower = boneName.toLowerCase();
    const sorted = FBX_BONE_MAP.slice().sort((a, b) => b.suffix.length - a.suffix.length);
    for (const entry of sorted) {
      if (lower.endsWith(entry.suffix.toLowerCase())) return entry.name;
      if (entry.alt && lower.endsWith(entry.alt.toLowerCase())) return entry.name;
    }
    return null;
  }

  function extractPositionsFromFBX(arrayBuffer, filename) {
    statusEl.textContent = 'Parsing FBX...';

    const loader = new THREE.FBXLoader();
    let group;
    try {
      group = loader.parse(arrayBuffer, '');
    } catch (e) {
      statusEl.textContent = 'FBX parse error: ' + e.message;
      console.error(e);
      return;
    }

    // Find bones and map them
    const allBones = [];
    group.traverse(obj => { if (obj.isBone) allBones.push(obj); });
    if (!allBones.length) { statusEl.textContent = 'No bones in FBX!'; return; }

    const boneMapping = {};
    let hipsBone = null;
    for (const bone of allBones) {
      const mapped = matchFBXBone(bone.name);
      if (mapped) {
        boneMapping[mapped] = bone;
        if (mapped === 'pelvis') hipsBone = bone;
      }
    }

    if (!hipsBone) { statusEl.textContent = 'No Hips bone found!'; return; }

    const clip = group.animations && group.animations[0];
    if (!clip) { statusEl.textContent = 'No animation clip in FBX!'; return; }

    // ── Extract BIND POSE (T-pose) BEFORE animation overrides bone transforms ──
    // In Mixamo FBX, the skeleton's default pose is the T-pose (bind pose).
    // We need these positions as the "rest" reference for delta retargeting,
    // NOT frame 0 of the walking animation (which is mid-walk).
    group.updateMatrixWorld(true);
    const bindPoseRaw = {};  // hips-centered, X-mirrored, UNscaled
    const hipsBindPos = new THREE.Vector3();
    hipsBindPos.setFromMatrixPosition(hipsBone.matrixWorld);
    for (const jn of Object.keys(boneMapping)) {
      const bone = boneMapping[jn];
      const bp = new THREE.Vector3();
      bp.setFromMatrixPosition(bone.matrixWorld);
      bindPoseRaw[jn] = [
        -(bp.x - hipsBindPos.x),
        bp.y - hipsBindPos.y,
        bp.z - hipsBindPos.z
      ];
    }

    // ── Extract BIND POSE quaternions (world-space, X-mirrored) ──
    const bindQuatRaw = {};
    const tmpQuat = new THREE.Quaternion();
    for (const jn of Object.keys(boneMapping)) {
      const bone = boneMapping[jn];
      bone.getWorldQuaternion(tmpQuat);
      // X-mirror: reflection across YZ plane → negate Y and Z components
      bindQuatRaw[jn] = [tmpQuat.x, -tmpQuat.y, -tmpQuat.z, tmpQuat.w];
    }

    // Setup mixer
    const mixer = new THREE.AnimationMixer(group);
    const action = mixer.clipAction(clip);
    action.play();

    // Measure skeleton height for auto-scale (from animated frame 0)
    mixer.setTime(0);
    group.updateMatrixWorld(true);
    let minY = Infinity, maxY = -Infinity;
    group.traverse(obj => {
      if (obj.isBone) {
        const wp = new THREE.Vector3();
        obj.getWorldPosition(wp);
        minY = Math.min(minY, wp.y);
        maxY = Math.max(maxY, wp.y);
      }
    });
    const sourceHeight = maxY - minY;
    const targetHeight = meshHeight || 1.0;
    const scaleFactor = (sourceHeight > 0.01) ? (targetHeight / sourceHeight) : 1.0;
    console.log('FBX skeleton height:', sourceHeight.toFixed(1),
      '→ cage mesh height:', targetHeight.toFixed(3),
      '→ scale:', scaleFactor.toFixed(4));

    // Extract positions — use clip's native rate or minimum 60fps for fast animations
    let nativeFPS = 30;
    if (clip.tracks.length > 0 && clip.tracks[0].times.length > 1) {
      const dt = clip.tracks[0].times[1] - clip.tracks[0].times[0];
      if (dt > 0) nativeFPS = Math.round(1 / dt);
    }
    const fps = Math.max(60, nativeFPS);
    const duration = clip.duration;
    const frameCount = Math.ceil(duration * fps) + 1;
    const jointNames = Object.keys(boneMapping);

    const result = {
      duration, fps, frame_count: frameCount,
      rest_pose: {},
      rest_quats: {},
      root_positions: [],
      joints: {}
    };
    for (const jn of jointNames) {
      result.joints[jn] = { times: [], positions: [], quaternions: [] };
    }

    const worldPos = new THREE.Vector3();
    const hipsPos = new THREE.Vector3();

    for (let f = 0; f < frameCount; f++) {
      const t = Math.min(f / fps, duration);
      mixer.setTime(t);
      group.updateMatrixWorld(true);

      // Get hips TRANSLATION only (not full inverse matrix).
      // This removes root motion drift while keeping world-space orientation,
      // so direction vectors between joints stay in world coordinates.
      hipsPos.setFromMatrixPosition(hipsBone.matrixWorld);
      // Negate X: Mixamo has left=+X, cage has left=-X (mirrored convention)
      result.root_positions.push([
        -hipsPos.x * scaleFactor,
        hipsPos.y * scaleFactor,
        hipsPos.z * scaleFactor
      ]);

      for (const jn of jointNames) {
        const bone = boneMapping[jn];
        worldPos.setFromMatrixPosition(bone.matrixWorld);

        // Subtract hips translation only — positions are now
        // "hips-centered but world-oriented"
        // Negate X to match cage convention (Mixamo left=+X, cage left=-X)
        const lx = -(worldPos.x - hipsPos.x) * scaleFactor;
        const ly = (worldPos.y - hipsPos.y) * scaleFactor;
        const lz = (worldPos.z - hipsPos.z) * scaleFactor;

        result.joints[jn].times.push(+t.toFixed(6));
        result.joints[jn].positions.push([
          +lx.toFixed(6), +ly.toFixed(6), +lz.toFixed(6)
        ]);

        // World quaternion (X-mirrored: negate Y and Z)
        bone.getWorldQuaternion(tmpQuat);
        result.joints[jn].quaternions.push([
          +tmpQuat.x.toFixed(6), +(-tmpQuat.y).toFixed(6),
          +(-tmpQuat.z).toFixed(6), +tmpQuat.w.toFixed(6)
        ]);
      }
    }

    // Use BIND POSE (T-pose) as rest_pose, NOT frame 0 of the animation.
    // This is critical: frame 0 of a walking animation is mid-walk, giving tiny
    // deltas. T-pose gives meaningful animation deltas for retargeting.
    for (const jn of jointNames) {
      const raw = bindPoseRaw[jn];
      if (raw) {
        result.rest_pose[jn] = [
          +(raw[0] * scaleFactor).toFixed(6),
          +(raw[1] * scaleFactor).toFixed(6),
          +(raw[2] * scaleFactor).toFixed(6)
        ];
      } else {
        result.rest_pose[jn] = result.joints[jn].positions[0].slice();
      }
    }

    // Rest quaternions from bind pose (T-pose)
    for (const jn of jointNames) {
      result.rest_quats[jn] = bindQuatRaw[jn] ||
        (result.joints[jn].quaternions[0] || [0, 0, 0, 1]).slice();
    }

    // Diagnostic: compare bind pose (T-pose) vs frame 0 (animated) positions
    console.log('=== BIND POSE (T-pose) vs FRAME 0 (animated) ===');
    for (const jn of jointNames) {
      const bp = result.rest_pose[jn];
      const f0 = result.joints[jn].positions[0];
      const dist = Math.sqrt((bp[0]-f0[0])**2 + (bp[1]-f0[1])**2 + (bp[2]-f0[2])**2);
      if (dist > 0.001) {
        console.log('  ' + jn + ': T-pose=[' + bp.map(v=>(+v).toFixed(4)).join(',') +
          '] frame0=[' + f0.map(v=>(+v).toFixed(4)).join(',') + '] diff=' + dist.toFixed(4));
      }
    }

    // ── TWIST SANITY CHECK (frame 0 vs bind pose — should be small) ──
    // Position-swing residual method: twist at frame 0 measures how far from
    // T-pose the animation starts. For walking, this is a few degrees.
    // Values > 30° may indicate extraction issues.
    {
      const twistChecks = [
        ['l_shoulder', 'l_elbow'], ['r_shoulder', 'r_elbow'],
        ['l_elbow', 'l_wrist'], ['r_elbow', 'r_wrist'],
        ['l_hip', 'l_knee'], ['r_hip', 'r_knee']
      ];
      const parts = [];
      for (const [bone, child] of twistChecks) {
        const rq = result.rest_quats[bone];
        const fq = result.joints[bone] && result.joints[bone].quaternions[0];
        const rpB = result.rest_pose[bone], rpC = result.rest_pose[child];
        if (rq && fq && rpB && rpC) {
          const bd = vnorm(vsub(rpC, rpB));
          const tw = extractQuatTwist(rq, fq, bd) * 180 / Math.PI;
          parts.push(bone + '=' + tw.toFixed(1) + '°');
        }
      }
      console.log('[TWIST SANITY frame0] ' + parts.join(' '));
    }

    // ── SKELETON PROPORTION CHECK ──
    // Compare FBX rest-pose proportions vs cage pivot proportions.
    // No alignment/rotation is applied — only direction vectors are used from FBX data.
    if (P) {
      console.log('=== FBX vs CAGE PROPORTION CHECK (no alignment applied) ===');
      const cagePelvis = P.pelvis || [0, 0, 0];
      const checkJoints = ['l_hip','r_hip','l_shoulder','r_shoulder','l_knee','r_knee','l_elbow','r_elbow','neck','head'];
      for (const jn of checkJoints) {
        const fp = result.rest_pose[jn];
        const cp = P[jn];
        if (fp && cp) {
          // FBX: hips-centered distance from pelvis
          const fd = Math.sqrt(fp[0]*fp[0] + fp[1]*fp[1] + fp[2]*fp[2]);
          // Cage: distance from cage pelvis
          const cd = Math.sqrt((cp[0]-cagePelvis[0])**2 + (cp[1]-cagePelvis[1])**2 + (cp[2]-cagePelvis[2])**2);
          const ratio = cd > 0.001 ? (fd / cd).toFixed(3) : 'N/A';
          console.log('  ' + jn + ': fbxDist=' + fd.toFixed(4) + ' cageDist=' + cd.toFixed(4) + ' ratio=' + ratio +
            '  fbx=[' + fp.map(v=>v.toFixed(4)).join(',') + '] cage=[' + cp.map(v=>v.toFixed(4)).join(',') + ']');
        }
      }
      // Direction check: FBX l_shoulder direction vs cage l_shoulder direction
      const fpLS = result.rest_pose.l_shoulder, fpRS = result.rest_pose.r_shoulder;
      const cpLS = P.l_shoulder, cpRS = P.r_shoulder;
      if (fpLS && fpRS && cpLS && cpRS) {
        console.log('  FBX shoulder axis: [' + [(fpLS[0]-fpRS[0]).toFixed(4), (fpLS[1]-fpRS[1]).toFixed(4), (fpLS[2]-fpRS[2]).toFixed(4)].join(',') + ']');
        console.log('  Cage shoulder axis: [' + [(cpLS[0]-cpRS[0]).toFixed(4), (cpLS[1]-cpRS[1]).toFixed(4), (cpLS[2]-cpRS[2]).toFixed(4)].join(',') + ']');
      }
      // Forward direction: knee should be in +Z when leg is forward
      const fpLK = result.rest_pose.l_knee, fpLH = result.rest_pose.l_hip;
      if (fpLK && fpLH) {
        console.log('  FBX l_thigh rest dir: [' + [(fpLK[0]-fpLH[0]).toFixed(4), (fpLK[1]-fpLH[1]).toFixed(4), (fpLK[2]-fpLH[2]).toFixed(4)].join(',') + ']');
      }
      const cpLK = P.l_knee, cpLH = P.l_hip;
      if (cpLK && cpLH) {
        console.log('  Cage l_thigh rest dir: [' + [(cpLK[0]-cpLH[0]).toFixed(4), (cpLK[1]-cpLH[1]).toFixed(4), (cpLK[2]-cpLH[2]).toFixed(4)].join(',') + ']');
      }
    }

    // Compute leg proportion ratio: cage leg length / FBX leg length (in cage-scaled space).
    // Used to scale pelvis translation so bob/sway matches the cage's actual leg reach.
    fbxLegPropRatio = 1.0;
    if (P) {
      const rp = result.rest_pose;
      // FBX leg length (hip→knee + knee→ankle) in cage-scaled space
      let fbxLegLen = 0;
      if (rp.l_hip && rp.l_knee && rp.l_ankle) {
        fbxLegLen = Math.sqrt(rp.l_knee[0]**2+rp.l_knee[1]**2+rp.l_knee[2]**2) > 0.001
          ? Math.sqrt((rp.l_knee[0]-rp.l_hip[0])**2+(rp.l_knee[1]-rp.l_hip[1])**2+(rp.l_knee[2]-rp.l_hip[2])**2) +
            Math.sqrt((rp.l_ankle[0]-rp.l_knee[0])**2+(rp.l_ankle[1]-rp.l_knee[1])**2+(rp.l_ankle[2]-rp.l_knee[2])**2)
          : 0;
      }
      // Cage leg length (hip→knee + knee→ankle)
      let cageLegLen = 0;
      if (P.l_hip && P.l_knee && P.l_ankle) {
        cageLegLen = Math.sqrt((P.l_knee[0]-P.l_hip[0])**2+(P.l_knee[1]-P.l_hip[1])**2+(P.l_knee[2]-P.l_hip[2])**2) +
          Math.sqrt((P.l_ankle[0]-P.l_knee[0])**2+(P.l_ankle[1]-P.l_knee[1])**2+(P.l_ankle[2]-P.l_knee[2])**2);
      }
      if (fbxLegLen > 0.01 && cageLegLen > 0.01) {
        fbxLegPropRatio = cageLegLen / fbxLegLen;
        console.log('=== LEG PROPORTION RATIO ===');
        console.log('  FBX leg len (scaled): ' + fbxLegLen.toFixed(4) +
          '  Cage leg len: ' + cageLegLen.toFixed(4) +
          '  ratio: ' + fbxLegPropRatio.toFixed(3));
      }
    }
    // ── CROSS-BODY CONTACT DETECTION ──
    // Scan all frames for joints that come close together across different limbs.
    // This detects hand-to-elbow, hands-together (prayer), hand-to-knee, etc.
    {
      const contactPairs = [
        ['l_wrist', 'r_elbow', 'L wrist→R elbow'],
        ['r_wrist', 'l_elbow', 'R wrist→L elbow'],
        ['l_wrist', 'r_wrist', 'L wrist→R wrist'],
        ['l_wrist', 'l_knee',  'L wrist→L knee'],
        ['r_wrist', 'r_knee',  'R wrist→R knee'],
        ['l_wrist', 'r_knee',  'L wrist→R knee'],
        ['r_wrist', 'l_knee',  'R wrist→L knee'],
        ['l_elbow', 'r_elbow', 'L elbow→R elbow'],
      ];
      // Estimate body height from rest pose for threshold
      const rp = result.rest_pose;
      let bodyH = 1.0;
      if (rp.l_ankle && rp.head) {
        bodyH = Math.abs(rp.head[1] - rp.l_ankle[1]) || 1.0;
      }
      const contactThresh = bodyH * 0.12; // 12% of body height = ~20cm for 170cm character
      console.log('=== CROSS-BODY CONTACT ANALYSIS (threshold=' + contactThresh.toFixed(4) + ', bodyH=' + bodyH.toFixed(4) + ') ===');
      let anyContact = false;
      for (const [jA, jB, label] of contactPairs) {
        const dataA = result.joints[jA], dataB = result.joints[jB];
        if (!dataA || !dataB) continue;
        let minDist = Infinity, minFrame = 0;
        const nf = dataA.positions.length;
        let contactFrames = 0;
        for (let f = 0; f < nf; f++) {
          const pA = dataA.positions[f], pB = dataB.positions[f];
          const d = Math.sqrt((pA[0]-pB[0])**2 + (pA[1]-pB[1])**2 + (pA[2]-pB[2])**2);
          if (d < minDist) { minDist = d; minFrame = f; }
          if (d < contactThresh) contactFrames++;
        }
        if (minDist < contactThresh * 2) { // report if within 2x threshold
          const pct = (contactFrames / nf * 100).toFixed(0);
          const contact = minDist < contactThresh;
          console.log('  ' + (contact ? 'CONTACT' : 'near') + ' ' + label +
            ': minDist=' + minDist.toFixed(4) + ' at frame ' + minFrame +
            '/' + nf + ' (' + pct + '% of frames in contact)');
          if (contact) anyContact = true;
        }
      }
      if (!anyContact) console.log('  No cross-body contacts detected');
    }

    // Reset diagnostics
    window._hipSmoothDiag = false;
    window._hipAnchorDiag = false;
    window._stretchDiagDone = false;

    // Apply to app
    jointPosData = result;
    cycleDuration = result.duration;
    time = 0;
    window._posDiagDone = false;  // reset diagnostic for fresh output
    window._kabschDiagDone = false;
    window._armAnalyticalDiagDone = false;
    window._twistDiagCount = 0;
    window._meshVertCheckDone = false;  // re-check mesh after FBX load
    window._armCollisionDiag = false;   // re-report collision with FBX data
    window._pivotDiagDone = false;      // re-report Kabsch vs analytical shin comparison
    // Reset temporal coherence state for new animation
    for (const k of Object.keys(prevBendAxis)) prevBendAxis[k] = null;

    const mappedNames = jointNames.join(', ');
    const info = filename + ': ' + frameCount + ' frames, ' +
      jointNames.length + ' joints, ' + duration.toFixed(2) + 's (scale ' + scaleFactor.toFixed(3) + ')';
    $('fbxInfo').textContent = info;
    statusEl.textContent = 'FBX loaded — position-based driving active';
    // Auto-enable DQS for FBX mode — LBS causes volume collapse at
    // body boundaries (crotch, shoulders) when diverging transforms blend.
    // DQS preserves volume by averaging quaternions instead of matrices.
    $('useDQS').checked = true;
    console.log('FBX extracted:', info);
    console.log('Mapped joints:', mappedNames);
  }

  // Drag-drop handlers
  const dropOverlay = $('dropOverlay');
  let dragCounter = 0;
  document.addEventListener('dragenter', e => {
    e.preventDefault();
    dragCounter++;
    dropOverlay.classList.add('active');
  });
  document.addEventListener('dragover', e => { e.preventDefault(); });
  document.addEventListener('dragleave', e => {
    dragCounter--;
    if (dragCounter <= 0) { dragCounter = 0; dropOverlay.classList.remove('active'); }
  });
  document.addEventListener('drop', e => {
    e.preventDefault();
    dragCounter = 0;
    dropOverlay.classList.remove('active');
    const file = e.dataTransfer.files[0];
    if (file && file.name.toLowerCase().endsWith('.fbx')) {
      const reader = new FileReader();
      reader.onload = () => extractPositionsFromFBX(reader.result, file.name);
      reader.readAsArrayBuffer(file);
    }
  });

  // File picker handler
  $('fbxFile').addEventListener('change', e => {
    const file = e.target.files[0];
    if (file) {
      const reader = new FileReader();
      reader.onload = () => extractPositionsFromFBX(reader.result, file.name);
      reader.readAsArrayBuffer(file);
    }
  });
